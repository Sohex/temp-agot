# Compatibility patch for AGOT Dragonlord Regime + Ancient Birthright.
#
# Dragonlord Regime exposes game rules controlling the culture/religion groups
# eligible to become Dragonlords, but its government-adoption decision hardcodes
# High Valyrian + Fourteen Flames. Reuse the mod's own admission-rule trigger so
# the founder follows those configured game rules too.

adopt_dragonlord_oligarchy_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_dragons.dds"
    }
    decision_group_type = major
    desc = adopt_dragonlord_oligarchy_decision_desc
    selection_tooltip = adopt_dragonlord_oligarchy_decision_tooltip
    confirm_text = adopt_dragonlord_oligarchy_decision_confirm

    ai_check_interval_by_tier = {
        barony = 0
        county = 0
        duchy = 0
        kingdom = 0
        empire = 0
        hegemony = 0
    }

    is_shown = {
        NOT = { government_has_flag = government_is_dragonlord_oligarchy }
        highest_held_title_tier >= tier_empire
        dragonlord_meets_family_admission_rules = yes
    }

    is_valid = {
        has_ep3_dlc_trigger = yes
        highest_held_title_tier >= tier_empire
        dragonlord_meets_family_admission_rules = yes
    }

    effect = {
        hidden_effect = {
            add_character_flag = dragonlord_oligarchy_adopted
            change_to_administrative_holding_effect = yes
            change_government = dragonlord_oligarchy_government
            dragonlord_refresh_bureaucracy_opinion_penalty_effect = yes
            primary_title = {
                set_title_name = dragonlord_valyrian_dragonlord_council
                set_definitive_form = yes
            }
            if = {
                limit = {
                    NOT = { has_realm_law = dragonlord_council_election_succession_law }
                }
                add_realm_law_skip_effects = dragonlord_council_election_succession_law
            }
            every_vassal_or_below = {
                limit = {
                    highest_held_title_tier >= tier_county
                }
                change_to_administrative_holding_effect = yes
                change_government = dragonlord_oligarchy_government
            }

            dragonlord_enforce_single_high_title_effect = yes

            dragonlord_initialize_noble_laws_effect = yes
            dragonlord_establish_family_titles_effect = yes
            add_character_flag = dragonlord_final_hierarchy_family_selection_migrated
            dragonlord_restore_family_primary_titles_effect = yes

            dragonlord_recalculate_fourteen_dragonlords_effect = yes
            dragonlord_recalculate_tamed_dragon_influence_effect = yes
            dragonlord_establish_default_provincial_laws_effect = yes
        }
        custom_tooltip = adopt_dragonlord_oligarchy_decision_effect_tooltip
    }

    ai_potential = { always = no }
    ai_will_do = { base = 0 }
}


# Save repair: Dragonlord Regime versions without this compatibility patch
# destroy Ancient Birthright's kingdom-tier Dragon Throne during adoption.
# This restores only the landless HoF title, bypassing the normal kingdom-title
# creation gate without re-awarding Ancient Birthright's one-time bonuses.
dragonlord_ab_restore_dragon_throne_decision = {
    picture = { reference = "gfx/interface/illustrations/decisions/decision_major_religion.dds" }
    decision_group_type = major
    desc = dragonlord_ab_restore_dragon_throne_decision_desc
    selection_tooltip = dragonlord_ab_restore_dragon_throne_decision_tooltip
    confirm_text = dragonlord_ab_restore_dragon_throne_decision_confirm

    is_shown = {
        government_has_flag = government_is_dragonlord_oligarchy
        faith = faith:imperial_valyrian
        NOT = { exists = title:k_dragon_throne.holder }
        is_target_in_global_variable_list = {
            name = unavailable_unique_decisions
            target = flag:flag_proclaimed_fire_and_blood
        }
    }

    is_valid = {
        top_liege = this
        is_landed = yes
        is_capable_adult = yes
        is_imprisoned = no
    }

    effect = {
        hidden_effect = {
            create_title_and_vassal_change = {
                type = returned
                save_scope_as = dragonlord_ab_dragon_throne_change
                add_claim_on_loss = no
            }
            title:k_dragon_throne = {
                change_title_holder = {
                    holder = root
                    change = scope:dragonlord_ab_dragon_throne_change
                }
            }
            resolve_title_and_vassal_change = scope:dragonlord_ab_dragon_throne_change
            if = {
                limit = { NOT = { has_trait = dragon_emperor } }
                add_trait = dragon_emperor
            }
        }
        custom_tooltip = dragonlord_ab_restore_dragon_throne_decision_effect_tooltip
    }

    ai_potential = { always = no }
    ai_will_do = { base = 0 }
}
