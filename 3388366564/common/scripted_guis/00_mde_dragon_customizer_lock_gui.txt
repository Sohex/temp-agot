
mde_dc_lock_primary_color_hue = {
	scope = character
	is_shown = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
			has_variable = gene_dragon_primary_color_hue_lock
		}
	}
	is_valid = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
		}
	}
	effect = {
		every_in_global_list = {
			variable = gl_dragon_variable_storage
			limit = { var:dragon_id ?= root.var:editing_dragon }
			save_scope_as = story
		}
		scope:story = {
			if = {
				limit = { has_variable = gene_dragon_primary_color_hue_lock }
				remove_variable = gene_dragon_primary_color_hue_lock
			}
			else = {
				set_variable = gene_dragon_primary_color_hue_lock
			}
		}
	}
}

	scope = character
	is_valid = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
		}
	}
	effect = {
		agot_dragon_customizer_clear_fire_effect = yes
		if = {
			limit = {
				agot_has_dragon_storage_system_global_list_editor_trigger = yes
			}
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = { var:dragon_id ?= root.var:editing_dragon }
				save_scope_as = story
			}
			scope:story = {
				mde_dc_randomize_effect = { EGG_COLOR = no }
			}
			agot_change_dragon_color_flags = { DRAGON = root.var:editing_dragon }
		}
	}
}
