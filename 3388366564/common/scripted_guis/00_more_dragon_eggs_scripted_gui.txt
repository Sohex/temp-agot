mde_view_dragons_in_dragonpit = {
	scope = title

	is_shown = {
		OR = {
			AND = {
				is_landless_type_title = no
				any_county_province = {
					OR = {
						has_building_or_higher = generic_dragon_pit_01
						has_building_or_higher = dragonpit_01 # Historical Version
						has_building = dragonpit_ruins_03 # Fully Restored Ruin Version
						has_building = agot_dragonmont_01 # Dragonmont
					}
				}
			}
			AND = {
				is_landless_type_title = yes
				holder.domicile ?= {
					has_domicile_parameter = camp_unlocks_dragonkeeper
				}
			}
		}
	}

	effect = {
		if = { # MDE Added
			limit = { has_variable = hdk }
			remove_variable = hdk
		}
		if = {
			limit = {
				scope:dp_county = {
					is_landless_type_title = yes
				}
			}
			scope:dp_county.holder.court_position:head_dragonkeeper_court_position ?= {
				root = {
					set_variable = {
						name = hdk
						value = prev
					}
				}
			}
		}
		else = {
			scope:dp_county.county.holder.court_position:head_dragonkeeper_court_position ?= {
				root = {
					set_variable = {
						name = hdk
						value = prev
					}
				}	
			}
		}
		set_variable = {
			name = viewing_county
			value = scope:dp_county
		}
	}	
}

mde_any_dragon_can_be_moved_into_pit = {
	scope = character
	is_shown = {
		exists = scope:dp_county
		scope:dp_county.holder = {
			OR = {
				this = root
				AND = {
					house ?= root.house
					target_is_liege_or_above = root
				}
                
			}
		}
		any_in_global_list = {
			variable = living_dragons
			has_character_flag = owned_dragon
			root = {
				allow_moving_of_dragon_trigger = { DRAGON = prev }
			}
			trigger_if = {
				limit = {
					has_variable = pitted_dragon_location
				}
				NOT = {
					var:pitted_dragon_location = scope:dp_county
				}
			}
			count > 0
		}
	}

	effect = {
		every_in_global_list = {
			variable = living_dragons
			limit = {
				root = {
					allow_moving_of_dragon_trigger = { DRAGON = prev }
				}
				trigger_if = {
					limit = {
						has_variable = pitted_dragon_location
					}
					NOT = {
						var:pitted_dragon_location = scope:dp_county
					}
				}
			}
			if = { # MDE Added
				limit = { this = { has_variable = pitted_dragon_location } }
				this = { save_scope_as = mde_moving_dragon }
				agot_remove_from_dragonpit_skip_county_effects = { DRAGON = scope:mde_moving_dragon }
			}
			agot_send_to_dragonpit_no_event = {
				DRAGON = this
				DRAGONPIT_COUNTY = scope:dp_county
			}
		}
	}
}
