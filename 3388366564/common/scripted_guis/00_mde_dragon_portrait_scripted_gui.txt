mde_dragon_portrait_size_baby = {
	scope = character
	is_shown = {
		any_relation = {
			type = agot_dragon
			is_alive = yes
			dragon_size < mde_dragon_portrait_normal_size_value
		}
	}
}

mde_dragon_portrait_size_normal = {
	scope = character
	is_shown = {
		any_relation = {
			type = agot_dragon
			is_alive = yes
			NOR = {
				dragon_size < mde_dragon_portrait_normal_size_value
				dragon_size >= mde_dragon_portrait_giant_size_value
			}
		}
	}
}

mde_dragon_portrait_size_giant = {

		}
	}
}


mde_player_has_living_dragon = {
	scope = character
	is_shown = {
		any_relation = {
			type = agot_dragon
			is_alive = yes
		}
	}
}

mde_dragon_portrait_toggle = {
	scope = character

	is_shown = {
		mde_dragon_portrait_shown_trigger = yes
	}

	effect = {
		if = {
			limit = { mde_dragon_portrait_shown_trigger = yes }
			mde_hide_dragon_portrait_effect = yes
		}
		else = {
			mde_show_dragon_portrait_effect = yes
		}
	}
}

mde_dragon_animation_idle = {
	scope = character

	is_shown = {
		mde_dragon_portrait_shown_trigger = yes
		any_relation = {
			type = agot_dragon
			is_alive = yes
			NOR = {
				has_character_flag = mde_dragon_roar
				has_character_flag = mde_dragon_roar_fire
			}
		}
	}

	is_valid = {
		mde_dragon_portrait_shown_trigger = yes
		any_relation = {
			type = agot_dragon
			is_alive = yes
		}
	}

	effect = {
		mde_toggle_dragon_animation_effect = { FLAG = mde_dragon_idle }
	}
}
