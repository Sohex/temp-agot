	is_valid = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
			NOT = {
				has_variable = gene_dragon_brow_width_lock
			}
		}
	}
	effect = {
		if = {
			limit = {
				agot_has_dragon_storage_system_global_list_editor_trigger = yes
			}
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = {
					var:dragon_id ?= root.var:editing_dragon
				}
				save_scope_as = story
			}
			scope:story = {
				set_variable = gene_dragon_brow_width_lock
			}
		}
	}
}
