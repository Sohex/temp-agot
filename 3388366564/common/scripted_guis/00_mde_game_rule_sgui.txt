player_map_egg_val_cycle = {
	scope = character

	effect = {
		if = {
			limit = { NOT = { has_global_variable = mde_egg_map_spawn_player } }
			set_global_variable = { name = mde_egg_map_spawn_player value = 0 }
		}
		if = {
			limit = { global_var:mde_egg_map_spawn_player < 11 }
			change_global_variable = { name = mde_egg_map_spawn_player add = 1 }
		}
		if = {
			limit = { global_var:mde_egg_map_spawn_player = 11 }
			change_global_variable = { name = mde_egg_map_spawn_player add = -11 }
		}
	}
}
