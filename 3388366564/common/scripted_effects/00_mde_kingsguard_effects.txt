
mde_build_current_kingsguard_list_effect = {
	clear_global_variable_list = mde_current_kingsguard
	every_living_character = {
		limit = {
			has_trait = kingsguard
			NOT = { has_variable = lord_commander }
			NOT = { has_title = title:d_kingsguard }
		}
		add_to_global_variable_list = {
			name = mde_current_kingsguard
			target = this
		}
	}
}

mde_update_kingsguard_candidates_effect = {
	if = {
		limit = {
			exists = title:h_the_iron_throne.holder
		}
		clear_global_variable_list = mde_potential_kingsguard_reorder

		every_in_global_list = {
			variable = mde_potential_kingsguard
			limit = {
				is_alive = yes
				mde_kingsguard_candidate_trigger = yes
			}
			add_to_global_variable_list = {
				name = mde_potential_kingsguard_reorder
				target = this
			}
		}

		every_living_character = {
			limit = {
				mde_kingsguard_candidate_trigger = yes
				NOT = {
					is_target_in_global_variable_list = {
						name = mde_potential_kingsguard_reorder
						target = this
					}
				}
			}
			add_to_global_variable_list = {
				name = mde_potential_kingsguard_reorder
				target = this
			}
		}

		clear_global_variable_list = mde_potential_kingsguard
		mde_add_kingsguard_candidate_tier_effect = { TIER = 1 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 2 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 3 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 4 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 5 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 6 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 7 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 8 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 9 }
		mde_add_kingsguard_candidate_tier_effect = { TIER = 10 }

		clear_global_variable_list = mde_potential_kingsguard_reorder
	}
	else = {
		clear_global_variable_list = mde_potential_kingsguard
	}
}

mde_add_kingsguard_candidate_tier_effect = {
	every_in_global_list = {
		variable = mde_potential_kingsguard_reorder
		limit = {
			mde_kingsguard_candidate_tier_$TIER$_trigger = yes
			NOT = {
				is_target_in_global_variable_list = {
					name = mde_potential_kingsguard
					target = this
				}
			}
		}
		add_to_global_variable_list = {
			name = mde_potential_kingsguard
			target = this
		}
	}
}
