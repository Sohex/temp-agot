### MDE version of agot_set_dragon_animation_flag_effect, driven by the MDE menu's dragon portrait buttons.
mde_set_dragon_animation = {
	mde_clear_dragon_animation_flags = yes
	add_character_flag = $FLAG$
}

mde_clear_dragon_animation_flags = {
	if = {
		limit = { has_character_flag = mde_dragon_idle }
		remove_character_flag = mde_dragon_idle
	}
	if = {
		limit = { has_character_flag = mde_dragon_roar }
		remove_character_flag = mde_dragon_roar
	}
	if = {
		limit = { has_character_flag = mde_dragon_roar_fire }
		remove_character_flag = mde_dragon_roar_fire
	}
}

mde_toggle_dragon_animation_effect = {
	every_relation = {
		type = agot_dragon
		limit = { is_alive = yes }
		if = {
			limit = { has_character_flag = $FLAG$ }
			mde_set_dragon_animation = { FLAG = mde_dragon_idle }
		}
		else = {
			mde_set_dragon_animation = { FLAG = $FLAG$ }
		}
	}
}

mde_show_dragon_portrait_effect = {
	set_variable = show_dragon_portrait
}

mde_hide_dragon_portrait_effect = {
	if = {
		limit = { has_global_variable = global_show_dragon_portrait }
		remove_global_variable = global_show_dragon_portrait
		every_player = {
			limit = { NOT = { this = root } }
			set_variable = show_dragon_portrait
		}
	}
	if = {
		limit = { has_variable = show_dragon_portrait }
		remove_variable = show_dragon_portrait
	}
	every_relation = {
		type = agot_dragon
		mde_clear_dragon_animation_flags = yes
	}
}
