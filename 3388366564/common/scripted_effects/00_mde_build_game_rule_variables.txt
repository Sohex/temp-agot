		limit = { global_var:$VAR$ = 0 }
		set_global_variable = { name = $VAR$ value = 1 }
	}
	else = {
		set_global_variable = { name = $VAR$ value = 0 }
	}
}

mde_cycle_game_rule_effect = {
	if = {
		limit = {
			has_global_variable = $VAR$
			global_var:$VAR$ < $MAX$
		}
		change_global_variable = { name = $VAR$ add = 1 }
	}
	else = {
		set_global_variable = { name = $VAR$ value = 0 }
	}
}
