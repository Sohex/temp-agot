### Bottom-right MDE buttons (dragon_management_buttons_widget.gui): dragonpit, dragon, egg, kingsguard, valyrian_steel, dragonblood, artifact_market, menu.

mde_set_menu_button_effect = {
	if = {
		limit = { has_multiple_players = no }
		if = {
			limit = { always = $ON$ }
			set_global_variable = mde_$BUTTON$_button_on
		}
		else_if = {
			limit = { has_global_variable = mde_$BUTTON$_button_on }
			remove_global_variable = mde_$BUTTON$_button_on
		}
	}
	else = {
		if = {
			limit = {
				NOT = {
					any_owned_story = { story_type = story_menu_buttons_variable_storage }
				}
			}
			create_story = {
				type = story_menu_buttons_variable_storage
				save_scope_as = mde_menu_button_story
			}
		}
		else = {
			random_owned_story = {
				limit = { story_type = story_menu_buttons_variable_storage }
				save_scope_as = mde_menu_button_story
			}
		}
		scope:mde_menu_button_story = {
			if = {
				limit = { always = $ON$ }
				set_variable = mde_$BUTTON$_button_toggle_on
			}
			else_if = {
				limit = { has_variable = mde_$BUTTON$_button_toggle_on }
				remove_variable = mde_$BUTTON$_button_toggle_on
			}
		}
	}
}

mde_toggle_menu_button_effect = {
	mde_refresh_menu_button_availability_effect = yes
	if = {
		limit = { mde_menu_button_on_trigger = { BUTTON = $BUTTON$ } }
		mde_set_menu_button_effect = { BUTTON = $BUTTON$ ON = no }
	}
	else = {
		mde_set_menu_button_effect = { BUTTON = $BUTTON$ ON = yes }
	}
}

mde_set_all_menu_buttons_effect = {
	mde_set_menu_button_effect = { BUTTON = dragonpit ON = $DRAGONPIT$ }
	mde_set_menu_button_effect = { BUTTON = dragon ON = $DRAGON$ }
	mde_set_menu_button_effect = { BUTTON = egg ON = $EGG$ }
	mde_set_menu_button_effect = { BUTTON = kingsguard ON = $KINGSGUARD$ }
	mde_set_menu_button_effect = { BUTTON = valyrian_steel ON = $VALYRIAN_STEEL$ }
	mde_set_menu_button_effect = { BUTTON = dragonblood ON = $DRAGONBLOOD$ }
	mde_set_menu_button_effect = { BUTTON = artifact_market ON = $ARTIFACT_MARKET$ }
	mde_set_menu_button_effect = { BUTTON = menu ON = $MENU$ }
}

mde_menu_buttons_standard_effect = {
	mde_refresh_menu_button_availability_effect = yes
	mde_set_all_menu_buttons_effect = { DRAGONPIT = yes DRAGON = yes EGG = yes KINGSGUARD = no VALYRIAN_STEEL = no DRAGONBLOOD = no ARTIFACT_MARKET = no MENU = yes }
}

mde_apply_menu_buttons_game_rule_effect = {
	mde_refresh_menu_button_availability_effect = yes
	if = {
		limit = { has_game_rule = mde_buttons_off }
		mde_set_all_menu_buttons_effect = { DRAGONPIT = no DRAGON = no EGG = no KINGSGUARD = no VALYRIAN_STEEL = no DRAGONBLOOD = no ARTIFACT_MARKET = no MENU = no }
	}
	else_if = {
		limit = { has_game_rule = mde_buttons_all }
		mde_set_all_menu_buttons_effect = { DRAGONPIT = yes DRAGON = yes EGG = yes KINGSGUARD = yes VALYRIAN_STEEL = yes DRAGONBLOOD = yes ARTIFACT_MARKET = yes MENU = yes }
	}
	else = {
		mde_menu_buttons_standard_effect = yes
	}
}

mde_refresh_menu_button_availability_effect = {
	if = {
		limit = { mde_live_dragonpits = yes }
		set_global_variable = mde_any_live_dragonpits
	}
	else_if = {
		limit = { has_global_variable = mde_any_live_dragonpits }
		remove_global_variable = mde_any_live_dragonpits
	}
	if = {
		limit = {
			any_artifact = { has_variable = dragon_egg }
		}
		set_global_variable = mde_any_dragon_eggs
	}
	else_if = {
		limit = { has_global_variable = mde_any_dragon_eggs }
		remove_global_variable = mde_any_dragon_eggs
	}
	if = {
		limit = {
			any_artifact = { has_variable = valyrian_steel }
		}
		set_global_variable = mde_any_valyrian_steel
	}
	else_if = {
		limit = { has_global_variable = mde_any_valyrian_steel }
		remove_global_variable = mde_any_valyrian_steel
	}
}
