mde_create_dragoncoop_scripted_effect = {
    $DRAGON_PIT_OWNER$ = {
		#  Adds a variable to the county holding the dragonpit so that the decision cannot be taken again
		if = {
			limit = {
                domicile ?= {
                    NOT = {
                        has_variable = has_dragonkeeper_order
                    }
                    has_domicile_parameter = camp_unlocks_dragonkeeper
                }
			}
			trigger_event = agot_dragonkeepers.0001  # Creates the Head Dragonkeeper Character, Provides a Little Bit of Context to Player # MDE Modified
            domicile = {
                set_variable = { name = has_dragonkeeper_order }
            }
            set_global_variable = mde_any_live_dragonpits # MDE Added
		}
	}
}
