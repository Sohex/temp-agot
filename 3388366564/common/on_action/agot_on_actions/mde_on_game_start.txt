fix_my_errors_please = {
    effect = {
        selected_set_error_fixer = yes
    }
}

apply_dragon_portrait_variable_on_start = {
    effect = {
        if = {
            limit = {
                has_game_rule = show_dragon_portrait_on_start_yes
            }
            set_global_variable = global_show_dragon_portrait
        }
    }
}

turn_on_dragon_management_buttons = {
    effect = {
        every_player = {
            mde_apply_menu_buttons_game_rule_effect = yes
        }
    }
}

mde_delay_egg_toasts_on_action = {
    effect = {
        random_player = {
            trigger_event = {
                on_action = mde_enable_egg_toasts_on_action
                days = 1
            }
        }
    }
}

mde_enable_egg_toasts_on_action = {
    effect = {
        set_global_variable = mde_egg_toasts_active
    }
}
