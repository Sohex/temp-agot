# Odds multiplier for the map spawn setting, as 299 / (target denominator - 1).
# whole accumulated weight scales. On a standard 299-weight nothing branch this
# lands exactly on the labelled rate. Offsets 0 and 10 never reach a random_list.
mde_egg_map_spawn_factor_player = {
	if = {
		limit = { NOT = { has_global_variable = mde_egg_map_spawn_player } }
		value = 1
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 1 }
		value = 0.24937
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 2 }
		value = 0.33259
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 3 }
		value = 0.49917
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 5 }
		value = 2.00671
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 6 }
		value = 6.10204
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 7 }
		value = 12.45833
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 8 }
		value = 74.75	# 1/5
	}
	else_if = {
		limit = { global_var:mde_egg_map_spawn_player = 9 }
		value = 299	# 1/2
	}
	else = {
		value = 1
	}
}

mde_cradle_hatch_likelihood_setting = {
	value = 3
	if = {
		limit = {
			OR = {
				is_ai = no
				court_owner ?= { is_ai = no }
				any_close_family_member = { is_ai = no }
			}
		}
		if = {
			limit = { has_global_variable = mde_cradle_hatch_likelihood_player }
			value = global_var:mde_cradle_hatch_likelihood_player
		}
	}
	else_if = {
		limit = { has_global_variable = mde_cradle_hatch_likelihood_ai }
		value = global_var:mde_cradle_hatch_likelihood_ai
	}
}

# How many egg availability cutoffs the campaign's start date is past.
# Each dragonlore event requires game_start_date > X, so an egg's in-game
# toggle is only relevant once this has reached that egg's step.
mde_game_start_era = {
	value = 0
	if = { limit = { game_start_date > 7900.1.1 } value = 1 }	# Pythax opens (mde_dragon_eggs_events.0014)
	if = { limit = { game_start_date > 7903.1.1 } value = 2 }	# Vassarion opens (mde_dragon_eggs_events.0016)
	if = { limit = { game_start_date > 8010.1.1 } value = 3 }	# Meraxes opens (mde_dragon_eggs_events.0012)
	if = { limit = { game_start_date > 8130.5.1 } value = 4 }	# Moondancer (mde_dragon_eggs_events.0008) and Vhagar (agot_dragon_hatching_dragonlore.0005, real cutoff 8130.1.1 - no bookmark starts in between)
	if = { limit = { game_start_date > 8142.1.1 } value = 5 }	# Cannibal and Sheepstealer open (mde_dragon_eggs_events.0001 / .0002)
}

		limit = { global_var:mde_hatch_success_chance = 4 }

	}

		limit = { global_var:mde_hatch_success_chance = 5 }
