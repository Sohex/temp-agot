on_birth_child = {
    on_actions = {
        on_birth_child_exotic_beauty
        on_birth_child_exotic_beauty_replace
    }
}

on_birth_child_exotic_beauty = {
   effect = {
       if = {
           limit = {
                culture = { 
                        has_cultural_parameter = unlock_exotic_beauty 
                }
           }
           random_list = {
               20 = { add_trait = exotic_beauty_good_1 }
               15 = { add_trait = exotic_beauty_good_2 }
               5 = { add_trait = exotic_beauty_good_3 }
               2 = { add_trait = exotic_beauty_good_4 }
               58 = {}
           }
       }
   }
} 

on_birth_child_exotic_beauty_replace = {
   effect = {
         if = {
              limit = {
                    AND = {
                         culture = {
                               has_cultural_parameter = unlock_exotic_beauty
                         }
		         OR = {
		               scope:real_father = { 
                                       has_trait = beauty_good_2
				}
				scope:mother = { 
                                       has_trait = beauty_good_2 
				}
			 }
                    }
              }
              random = {
                       chance = 20
                       remove_trait = beauty_good_1
                       remove_trait = beauty_good_2
                       remove_trait = beauty_good_3
              }
         }                 
         if = {
              limit = {
                    AND = {
                         culture = {
                               has_cultural_parameter = unlock_exotic_beauty
                         }
		         OR = {
		               scope:real_father = { 
                                       has_trait = beauty_good_2
				}
				scope:mother = { 
                                       has_trait = beauty_good_2 
				}
			 }
                    }
              }
              random_list = {
                  30 = {
                         remove_trait = beauty_good_1
                         remove_trait = beauty_good_2
                         remove_trait = beauty_good_3
                         add_trait = exotic_beauty_good_1
                  }
                  20 = { 
                         remove_trait = beauty_good_1
                         remove_trait = beauty_good_2
                         remove_trait = beauty_good_3
                         add_trait = exotic_beauty_good_2
                  }
                  50 = {
                  }
              }
          }
          if = {
               limit = {
                     AND = {
                          culture = {
                                has_cultural_parameter = unlock_exotic_beauty
                          }
		          OR = {
		                scope:real_father = { 
                                        has_trait = beauty_good_3
				 }
				 scope:mother = { 
                                        has_trait = beauty_good_3 
				 }
			  }
                     }
               }
               random_list = {
                   40 = {
                          remove_trait = beauty_good_1
                          remove_trait = beauty_good_2
                          remove_trait = beauty_good_3
                          add_trait = exotic_beauty_good_1
                   }
                   30 = { 
                          remove_trait = beauty_good_1
                          remove_trait = beauty_good_2
                          remove_trait = beauty_good_3
                          add_trait = exotic_beauty_good_2
                   }
                   20 = {
                          remove_trait = beauty_good_1
                          remove_trait = beauty_good_2
                          remove_trait = beauty_good_3
                          add_trait = exotic_beauty_good_3
                   }
                   10 = {
                   }
               }
           }

    }
}
