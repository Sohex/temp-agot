restore_dragon_throne_scripted_effect = {
	add_piety = major_piety_gain
 	save_scope_as = reformer
	root.dynasty = {
		add_dynasty_prestige = 10000
		add_dynasty_prestige_level = 1
	}
	create_title_and_vassal_change = {
		type = returned
		save_scope_as = change
		add_claim_on_loss = no
	}
	title:k_dragon_throne = {
		save_scope_as = dragon_throne
		change_title_holder = {
			holder = scope:reformer
			change = scope:change
		}
	}
	resolve_title_and_vassal_change = scope:change
	faith:imperial_valyrian = {
		change_fervor = {
			value = 15
			desc = fervor_gain_restored_dragon_throne
		}
	}
}
