namespace = ab_electorate_events

#Yontaros
ab_electorate_events.0001 = {
                        type = character_event
                        hidden = yes

                        trigger = {
                                AND = {
                                      has_title = title:d_jarqanor
                                      has_title = title:c_jarqanor

                                       OR = {
                                            religion = religion:the_flames_religion
                                            religion = religion:maegor_religion
                                       }
                                       culture = {
                                            has_cultural_parameter = restore_old_titles
                                       }
                                       NOT = { 
                                            has_trait = guardian_yontaryos 
                                       } 
                                }
                         }
immediate = { add_trait = guardian_yontaryos }
}
