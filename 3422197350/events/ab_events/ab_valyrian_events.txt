 # on_new_holder: New Emperor gets the Dragon Emperor Trait.
 ab_valyrian_events.0002 = {
           type = character_event
           title = ab_valyrian_events.0002.title
           desc = ab_valyrian_events.0002.desc
           theme = realm
           left_portrait = {
                   character = root
                   animation = personality_bold
           }

           trigger = {
                    AND = {
                          culture = {
                                     has_cultural_tradition = tradition_rule_of_the_dragon_emperor
                          }
                          scope:title = title:k_dragon_throne 
                    }
                    NOT = { has_trait = dragon_emperor } 
           }

           option = {
                   name = ab_valyrian_events.0002.a
                   add_trait = dragon_emperor
           }
}

# If old holder is still alive, he will lose the trait.
ab_valyrian_events.0003 = {
          type = character_event
          title = ab_valyrian_events.0003.title
          desc = ab_valyrian_events.0003.desc
          theme = realm
          left_portrait = {
                   character = root
                   animation = shame
          }

          trigger = {
                  AND = {
                         is_alive = yes
                         has_trait = dragon_emperor
                  }
                  NOT = { has_title = title:k_dragon_throne }
          }

          option = {
                     name = ab_valyrian_events.0003.a
                     remove_trait = dragon_emperor
                     stress_impact = {
                             base = minor_stress_impact_gain
                     }
          }
}

ab_valyrian_events.0004 = {
          type = character_event
          title = ab_valyrian_events.0004.title
          desc = ab_valyrian_events.0004.desc
          theme = dragon
	  override_background = { reference = agot_dragonstone_throneroom }
          left_portrait = {
                  character = root
                  animation = marshal
          }

          immediate = {
		if = { #remove the HoF blocker 
			limit = {
				religion = {
					any_faith = { has_doctrine = ab_broken_crown_doctrine }
				}
			}
			religion = {
				every_faith = {
					limit = {
						has_doctrine = ab_broken_crown_doctrine
					}
					remove_doctrine = ab_broken_crown_doctrine
				}
			}
		 }
          }

          option = {
                name = ab_valyrian_events.0004.a
                set_character_faith = faith:imperial_valyrian
                restore_dragon_throne_scripted_effect = yes
                add_trait = dragon_emperor
                add_trait = conqueror
                culture = {
                        add_culture_tradition = tradition_rule_of_the_dragon_emperor
                        add_culture_tradition = tradition_ancient_birthright
                }
          }         
}
