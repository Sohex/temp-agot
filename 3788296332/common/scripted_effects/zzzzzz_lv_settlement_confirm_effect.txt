lv_settlement_confirm_effect = {
	if = {
		limit = { scope:liege = { is_alive = yes } }
		save_scope_as = settler
		set_variable = {
			name = lv_settlement_expedition_liege
			value = scope:liege
		}
		set_variable = {
			name = lv_settlement_expedition_destination
			value = scope:ruin
		}
		set_variable = {
			name = lv_settlement_cost
			value = scope:lv_settlement_cost
		}
		remove_variable = si_selecting_character
		scope:liege = {
			remove_variable = si_selected_character
			remove_variable = lv_settlement_state
			remove_variable = lv_restore_ruins_tooltip
			if = {
				limit = { owns_story_of_type = lv_settlement_story_cycle }
				random_owned_story = {
					limit = { story_type = lv_settlement_story_cycle }
					add_to_variable_list = {
						name = lv_settlement_expeditions
						target = scope:ruin
					}
				}
			}
			else = {
				create_story = lv_settlement_story_cycle
				random_owned_story = {
					limit = { story_type = lv_settlement_story_cycle }
					add_to_variable_list = {
						name = lv_settlement_expeditions
						target = scope:ruin
					}
				}
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = lv_settlement_accepted
				left_icon = root
				right_icon = scope:liege
				custom_tooltip = lv_settlement_accepted_desc
			}
		}
		set_variable = { name = lv_settlement_expedition_started value = yes }
		if = {
			limit = { exists = var:lv_settlement_cost }
			scope:liege = { remove_short_term_gold = scope:settler.var:lv_settlement_cost }
			remove_variable = lv_settlement_cost
		}
		if = {
			limit = { NOT = { scope:liege ?= scope:settler } }
			scope:liege = { trigger_event = lv_settlement_event.0010 }
		}
		trigger_event = lv_settlement_event.0010
		trigger_event = { id = lv_settlement_event.0099 days = 30 }
	}
}
