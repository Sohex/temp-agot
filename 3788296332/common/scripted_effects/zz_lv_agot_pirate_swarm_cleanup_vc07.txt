lv_agot_pirate_swarm_cleanup_effect = {
	if = {
		limit = {
			any_county_in_region = {
				region = world_stepstones
				title_province = {
					any_province_domicile = {
						is_domicile_type = agot_pirate_ship
						owner = {
							is_ai = yes
							is_landed = no
							any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
						}
						count >= 10
					}
				}
			}
		}

		if = {
			limit = { exists = dynasty:dynn_Placeholder_Pirate.dynasty_founder.house.house_head }
			dynasty:dynn_Placeholder_Pirate.dynasty_founder.house.house_head = {
				save_scope_as = lv_pirate_swarm_placeholder
			}
		}

		every_county_in_region = {
			region = world_stepstones
			every_county_province = {
				limit = {
					any_province_domicile = {
						is_domicile_type = agot_pirate_ship
						owner = {
							is_ai = yes
							is_landed = no
							any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
						}
						count >= 10
					}
				}
				random_province_domicile = {
					limit = {
						is_domicile_type = agot_pirate_ship
						owner = {
							is_ai = yes
							is_landed = no
							any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
						}
					}
					owner = { save_scope_as = lv_pirate_swarm_keeper }
				}
				every_province_domicile = {
					limit = {
						is_domicile_type = agot_pirate_ship
						owner = {
							is_ai = yes
							is_landed = no
							any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
							NOT = { this = scope:lv_pirate_swarm_keeper }
							trigger_if = {
								limit = { exists = scope:lv_pirate_swarm_placeholder }
								NOT = { this = scope:lv_pirate_swarm_placeholder }
							}
						}
					}
					owner = { add_to_list = lv_pirate_swarm }
				}
			}
		}

		every_in_list = {
			list = lv_pirate_swarm
			agot_pirate_becoming_a_laamp_domicile_destruction_workaround_effect = yes
			death = { death_reason = death_erased_for_script }
		}
	}
}
