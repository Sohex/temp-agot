
lv_place_dummy_ruler_effect = {
	save_scope_as = lv_dummy_ruler

	if = {
		limit = { has_title = $PLACEHOLDER_TITLE$ }
		set_realm_capital = $PLACEHOLDER_TITLE$
	}

	if = {
		limit = {
			any_held_title = {
				tier = tier_county
				NOT = { this = $PLACEHOLDER_TITLE$ }
				exists = title_province
			}
		}
		random_held_title = {
			limit = {
				tier = tier_county
				NOT = { this = $PLACEHOLDER_TITLE$ }
				exists = title_province
			}
			save_scope_as = lv_dummy_reachable_location
		}
		set_location = scope:lv_dummy_reachable_location.title_province
	}
	else_if = {
		limit = { exists = title:c_kings_landing.title_province }
		set_location = title:c_kings_landing.title_province
	}
}

lv_place_all_dummy_rulers_effect = {
	title:c_ruins.holder ?= { lv_place_dummy_ruler_effect = { PLACEHOLDER_TITLE = title:c_ruins } }
	title:c_unknown.holder ?= { lv_place_dummy_ruler_effect = { PLACEHOLDER_TITLE = title:c_unknown } }
	title:c_wilderness.holder ?= { lv_place_dummy_ruler_effect = { PLACEHOLDER_TITLE = title:c_wilderness } }
}
