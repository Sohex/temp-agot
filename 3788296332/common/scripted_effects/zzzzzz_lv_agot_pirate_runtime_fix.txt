
pirate_adventurer_start_war_effect = {
	$PIRATE$ = { save_scope_as = pirate_adventurer }
	$TARGET$ = { save_scope_as = adventurer_target }

	if = {
		limit = {
			exists = scope:pirate_adventurer
			exists = scope:adventurer_target
			scope:adventurer_target = {
				exists = holder
				exists = title_province
			}
		}

		scope:adventurer_target.holder = {
			save_scope_as = pirate_target_holder
			top_liege ?= { save_scope_as = pirate_target_top_liege }
		}
		if = {
			limit = { NOT = { exists = scope:pirate_target_top_liege } }
			scope:pirate_target_holder = { save_scope_as = pirate_target_top_liege }
		}

		if = {
			limit = {
				scope:pirate_adventurer = {
					any_held_title = { agot_is_pirate_domicile_title_trigger = no }
				}
			}
			if = {
				limit = { agot_dlc_check_for_landless_pirates = yes }
				agot_create_pirate_domicile_effect = { PIRATE = scope:pirate_adventurer }
				scope:adventurer_target = {
					random_title_to_title_neighboring_and_across_water_county = {
						limit = { title_province = { has_holding_type = pirate_den_holding } }
						alternative_limit = {
							title_province = { NOT = { has_holding_type = ruin_holding } }
							OR = {
								NOT = { exists = holder }
								holder = {
									NOT = { this = scope:pirate_target_holder }
									NOT = { is_allied_to = scope:pirate_target_holder }
								}
							}
						}
						save_scope_as = pirate_nearby_safe_location
					}
				}
				if = {
					limit = { NOT = { exists = scope:pirate_nearby_safe_location } }
					scope:adventurer_target = {
						random_county_in_region = {
							region = world_stepstones
							limit = {
								title_province = { NOT = { has_holding_type = ruin_holding } }
								OR = {
									NOT = { exists = holder }
									holder = { NOT = { this = scope:pirate_target_holder } }
								}
							}
							save_scope_as = pirate_nearby_safe_location
						}
					}
				}
				if = {
					limit = {
						exists = scope:pirate_nearby_safe_location
						scope:pirate_adventurer = { exists = domicile }
					}
					scope:pirate_adventurer.domicile = { move_domicile = scope:pirate_nearby_safe_location }
				}
			}
			else = {
				scope:pirate_adventurer = {
					create_dynamic_title = { tier = duchy name = PIRATE_ARMY_NEUTRAL_NAME }
				}
				create_title_and_vassal_change = { type = created save_scope_as = change add_claim_on_loss = no }
				scope:new_title = {
					set_capital_county = scope:adventurer_target
					set_landless_title = yes
					set_destroy_on_gain_same_tier = yes
					set_no_automatic_claims = yes
					set_can_be_named_after_dynasty = no
					change_title_holder = { holder = scope:pirate_adventurer change = scope:change }
				}
				resolve_title_and_vassal_change = scope:change
				scope:new_title = {
					generate_coa = yes
					set_variable = { name = temporary_title value = yes }
				}
			}
		}

		if = {
			limit = {
				scope:pirate_target_top_liege = { has_realm_law = top_liege_not_redirected_to_border_wars_law }
			}
			scope:pirate_target_holder = { save_scope_as = pirate_war_defender }
		}
		else = { scope:pirate_target_top_liege = { save_scope_as = pirate_war_defender } }

		if = {
			limit = {
				exists = scope:pirate_war_defender
				scope:pirate_war_defender = { is_alive = yes }
				scope:pirate_adventurer = {
					is_alive = yes
					NOT = { is_at_war_with = scope:pirate_war_defender }
				}
			}
			scope:pirate_adventurer = {
				start_war = {
					casus_belli = pirate_adventurer_conquest
					target = scope:pirate_war_defender
					target_title = scope:adventurer_target.duchy
				}
				add_character_modifier = pirate_adventurer_modifier
				add_trait = adventurer
				spawn_army = {
					levies = {
						add = {
							add = scope:pirate_war_defender.max_military_strength
							multiply = 1.5
							scope:pirate_war_defender = { every_ally = { add = this.max_military_strength } }
							max = 8000
							min = 1000
						}
					}
					inheritable = no
					location = scope:adventurer_target.title_province
					name = pirate_adventurer_event_troops
				}
				spawn_army = {
					men_at_arms = { type = westerosi_sellswords stacks = { value = 1 multiply = scope:pirate_war_defender.primary_title.tier } }
					inheritable = yes
					location = scope:adventurer_target.title_province
					name = pirate_adventurer_event_troops
				}
				spawn_army = {
					men_at_arms = { type = stepstone_sailors stacks = { value = 1 multiply = scope:pirate_war_defender.primary_title.tier } }
					inheritable = yes
					location = scope:adventurer_target.title_province
					name = pirate_adventurer_event_troops
				}
				spawn_army = {
					men_at_arms = { type = mangonel stacks = 2 }
					inheritable = yes
					location = scope:adventurer_target.title_province
					name = pirate_adventurer_event_troops
				}
				add_gold = pirate_adventurer_efficacy_loan_value
			}
		}
	}
}
