
lv_agot_posthistory_government_fallback_effect = {
	every_county = {
		every_past_holder = {
			limit = {
				is_alive = no
				primary_title.title_capital_county.title_province ?= {
					OR = {
						geographical_region = world_asshai
						geographical_region = world_sothoryos_basilisk_isles
					}
				}
				agot_is_in_historical_government_list = no
			}
			save_temporary_scope_as = temp_ruler_scope

			if = {
				limit = { primary_title.holder ?= { agot_ruler_is_government_trigger = { GOV = administrative } } }
				assign_government_data_effect = {
					GOV = administrative
					SCOPE = scope:temp_ruler_scope
				}
			}
			else_if = {
				limit = { primary_title.holder ?= { agot_ruler_is_government_trigger = { GOV = republic } } }
				assign_government_data_effect = {
					GOV = republic
					SCOPE = scope:temp_ruler_scope
				}
			}
			else_if = {
				limit = { primary_title.holder ?= { agot_ruler_is_government_trigger = { GOV = oligarchic } } }
				assign_government_data_effect = {
					GOV = oligarchic
					SCOPE = scope:temp_ruler_scope
				}
			}
			else_if = {
				limit = { primary_title.holder ?= { agot_ruler_is_government_trigger = { GOV = pirate } } }
				assign_government_data_effect = {
					GOV = pirate
					SCOPE = scope:temp_ruler_scope
				}
			}
			else_if = {
				limit = { primary_title.holder ?= { agot_ruler_is_government_trigger = { GOV = pirate_no_dlc } } }
				assign_government_data_effect = {
					GOV = pirate_no_dlc
					SCOPE = scope:temp_ruler_scope
				}
			}
			else = {
				assign_government_data_effect = {
					GOV = feudal
					SCOPE = scope:temp_ruler_scope
				}
			}
		}
	}
}
