lv_aurion_forsake_hegemony_effect = {
	save_scope_as = lv_aurion

	if = {
		limit = { has_title = title:h_valyria }
		destroy_title = title:h_valyria
		add_unpressed_claim = title:h_valyria
	}

	if = {
		limit = {
			title:k_qohor.holder ?= {
				exists = liege
				NOT = { this = scope:lv_aurion }
			}
		}
		create_title_and_vassal_change = {
			type = independency
			save_scope_as = lv_aurion_qohor_change
		}
		title:k_qohor.holder ?= {
			becomes_independent = { change = scope:lv_aurion_qohor_change }
		}
		resolve_title_and_vassal_change = scope:lv_aurion_qohor_change
	}

	if = {
		limit = {
			has_title = title:c_artisans_quarter
			exists = character:Ostorlat_15
			character:Ostorlat_15 = { is_alive = yes }
		}
		character:Ostorlat_15 = { get_title = title:c_artisans_quarter }
	}
	else_if = {
		limit = {
			has_title = title:c_artisans_quarter
			exists = title:k_qohor.holder
			NOT = { title:k_qohor.holder = { this = scope:lv_aurion } }
		}
		title:k_qohor.holder ?= { get_title = title:c_artisans_quarter }
	}

	if = {
		limit = {
			any_held_title = { has_variable = adventurer_creation_reason }
			NOT = { has_government = landless_adventurer_government }
		}
		change_government = landless_adventurer_government
	}

	if = {
		limit = { exists = title:c_rhoford.title_province }
		if = {
			limit = {
				has_domicile = yes
				NOT = { domicile.domicile_location = title:c_rhoford.title_province }
			}
			domicile ?= {
				move_domicile = title:c_rhoford.title_province
			}
		}
	}
}

lv_aurion_keep_hegemony_effect = {
	save_scope_as = lv_aurion

	random_owned_story = {
		limit = { story_type = story_cob_aurion }
		end_story = yes
	}

	if = {
		limit = { any_held_title = { has_variable = adventurer_creation_reason } }
		destroy_laamp_effect = { ADVENTURER = scope:lv_aurion }
	}

	if = {
		limit = { NOT = { has_government = administrative_government } }
		change_government = administrative_government
	}

	hidden_effect = {
		if = {
			limit = { has_trait = adventurer }
			remove_trait = adventurer
		}
	}

	if = {
		limit = { has_title = title:h_valyria }
		set_primary_title_to = title:h_valyria
	}
}

lv_aurion_settle_valyria_effect = {
	save_scope_as = lv_aurion
	title:c_valyria.title_province = { save_scope_as = lv_aurion_seat }

	get_title = title:c_valyria

	scope:lv_aurion_seat = {
		if = {
			limit = { has_province_modifier = agot_ruin_reconstruction }
			remove_province_modifier = agot_ruin_reconstruction
		}
		if = {
			limit = { has_building = large_ruin_04 }
			remove_building = large_ruin_04
		}
		else_if = {
			limit = { has_building = large_ruin_03 }
			remove_building = large_ruin_03
		}
		else_if = {
			limit = { has_building = large_ruin_02 }
			remove_building = large_ruin_02
		}
		else_if = {
			limit = { has_building = large_ruin_01 }
			remove_building = large_ruin_01
		}
		set_holding_type = castle_holding
		add_building = castle_02
	}

	if = {
		limit = { any_held_title = { has_variable = adventurer_creation_reason } }
		destroy_laamp_effect = { ADVENTURER = scope:lv_aurion }
	}
	if = {
		limit = { government_has_flag = government_is_landless_adventurer }
		change_government = feudal_government
	}

	title:c_valyria = {
		set_county_faith = scope:lv_aurion.faith
		if = {
			limit = { culture = culture:unknown_culture }
			set_county_culture = scope:lv_aurion.culture
		}
		generate_coa = yes
		every_county_province = { barony ?= { generate_coa = yes } }
	}
	if = {
		limit = { exists = scope:lv_aurion.house }
		scope:lv_aurion_seat.barony ?= { set_coa = scope:lv_aurion.house }
	}

	set_primary_title_to = title:c_valyria
}
