lv_agot_reconcile_pirate_succession_effect = {
	every_ruler = {
		every_held_title = {
			limit = {
				tier >= tier_county
				can_title_have_law_general_trigger = yes
			}
			if = {
				limit = {
					tier > tier_county
					holder ?= {
						government_has_flag = government_is_pirate_trigger_check
					}
					NOT = {
						has_title_law = pirate_succession_law
					}
				}
				add_title_law = pirate_succession_law
			}
			else_if = {
				limit = {
					has_title_law = pirate_succession_law
					holder ?= {
						NOT = {
							government_has_flag = government_is_pirate_trigger_check
						}
					}
				}
				remove_title_law = pirate_succession_law
			}
		}
	}
}

lv_agot_reconcile_pirate_government_effect = {
	every_ruler = {
		limit = {
			NOT = { government_has_flag = government_is_pirate_trigger_check }
			NOT = { any_held_title = { tier > tier_county } }
			any_held_title = { tier = tier_county }
			NOT = {
				any_held_title = {
					tier = tier_county
					NOT = { title_province = { has_holding_type = pirate_den_holding } }
				}
			}
			NOT = {
				any_held_title = {
					tier = tier_barony
					NOT = { title_province = { has_holding_type = pirate_den_holding } }
				}
			}
		}
		change_government = pirate_no_dlc_government
	}
}
