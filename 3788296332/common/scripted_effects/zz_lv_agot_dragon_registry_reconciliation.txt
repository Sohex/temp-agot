lv_agot_reconcile_dragon_registry_effect = {
	if = {
		limit = { agot_is_any_dragon_alive = no }
		set_global_variable = {
			name = lv_dragon_registry_was_empty
			value = yes
		}
	}

	if = {
		limit = { has_global_variable_list = living_dragons }
		every_in_global_list = {
			variable = living_dragons
			limit = {
				OR = {
					is_alive = no
					NOT = { has_trait = dragon }
				}
			}
			add_to_global_variable_list = {
				name = lv_stale_dragon_registry_entries
				target = this
			}
		}
	}
	if = {
		limit = {
			has_global_variable_list = lv_stale_dragon_registry_entries
		}
		every_in_global_list = {
			variable = lv_stale_dragon_registry_entries
			remove_list_global_variable = {
				name = living_dragons
				target = this
			}
		}
		remove_global_variable = lv_stale_dragon_registry_entries
	}

	every_living_dragon = {
		save_scope_as = lv_reconciled_dragon
		if = {
			limit = {
				NOT = {
					is_target_in_global_variable_list = {
						name = living_dragons
						target = this
					}
				}
			}
			if = {
				limit = {
					NOT = {
						any_in_global_list = {
							variable = gl_dragon_variable_storage
							var:dragon_id ?= scope:lv_reconciled_dragon
						}
					}
				}
				agot_dragon_transfer_vars_to_story_cycle_effect = {
					DRAGON = this
				}
			}
			else = {
				add_to_global_variable_list = {
					name = living_dragons
					target = this
				}
			}
		}
	}

	agot_init_magic_level_effect = yes

	if = {
		limit = {
			has_global_variable = lv_dragon_registry_was_empty
			agot_is_any_dragon_alive = yes
		}
		agot_update_glass_candles_effect = yes
		agot_dragon_revival_effect = yes
	}
	remove_global_variable = lv_dragon_registry_was_empty
}
