
lv_agot_set_colonization_cooldown_effect = {
	if = {
		limit = { has_game_rule = lv_agot_colonization_cooldown_none }
		remove_variable = colonization_cooldown
	}
	else_if = {
		limit = { has_game_rule = lv_agot_colonization_cooldown_fast }
		set_variable = {
			name = colonization_cooldown
			years = $FAST$
		}
	}
	else_if = {
		limit = { has_game_rule = lv_agot_colonization_cooldown_slow }
		set_variable = {
			name = colonization_cooldown
			years = $SLOW$
		}
	}
	else = {
		set_variable = {
			name = colonization_cooldown
			years = $DEFAULT$
		}
	}
}
