coronation_save_crowning_artifact_effect = {
	if = {
		limit = { coronation_proper_artifact_regalia_trigger = no }
		ordered_character_artifact = {
			order_by = artifact_rarity
			limit = {
				artifact_slot_type = helmet
				has_variable = artifact_succession_title
				var:artifact_succession_title = { this = $HOST$.primary_title }
			}
			alternative_limit = {
				artifact_slot_type = helmet
				NOT = { has_variable = artifact_succession_title }
			}
			alternative_limit = { artifact_slot_type = helmet }
			save_scope_as = crowning_artifact
		}
		if = {
			limit = { exists = scope:crowning_artifact }
			set_variable = {
				name = crowning_artifact
				value = scope:crowning_artifact
				years = 1
			}
		}
	}
	else = {
		ordered_character_artifact = {
			order_by = artifact_rarity
			limit = {
				artifact_slot_type = regalia
				has_variable = artifact_succession_title
				var:artifact_succession_title = { this = $HOST$.primary_title }
			}
			alternative_limit = {
				artifact_slot_type = regalia
				NOT = { has_variable = artifact_succession_title }
			}
			alternative_limit = { artifact_slot_type = regalia }
			save_scope_as = crowning_artifact
		}
	}
}
