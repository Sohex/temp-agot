

populist_support_multiplier = {
	value = 1
	if = {
		limit = { exists = root.top_liege }
		root.top_liege = {
			if = {
				limit = { has_realm_law = franchise_democratic_full }
				value = 5
			}
			else_if = {
				limit = { has_realm_law = franchise_lean_democratic }
				value = 2
			}
			else_if = {
				limit = { has_realm_law = franchise_aristocratic_full }
				value = 0.25
			}
		}
	}
}
