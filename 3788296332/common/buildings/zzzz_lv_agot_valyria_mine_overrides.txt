	}

	can_construct_potential = { lv_valyrian_mine = yes }

	can_construct = {
		scope:holder = { agot_is_dummy_character = no }
	}

	province_modifier = {
		travel_danger = 30
		monthly_income = special_mine_building_tax_tier_2
		stationed_heavy_infantry_damage_mult = normal_maa_damage_tier_1
		stationed_heavy_infantry_toughness_mult = normal_maa_toughness_tier_2
	}

	county_modifier = {
		build_speed = 0.15
		development_growth_factor = -0.25
		monthly_county_control_growth_factor = -0.25
		tax_mult = 0.1
	}

	character_modifier = {
		stress_gain_mult = 0.2
		stress_loss_mult = -0.2
	}

	on_complete = {
		save_scope_as = valyria_mine_province
		hidden_effect = {
			county.holder = {
				if = {
					limit = { root = { NOT = { has_variable = upgraded_mine } } }
					random_list = {
						2 = { trigger_event = special_blockers.0002 }
						6 = {
							send_interface_toast = {
								title = find_gold_cache
								left_icon = this
								add_gold = 800
							}
						}
					}
					root = { set_variable = { name = upgraded_mine } }
				}
			}
		}
	}

	ai_value = { base = 100 }

	flag = obstacle

	effect_desc = lv_wilderness_blocker_effect_mines

	flag = special_building_enabled_in_ruin_county
	flag = lv_agot_chain_valyria_mines
}
