small_ruin_04 = {
	construction_time = 2190

	can_construct_potential = {
		has_building = small_ruin_03
		custom_tooltip = {
			text = "RUIN_MUST_CLEAR_BLOCKERS"
			NOT = { has_building_with_flag = obstacle }
		}
		scope:holder = {
			NOT = {
				has_trait = ruin
			}
		}
	}

	on_complete = {
		if = {
			limit = {
				county.holder = { is_human = yes }
			}
			save_scope_as = completed_ruin
			county.holder = { trigger_event = agot_ruins.1000 }
		}
		else = {
			set_variable = small_ruin
			trigger_event = agot_colonization_events.9001
		}
	}

	cost_gold = 600

	character_modifier = {
		monthly_income = -2
	}

	province_modifier = {
		fort_level = 2
	}

	levy = 300
	max_garrison = 300
	garrison_reinforcement_factor = 0.09

	type_icon = "icon_building_ruin.dds"
}
