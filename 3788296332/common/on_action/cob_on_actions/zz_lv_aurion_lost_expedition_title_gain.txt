on_title_gain = {
    on_actions = {
        lv_aurion_lost_expedition_recovery_on_title_gain
    }
}

lv_aurion_lost_expedition_recovery_on_title_gain = {
    trigger = {
        exists = root
        scope:title ?= {
            tier = tier_county
            title_province ?= {
                has_building = lv_aurions_lost_expedition_01
            }
        }
        root = {
            NOT = { has_character_flag = lv_aurion_recovery_pending }
            is_target_in_global_variable_list = {
                name = lv_unavailable_unique_items
                target = flag:aurion_sword_lost
            }
            is_target_in_global_variable_list = {
                name = lv_unavailable_unique_items
                target = flag:aurion_armor_lost
            }
        }
    }
    effect = {
        root = {
            add_character_flag = lv_aurion_recovery_pending
            trigger_event = {
                id = cob_7899_scenerio.9998
            }
        }
    }
}
