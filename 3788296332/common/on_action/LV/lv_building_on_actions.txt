
lv_on_building_completed_maintenance = {
	effect = {
		# Blocker clearance
		if = { # Execute only for ruins and settlements to save performance (I love unnecessary premature optimizations)
			limit = {
				OR = {
					has_holding_type = ruin_holding
					has_holding_type = settlement_holding
				}
			}
			# Failsafe for if, somehow, blockers that should be cleared are left over
			lv_remove_all_cleared_blockers_effect = yes
		}
		if = {
			limit = { county = { exists = holder } }
			county.holder = { lv_agot_repair_dragonkeeper_order_effect = yes }
		}
	}
}

lv_dragon_pits_on_actions = {
	trigger = {
		NOT = { zz_has_natural_dragon_pit_barony = yes }
		county = { can_this_dragonpit_be_destroyed_when_dragonpit_building_is_replaced = yes }
	}
	effect = {
		county = {
			if = {
				limit = { has_variable_list = dragons_in_pit }
				clear_variable_list = dragons_in_pit
			}
			holder = {
				if = {
					limit = { has_realm_law = dragonpit_close_family_law }
					remove_realm_law = dragonpit_close_family_law
				}
				else_if = {
					limit = { has_realm_law = dragonpit_house_law }
					remove_realm_law = dragonpit_house_law
				}
				else_if = {
					limit = { has_realm_law = dragonpit_dynasty_law }
					remove_realm_law = dragonpit_dynasty_law
				}
				else_if = {
					limit = { has_realm_law = dragonpit_everyone_law }
					remove_realm_law = dragonpit_everyone_law
				}
			}
			remove_variable = has_dragonkeeper_order
			holder = { lv_agot_repair_dragonkeeper_order_effect = yes }
		}
	}
}


lv_agot_dragonkeeper_order_yearly_repair = {
	trigger = { is_landed = yes }
	effect = { lv_agot_repair_dragonkeeper_order_effect = yes }
}
