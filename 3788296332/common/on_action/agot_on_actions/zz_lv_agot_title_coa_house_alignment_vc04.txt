on_game_start_after_lobby = {
	on_actions = {
		delay = { days = 2 }
		lv_agot_title_coa_house_alignment_on_load
	}
}

lv_agot_title_coa_house_alignment_on_load = {
	effect = {
		every_ruler = {
			every_held_title = {
				limit = {
					is_noble_family_title = yes
					exists = holder.house
				}
				set_coa = holder.house
			}
		}
		title:k_dragonstone ?= {
			if = {
				limit = { exists = holder.house }
				set_coa = holder.house
			}
		}
		title:d_dragonstone ?= {
			if = {
				limit = { exists = holder.house }
				set_coa = holder.house
			}
		}
	}
}

# root = neuer Halter, scope:title = gewonnener Titel
on_title_gain = {
	on_actions = {
		lv_agot_title_coa_house_alignment_on_title_gain
	}
}

lv_agot_title_coa_house_alignment_on_title_gain = {
	trigger = {
		exists = root
		exists = root.house
		scope:title ?= {
			OR = {
				is_noble_family_title = yes
				this = title:k_dragonstone
				this = title:d_dragonstone
			}
		}
	}
	effect = {
		scope:title = { set_coa = root.house }
	}
}
