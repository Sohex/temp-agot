	type = character

	text = {
		trigger = {
			has_government = administrative_government

			has_realm_law = free_city_election_succession_law
			OR = {
				culture = culture:volantene
				faith = faith:fc_pan_volantis
			}
		}
		localization_key = volantene_male_triarch
	}
	text = {
		trigger = {
			any_liege_or_above = {
