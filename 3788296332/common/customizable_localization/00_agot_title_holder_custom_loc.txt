AGOTGetTitle = {
	type = character

	text = {
		setup_scope = {
			save_scope_as = character
			if = {
				limit = {
					exists = scope:character
					scope:character ?= { character_title_visible = yes }
				}

				#1. Assign Character Parameters
				assign_character_parameter_effect = { SCOPE = scope:character }

				#2. Localize Dowager Title
				apply_character_special_title_data = { SCOPE = scope:character }

				#3. Localize Regent Title
				apply_character_special_regent_title_data = { SCOPE = scope:character }

				#4. Localize Character Title
				if = {
					limit = { scope:ruler_status ?= flag:royal }
					apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:character.primary_title }
				}
				else_if = {
					limit = { scope:spouse_status ?= flag:royal }
					apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:spouse_scope.primary_title }
				}
				else_if = {
					limit = {
						scope:scion_status ?= flag:royal
						NOR = {
							exists = scope:maester
							exists = scope:silent_sister
							exists = scope:septon
							is_titled_ruler_or_spouse = yes
						}
					}
					if = {
						limit = { exists = scope:scion_1_title }
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:scion_1_title }
					}
					else_if = {
						limit = { exists = scope:scion_2_title }
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:scion_2_title }
					}
					else_if = {
						limit = { exists = scope:scion_3_title }
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:scion_3_title }
					}
				}
				else = {
					if = {
						limit = { scope:ruler_status ?= flag:active }
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:character.primary_title }
					}
					else_if = {
						limit = { scope:spouse_status ?= flag:active }
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:spouse_scope.primary_title }
					}
					else = {
						apply_character_title_data = { SCOPE = scope:character SCOPED_TITLE = scope:dummy_gender }
					}
				}
			}
		}
		trigger = {
			is_titled_character = yes
			exists = scope:regent_status
			exists = scope:get_title_regent
			OR = {
				AND = {
					exists = scope:ruler_status
					OR = {
						agot_is_historical_government_trigger = { GOV = lp_feudal }
						agot_is_historical_government_trigger = { GOV = feudal }
					}
				}
				AND = {
					exists = scope:spouse_status
					scope:spouse_scope ?= {
						OR = {
							agot_is_historical_government_trigger = { GOV = lp_feudal }
							agot_is_historical_government_trigger = { GOV = feudal }
						}
					}
				}
				exists = scope:scion_status
				AND = {
					exists = scope:councillor
					NOR = {
						exists = scope:knighthood
						exists = scope:maester
						exists = scope:silent_sister
						exists = scope:septon
					}
				}
			}
		}
		localization_key = AGOTGetTitleRegent
	}

	text = {

			}

				ruler_is_royal = yes
				historical_independent_ruler = yes
			}
		}
		localization_key = AGOTGetTitleDowager
	}

					scope:character ?= { character_title_visible = yes }

		trigger = {
			is_titled_character = yes
			exists = scope:get_title
		}

	}

		fallback = yes
		localization_key = AGOTGetTitleEmpty
	}
}

AGOTGetRegentTitle = {
	type = character

	text = {
		trigger = {
			character_is_regent = yes
			NOR = {
				ruler_is_royal = yes
				is_royal_scion = yes
				any_spouse = {
					even_if_dead = yes
					ruler_is_royal = yes
				}
			}
			liege ?= {
				OR = {
					AND = {
						ruler_is_royal = yes
						NOT = { primary_title ?= title:k_dragonstone }
					}
					primary_title.tier >= tier_empire
				}
			}
			is_female = yes
		}
		localization_key = REGENT_PROTECTRESS
	}

	text = {
		trigger = {
			character_is_regent = yes
			NOR = {
				ruler_is_royal = yes
				is_royal_scion = yes
				any_spouse = {
					even_if_dead = yes
					ruler_is_royal = yes
				}
			}
			liege ?= {
				OR = {
					AND = {
						ruler_is_royal = yes
						NOT = { primary_title ?= title:k_dragonstone }
					}
					primary_title.tier >= tier_empire
				}
			}
			is_female = no
		}
		localization_key = REGENT_PROTECTOR
	}

	text = {
		trigger = {
			character_is_regent = yes
			is_female = yes
		}
		localization_key = REGENT_FEMALE
	}

	text = {
		trigger = {
			character_is_regent = yes
		}
		localization_key = REGENT_MALE
	}
}
