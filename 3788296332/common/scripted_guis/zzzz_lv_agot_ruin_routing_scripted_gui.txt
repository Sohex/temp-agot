lv_use_custom_colonization = {
	scope = province
	is_shown = {
		NOT = { lv_agot_ruin_uses_agot_restoration_trigger = yes }
	}
}

lv_restore_ruins_tooltip_show = {
	scope = province
	effect = {
		if = {
			limit = { NOT = { lv_agot_ruin_uses_agot_restoration_trigger = yes } }
			save_scope_as = ruin
			scope:actor = {
				set_variable = {
					name = lv_restore_ruins_tooltip
					value = scope:ruin
				}
			}
		}
	}
}
