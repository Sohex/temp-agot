# Ancient Birthright compatch - the Shrouded Sisters camp.
#
# Every working landless adventurer title in AGOT sets the variable adventurer_creation_reason
# on itself when it is created. Ancient Birthright's d_laamp_Anogrion_1, the Shrouded Sisters,
# does not. Without that variable the game does not treat the title as an adventurer camp, so
# its holder is counted as landless and the "landless for a year" game over fires on them.
# Confirmed against a save: Aurion's and Justman's camps carry the variable, this one carries
# only legitimate_house_title.
#
# Set from on_game_start_after_lobby rather than from title history, so we never risk clobbering
# Ancient Birthright's own setup effects, and so existing saves are repaired on load.
# RE-CHECK if Ancient Birthright adds the variable themselves, then this can be dropped.

eetlv_ab_fix_shrouded_sisters_camp_effect = {
	title:d_laamp_Anogrion_1 ?= {
		if = {
			limit = {
				NOT = { has_variable = adventurer_creation_reason }
			}
			set_variable = { name = adventurer_creation_reason value = flag:historical }
		}
	}
}
