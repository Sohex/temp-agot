# The proposer chooses how to frame the motion. 80% of the chosen skill becomes
# the starting speech score; a later interruption tests the delivery under pressure.

	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_proposal.1000.t
	desc = dragonlord_council_proposal.1000.desc
	theme = realm

	left_portrait = {
		character = root
		animation = chancellor
	}

	trigger = {
		this = scope:host
		scope:activity = { has_current_phase = dragonlord_council_phase_proposal }
	}

	option = {
		name = dragonlord_council_proposal.1000.a
		set_variable = { name = dragonlord_council_speech_style value = flag:oratorical }

# One attending voter challenges the speech.  The proposer may stay with the
# prepared style or pivot to another skill; matching the prepared style grants
# a modest edge without locking the player into one response.
dragonlord_council_proposal.1010 = {
	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_proposal.1010.t
	desc = dragonlord_council_proposal.1010.desc
	theme = realm

	left_portrait = {
		character = root
		animation = chancellor
	}
	right_portrait = {
		character = scope:dragonlord_council_critic
		animation = anger
	}

	trigger = {
		this = scope:host
		scope:activity = {
			has_current_phase = dragonlord_council_phase_proposal
			any_attending_character = {
				this != root
				is_ai = yes
				has_variable = dragonlord_council_meeting_base_support
			}
		}
	}

	immediate = {
		scope:activity = {
			random_attending_character = {
				limit = {
					this != root
					is_ai = yes
					has_variable = dragonlord_council_meeting_base_support
				}
				save_scope_as = dragonlord_council_critic
			}
		}
	}

	option = {
		name = dragonlord_council_proposal.1010.a
		duel = {
			skill = diplomacy
			target = scope:dragonlord_council_critic
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				modifier = { add = 15 var:dragonlord_council_speech_style = flag:oratorical }
				desc = dragonlord_council_proposal.challenge.success
				change_variable = { name = dragonlord_council_speech_score add = 10 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = dragonlord_council_proposal.challenge.success
					left_icon = root
					right_icon = scope:dragonlord_council_critic
				}
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
				desc = dragonlord_council_proposal.challenge.failure
				change_variable = { name = dragonlord_council_speech_score add = -10 }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = dragonlord_council_proposal.challenge.failure

				}

# When an AI Chief Dragonlord presides, the attending player may either make a
# public challenge or leave the existing interruption to another AI delegate.
dragonlord_council_proposal.1030 = {
	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_proposal.1030.t
	desc = dragonlord_council_proposal.1030.desc
	theme = realm

	left_portrait = {
		character = root
		animation = debating
	}
	right_portrait = {
		character = scope:host
		animation = chancellor
	}

	trigger = {
		this != scope:host
		is_ai = no
		has_variable = dragonlord_council_meeting_base_support
		scope:host = { is_ai = yes }
		scope:activity = { has_current_phase = dragonlord_council_phase_proposal }
	}

	immediate = {
		# Keep attendee event scenery tied to the presiding Dragonlord's realm.
		scope:host = { save_scope_as = bg_override_char }
	}

	option = {
		name = dragonlord_council_proposal.1030.a
		scope:host = {
			add_opinion = {
				target = root
				modifier = dragonlord_council_embarrassed_opinion
				opinion = -10
			}
		}
		duel = {
			skill = diplomacy
			target = scope:host
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_proposal.guest_challenge.success
				scope:host = { change_variable = { name = dragonlord_council_speech_score add = -10 } }
				send_interface_toast = {
					type = event_toast_effect_good
					title = dragonlord_council_proposal.guest_challenge.success
					left_icon = root
					right_icon = scope:host
				}
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
				desc = dragonlord_council_proposal.guest_challenge.failure
				scope:host = { change_variable = { name = dragonlord_council_speech_score add = 10 } }
				send_interface_toast = {
					type = event_toast_effect_bad
					title = dragonlord_council_proposal.guest_challenge.failure
					left_icon = root
					right_icon = scope:host
				}
			}
		}
		scope:host = { trigger_event = { id = dragonlord_council_proposal.1020 days = 2 } }
	}

	option = {
		name = dragonlord_council_proposal.1030.b
		if = {
			limit = {
				scope:activity = {
					any_attending_character = {
						this != scope:host
						is_ai = yes
						has_variable = dragonlord_council_meeting_base_support
					}
				}
			}
			scope:host = { trigger_event = { id = dragonlord_council_proposal.1010 days = 2 } }
		}
		else = {
			scope:host = { trigger_event = { id = dragonlord_council_proposal.1020 days = 2 } }
		}
	}
}

# Routes the accumulated score into one of seven quality tiers, including a
# deliberately neutral result between the negative and positive bands.
dragonlord_council_proposal.1020 = {
	type = activity_event
	hidden = yes
	trigger = {
		this = scope:host
		has_variable = dragonlord_council_speech_score
		scope:activity = { has_current_phase = dragonlord_council_phase_proposal }
	}
	immediate = {
		if = {
			limit = { var:dragonlord_council_speech_score <= 0 }
			trigger_event = dragonlord_council_proposal.1021
		}
		else_if = {
			limit = { var:dragonlord_council_speech_score <= 7 }
			trigger_event = dragonlord_council_proposal.1022
		}
		else_if = {
			limit = { var:dragonlord_council_speech_score <= 14 }
			trigger_event = dragonlord_council_proposal.1023
		}
		else_if = {
			limit = { var:dragonlord_council_speech_score <= 19 }
			trigger_event = dragonlord_council_proposal.1027
		}
		else_if = {
			limit = { var:dragonlord_council_speech_score <= 24 }
			trigger_event = dragonlord_council_proposal.1024
		}
		else_if = {
			limit = { var:dragonlord_council_speech_score <= 29 }
			trigger_event = dragonlord_council_proposal.1025
		}
		else = { trigger_event = dragonlord_council_proposal.1026 }
	}
}
