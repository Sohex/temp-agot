# A voter publicly challenges the motion.  Characters close to the opposition
# threshold are strongly preferred, so a good answer can visibly change a vote.
dragonlord_council_discussion.2000 = {
	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_discussion.2000.t
	desc = dragonlord_council_discussion.2000.desc
	theme = realm

	left_portrait = {
		character = root
		animation = chancellor
	}
	right_portrait = {
		character = scope:dragonlord_council_discussion_speaker
		animation = debating
	}

	trigger = {
		this = scope:host
		scope:activity = { has_current_phase = dragonlord_council_phase_discussion }
		any_in_list = {
			variable = dragonlord_council_meeting_voters
			this != root
			is_ai = yes
			is_alive = yes
			has_variable = dragonlord_council_meeting_base_support
		}
	}

	immediate = {
		random_in_list = {
			variable = dragonlord_council_meeting_voters
			limit = {
				this != root
				is_ai = yes
				is_alive = yes
				has_variable = dragonlord_council_meeting_base_support
			}
			weight = {
				base = 1
				modifier = {
					add = 200
					dragonlord_council_meeting_final_support <= 0
					dragonlord_council_meeting_final_support >= -15
				}
				modifier = {
					add = 50
					dragonlord_council_meeting_final_support <= 0
					dragonlord_council_meeting_final_support >= -30
				}
				modifier = {
					add = 20
					dragonlord_council_meeting_final_support > 0
					dragonlord_council_meeting_final_support <= 15
				}
			}
			save_scope_as = dragonlord_council_discussion_speaker
		}
	}

	option = {
		name = dragonlord_council_discussion.2000.a
		duel = {
			skill = diplomacy
			target = scope:dragonlord_council_discussion_speaker
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_discussion.answer.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = dragonlord_council_discussion.answer.success
					left_icon = root
					right_icon = scope:dragonlord_council_discussion_speaker
					scope:dragonlord_council_discussion_speaker = {
						dragonlord_council_change_voter_support_effect = { VALUE = 10 }
						add_opinion = {
							target = root
							modifier = respect_opinion
							opinion = 10
						}
					}
					custom_tooltip = dragonlord_council_discussion.speaker_support_plus_10_tt
				}
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
				desc = dragonlord_council_discussion.answer.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = dragonlord_council_discussion.answer.failure
					left_icon = root
					right_icon = scope:dragonlord_council_discussion_speaker
					scope:dragonlord_council_discussion_speaker = {
						dragonlord_council_change_voter_support_effect = { VALUE = -5 }
						add_opinion = {
							target = root
							modifier = disrespect_opinion
							opinion = -10
						}
					}
					custom_tooltip = dragonlord_council_discussion.speaker_support_minus_5_tt
				}
			}
		}
		ai_chance = { base = 30 modifier = { add = diplomacy } }
	}

	option = {
		name = dragonlord_council_discussion.2000.b
		duel = {
			skill = learning
			target = scope:dragonlord_council_discussion_speaker
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_discussion.answer.success
				send_interface_toast = {
					type = event_toast_effect_good

					right_icon = scope:dragonlord_council_discussion_speaker
					scope:dragonlord_council_discussion_speaker = {

							opinion = 5
						}
					}
					custom_tooltip = dragonlord_council_discussion.speaker_support_plus_10_tt
				}

			50 = {

				desc = dragonlord_council_discussion.answer.failure
				send_interface_toast = {
					type = event_toast_effect_bad

# Two voters turn the chamber into a personal quarrel.  This mirrors feast
# relationship events: council support changes remain meeting-only, while the
# opinion changes persist after the activity.
dragonlord_council_discussion.2010 = {
	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_discussion.2010.t
	desc = dragonlord_council_discussion.2010.desc
	theme = realm

	left_portrait = {
		character = root
		animation = chancellor
	}
	center_portrait = {
		character = scope:dragonlord_council_disputant_1
		animation = anger
	}
	right_portrait = {
		character = scope:dragonlord_council_disputant_2
		animation = anger
	}

	trigger = {
		this = scope:host
		scope:activity = { has_current_phase = dragonlord_council_phase_discussion }
		any_in_list = {
			variable = dragonlord_council_meeting_voters
			count >= 2
			this != root
			is_ai = yes
			is_alive = yes
			has_variable = dragonlord_council_meeting_base_support
		}
	}

	immediate = {
		random_in_list = {
			variable = dragonlord_council_meeting_voters
			limit = {
				this != root
				is_ai = yes
				is_alive = yes
				has_variable = dragonlord_council_meeting_base_support
			}
			save_scope_as = dragonlord_council_disputant_1
		}
		random_in_list = {
			variable = dragonlord_council_meeting_voters
			limit = {
				this != root
				this != scope:dragonlord_council_disputant_1
				is_ai = yes
				is_alive = yes
				has_variable = dragonlord_council_meeting_base_support
			}
			save_scope_as = dragonlord_council_disputant_2
		}
	}

	option = {
		name = dragonlord_council_discussion.2010.a
		duel = {
			skill = diplomacy
			value = medium_skill_rating
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_discussion.mediate.success
				send_interface_toast = {
					type = event_toast_effect_good
					title = dragonlord_council_discussion.mediate.success
					left_icon = scope:dragonlord_council_disputant_1
					right_icon = scope:dragonlord_council_disputant_2
					scope:dragonlord_council_disputant_1 = {
						dragonlord_council_change_voter_support_effect = { VALUE = 5 }
						add_opinion = { target = root modifier = respect_opinion opinion = 10 }
						add_opinion = { target = scope:dragonlord_council_disputant_2 modifier = respect_opinion opinion = 10 }
					}
					scope:dragonlord_council_disputant_2 = {
						dragonlord_council_change_voter_support_effect = { VALUE = 5 }
						add_opinion = { target = root modifier = respect_opinion opinion = 10 }
						add_opinion = { target = scope:dragonlord_council_disputant_1 modifier = respect_opinion opinion = 10 }
					}
					custom_tooltip = dragonlord_council_discussion.both_support_plus_5_tt
				}
			}
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = -3.5 min = -49 }
				desc = dragonlord_council_discussion.mediate.failure
				send_interface_toast = {
					type = event_toast_effect_bad
					title = dragonlord_council_discussion.mediate.failure
					left_icon = scope:dragonlord_council_disputant_1
					right_icon = scope:dragonlord_council_disputant_2
					scope:dragonlord_council_disputant_1 = {
						dragonlord_council_change_voter_support_effect = { VALUE = -5 }
						add_opinion = { target = root modifier = disrespect_opinion opinion = -5 }
						add_opinion = { target = scope:dragonlord_council_disputant_2 modifier = disrespect_opinion opinion = -10 }
					}
					scope:dragonlord_council_disputant_2 = {
						dragonlord_council_change_voter_support_effect = { VALUE = -5 }
						add_opinion = { target = root modifier = disrespect_opinion opinion = -5 }
						add_opinion = { target = scope:dragonlord_council_disputant_1 modifier = disrespect_opinion opinion = -10 }
					}
					custom_tooltip = dragonlord_council_discussion.both_support_minus_5_tt
				}
			}
		}
		ai_chance = { base = 50 modifier = { add = diplomacy } }
	}

	option = {
		name = dragonlord_council_discussion.2010.b
		scope:dragonlord_council_disputant_1 = {
			dragonlord_council_change_voter_support_effect = { VALUE = 10 }
			add_opinion = { target = root modifier = respect_opinion opinion = 10 }
			add_opinion = { target = scope:dragonlord_council_disputant_2 modifier = disrespect_opinion opinion = -10 }
		}
		scope:dragonlord_council_disputant_2 = {
			dragonlord_council_change_voter_support_effect = { VALUE = -10 }
			add_opinion = { target = root modifier = disrespect_opinion opinion = -15 }

		}

# An attending player under an AI Chief may intervene once in the public
# discussion. A failed speech has no reverse mechanical effect.
dragonlord_council_discussion.2020 = {
	type = activity_event
	window = dragonlord_council_event
	title = dragonlord_council_discussion.2020.t
	desc = dragonlord_council_discussion.2020.desc
	theme = realm

	left_portrait = {
		character = root
		animation = debating
	}
	right_portrait = {
		character = scope:host
		animation = thinking
	}

	trigger = {
		this != scope:host
		is_ai = no
		has_variable = dragonlord_council_meeting_base_support
		scope:host = { is_ai = yes }
		scope:activity = { has_current_phase = dragonlord_council_phase_discussion }
	}

	immediate = {
		scope:host = { save_scope_as = bg_override_char }
		save_scope_as = dragonlord_council_guest_speaker
	}

	option = {
		name = dragonlord_council_discussion.2020.a
		scope:host = {
			add_opinion = {
				target = root
				modifier = dragonlord_council_supported_motion_opinion
				opinion = 10
			}
		}
		duel = {
			skill = diplomacy
			value = medium_skill_rating
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_discussion.guest_support.success

					type = event_toast_effect_good

			skill = diplomacy
			value = medium_skill_rating
			50 = {
				compare_modifier = { value = scope:duel_value multiplier = 3.5 min = -49 }
				desc = dragonlord_council_discussion.guest_opposition.success
				dragonlord_council_change_other_voter_support_effect = { VALUE = -5 }
