# Character-name widgets such as the House view call the engine's native
# title flavourization directly instead of AGOT's scripted title dispatcher.
# Give the Dragonlord Oligarchy explicit, high-priority holder titles so those
# widgets never fall back to Emperor/Empress or Free City ranks.

# Match AGOT's Free City pattern: the ruler and the ruler's spouse need
# separate empire-tier entries. Without the spouse entry, a landed
# administrative spouse inherits the generic Emperor/Empress title.

	type = character
	gender = male
	special = holder
	tier = empire
	priority = 1000
	governments = { dragonlord_oligarchy_government }
}
