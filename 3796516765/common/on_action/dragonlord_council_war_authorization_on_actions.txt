on_war_started = {
	on_actions = {
		dragonlord_council_consume_war_authorization_on_action
	}
}

# on_war_started exposes the same attacker/defender scopes as CB declaration.
# Any authorized war against the approved target consumes the permit once.
dragonlord_council_consume_war_authorization_on_action = {
	effect = {
		scope:attacker = {
			if = {
				limit = {
					has_character_flag = dragonlord_council_war_authorized
					has_variable = dragonlord_council_war_authorization_target
					var:dragonlord_council_war_authorization_target ?= scope:defender
				}
				dragonlord_council_clear_war_authorization_effect = yes
			}
		}
	}
}
