# Candidate scope. Each living, tamed dragon held by a member of the
# candidate's House contributes twenty points to the council's acclamation.
dragonlord_house_dragon_acclamation_score = {
	value = 0
	house ?= {
		every_house_member = {
			limit = {
				is_alive = yes
				has_trait = dragonrider
			}
			every_relation = {
					type = agot_dragon
					limit = {
						is_alive = yes
					}
					add = 20
				}
		}
	}
}
