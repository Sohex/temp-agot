# Late-loading extension of AGOT's taming roll.  All AGOT inputs and caps are
# retained; the Dragon Lords family attribute adds 5/10/15 percentage points.
dragon_taming_attempt_chance = {
	value = 0
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				has_relation_agot_dragon = scope:dragon_taming_modifier_dragon
			}
		}
		add = dragon_scheme_base_taming_bonded_value
	}
	else = { add = dragon_scheme_base_taming_unbonded_value }

	if = {
		limit = {
			scope:dragon_taming_modifier_dragon = {
				NOT = { has_character_flag = owned_dragon }
			}
		}
		add = dragon_scheme_base_wild_value
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = { has_trait = dragonless_dragonrider }
		}
		add = -30
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				knows_language_of_culture = scope:dragon_taming_modifier_dragon.culture
			}
		}
		add = dragon_scheme_base_common_language_value
	}

	add = {
		value = dragon_traits_in_common
		multiply = dragon_scheme_base_common_traits_value
	}
	if = {
		limit = { scope:dragon_taming_modifier_tamer = { has_trait = diligent } }
		add = dragon_scheme_diligent_value
	}
	if = {
		limit = { scope:dragon_taming_modifier_tamer = { has_trait = lazy } }
		add = dragon_scheme_lazy_value
	}

	add = scope:dragon_taming_modifier_dragon.temperament
	add = scope:dragon_taming_modifier_dragon.taming_chance

	if = {
		limit = {
			scope:dragon_taming_modifier_dragon = { has_character_flag = in_dragonpit }
		}
		add = 10
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = { has_character_modifier = agot_taught_dragonlore }
		}
		add = 10
	}

	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				exists = court_position:head_dragonkeeper_court_position
				court_position:head_dragonkeeper_court_position = {
					aptitude:head_dragonkeeper_court_position = 0
				}
			}
		}
		add = 1
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				exists = court_position:head_dragonkeeper_court_position
				court_position:head_dragonkeeper_court_position = {
					aptitude:head_dragonkeeper_court_position = 1
				}
			}
		}
		add = 2
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				exists = court_position:head_dragonkeeper_court_position
				court_position:head_dragonkeeper_court_position = {
					aptitude:head_dragonkeeper_court_position = 2
				}
			}
		}
		add = 5
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				exists = court_position:head_dragonkeeper_court_position
				court_position:head_dragonkeeper_court_position = {
					aptitude:head_dragonkeeper_court_position = 3
				}
			}
		}
		add = 10
	}
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				exists = court_position:head_dragonkeeper_court_position
				court_position:head_dragonkeeper_court_position = {
					aptitude:head_dragonkeeper_court_position = 4
				}
			}
		}
		add = 15
	}

	add = {
		value = scope:dragon_taming_modifier_tamer.dragonblood_percent
		divide = 2
		desc = "SCHEME_BOND_VALYRIAN_BLOOD"
	}
	if = {
		limit = {
			agot_is_canon_rider_dragon_pair = {
				RIDER = scope:dragon_taming_modifier_tamer
				DRAGON = scope:dragon_taming_modifier_dragon
			}
		}
		add = 100
	}

	# The highest active level is checked first because level parameters are
	# cumulative for upgraded family attributes.
	if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				dragonlord_is_registered_dragonlord_member_trigger = yes
				house ?= { has_house_aspiration_parameter = dragon_lords_taming_chance_3 }
			}
		}
		add = { value = 15 desc = dragon_lords_taming_chance_bonus }
	}
	else_if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				dragonlord_is_registered_dragonlord_member_trigger = yes
				house ?= { has_house_aspiration_parameter = dragon_lords_taming_chance_2 }
			}
		}
		add = { value = 10 desc = dragon_lords_taming_chance_bonus }
	}
	else_if = {
		limit = {
			scope:dragon_taming_modifier_tamer = {
				dragonlord_is_registered_dragonlord_member_trigger = yes
				house ?= { has_house_aspiration_parameter = dragon_lords_taming_chance_1 }
			}
		}
		add = { value = 5 desc = dragon_lords_taming_chance_bonus }
	}

	max = {
		value = scope:dragon_taming_modifier_tamer.dragonblood_percent
		max = 85
		if = {
			limit = {
				scope:dragon_taming_modifier_tamer = {
					has_relation_agot_dragon = scope:dragon_taming_modifier_dragon
				}
			}
			add = 100
		}
		if = {
			limit = {
				agot_canon_dragons_enabled = yes
				agot_is_canon_rider_dragon_pair = {
					RIDER = scope:dragon_taming_modifier_tamer
					DRAGON = scope:dragon_taming_modifier_dragon
				}
			}
			add = 100
		}
		max = 100
	}
}
