# Remove modifiers left by the former character-modifier implementation. The
# independent Dragonlord law now owns these values directly.
dragonlord_clear_legacy_bureaucracy_modifiers_effect = {
	remove_character_modifier = dragonlord_bureaucracy_opinion_penalty_1
	remove_character_modifier = dragonlord_bureaucracy_opinion_penalty_2
	remove_character_modifier = dragonlord_bureaucracy_opinion_penalty_3

	if = {
		limit = { has_character_modifier = liege_authority_boost_has_scapegoat_modifier }
		remove_character_modifier = liege_authority_boost_has_scapegoat_modifier
	}
}

# Migrate an existing Dragonlord realm once from the original authority tier.
# The original tier remains a hidden compatibility proxy because vanilla and
# AGOT contain direct checks for its four database keys. After migration it is
# synchronized only by the custom law's on_pass effect, never by this pulse.
dragonlord_refresh_bureaucracy_opinion_penalty_effect = {
	dragonlord_clear_legacy_bureaucracy_modifiers_effect = yes

	if = {
		limit = {
			top_liege = this
			has_government = dragonlord_oligarchy_government
		}

		if = {
			limit = {
				OR = {
					primary_title = {
						NOT = { has_variable = dragonlord_bureaucracy_migrated }
					}
					NOR = {
						has_realm_law = dragonlord_imperial_bureaucracy_0
						has_realm_law = dragonlord_imperial_bureaucracy_1
						has_realm_law = dragonlord_imperial_bureaucracy_2
						has_realm_law = dragonlord_imperial_bureaucracy_3
					}
				}
			}

			# Existing Dragonlord laws are already authoritative. Merely record
			# migration on the realm title; re-adding the same law emits a false
			# "ruler passed a law" notification whenever the ruler changes.
			if = {
				limit = {
					NOR = {
						has_realm_law = dragonlord_imperial_bureaucracy_0
						has_realm_law = dragonlord_imperial_bureaucracy_1
						has_realm_law = dragonlord_imperial_bureaucracy_2
						has_realm_law = dragonlord_imperial_bureaucracy_3
					}
				}

				if = {
					limit = { has_realm_law = imperial_bureaucracy_3 }
					add_realm_law_skip_effects = dragonlord_imperial_bureaucracy_3
				}
				else_if = {
					limit = { has_realm_law = imperial_bureaucracy_2 }
					add_realm_law_skip_effects = dragonlord_imperial_bureaucracy_2
				}
				else_if = {
					limit = { has_realm_law = imperial_bureaucracy_1 }
					add_realm_law_skip_effects = dragonlord_imperial_bureaucracy_1
				}
				else_if = {
					limit = { has_realm_law = imperial_bureaucracy_0 }
					add_realm_law_skip_effects = dragonlord_imperial_bureaucracy_0
				}
				else = {
					# Tier 1 is the default for old realms with neither law group.
					add_realm_law_skip_effects = imperial_bureaucracy_1
					add_realm_law_skip_effects = dragonlord_imperial_bureaucracy_1
				}
			}

			primary_title = {
				set_variable = { name = dragonlord_bureaucracy_migrated value = yes }
			}
		}
	}
}

# Preserve the vanilla cleanup hook, but do not re-add the removed legacy
# modifiers. Custom-law modifiers are applied by the law database itself.
remove_law_related_modifiers_effect = {
	hidden_effect = {
		dragonlord_clear_legacy_bureaucracy_modifiers_effect = yes
	}
}
