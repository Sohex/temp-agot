dragonlord_refresh_fourteen_attribute_modifiers_effect = {
	# Root scope: a House supplied by an aspiration on_changed hook. Rebuild
	# the entire Freehold so top-liege modifiers from the other Houses are not
	# accidentally removed when one House changes its attribute.
	if = {
		limit = {
			house_head ?= {
			government_has_flag = government_is_dragonlord_oligarchy
			top_liege ?= {
				NOT = { has_variable = dragonlord_attribute_recalc_in_progress }
			}
			}
		}
		house_head ?= {
			top_liege ?= {
				set_variable = { name = dragonlord_attribute_recalc_in_progress value = yes }
				dragonlord_recalculate_fourteen_dragonlords_effect = yes
				remove_variable = dragonlord_attribute_recalc_in_progress
			}
		}
	}
}
