# Run once per Dragonlord realm. The title marker survives ruler succession, so
# an enacted council law is never reset to the original game-rule setting.
dragonlord_initialize_noble_laws_effect = {
	if = {
		limit = {
			top_liege = this
			has_government = dragonlord_oligarchy_government
			primary_title = { NOT = { has_variable = dragonlord_noble_laws_initialized } }
		}
		if = {
			limit = { has_game_rule = dragonlord_culture_all }
			add_realm_law_skip_effects = dragonlord_culture_all_law
		}
		else_if = {
			limit = { has_game_rule = dragonlord_culture_valyrian_heritage }
			add_realm_law_skip_effects = dragonlord_culture_valyrian_heritage_law
		}
		else = { add_realm_law_skip_effects = dragonlord_culture_high_valyrian_law }

		if = {
			limit = { has_game_rule = dragonlord_religion_all }
			add_realm_law_skip_effects = dragonlord_religion_all_law
		}
		else_if = {
			limit = { has_game_rule = dragonlord_religion_valyrian }
			add_realm_law_skip_effects = dragonlord_religion_valyrian_law
		}
		else = { add_realm_law_skip_effects = dragonlord_religion_fourteen_flames_law }

		primary_title = { set_variable = { name = dragonlord_noble_laws_initialized value = yes } }
	}
}
