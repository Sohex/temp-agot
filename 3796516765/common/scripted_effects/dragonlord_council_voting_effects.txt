# Counts the frozen voting weight of every living attendee.  The same stance
# boundary used by the agenda forecast applies here: a final support score
# above zero votes in favor, while zero or less votes against.  A tie fails.
dragonlord_council_tally_meeting_votes_effect = {
	clear_variable_list = dragonlord_council_meeting_supporters
	clear_variable_list = dragonlord_council_meeting_opponents
	set_variable = { name = dragonlord_council_meeting_support_votes value = 0 }
	set_variable = { name = dragonlord_council_meeting_opposition_votes value = 0 }
	remove_variable = dragonlord_council_vote_passed

	every_in_list = {
		variable = dragonlord_council_meeting_voters
		limit = {
			is_alive = yes
			has_variable = dragonlord_council_meeting_base_support
			has_variable = dragonlord_council_meeting_vote_weight
		}
		save_scope_as = dragonlord_council_ballot_voter
		if = {
			limit = { dragonlord_council_meeting_final_support > 0 }
			root = {
				add_to_variable_list = {
					name = dragonlord_council_meeting_supporters
					target = scope:dragonlord_council_ballot_voter
				}
				change_variable = {
					name = dragonlord_council_meeting_support_votes
					add = scope:dragonlord_council_ballot_voter.var:dragonlord_council_meeting_vote_weight
				}
			}
		}
		else = {
			root = {
				add_to_variable_list = {
					name = dragonlord_council_meeting_opponents
					target = scope:dragonlord_council_ballot_voter
				}
				change_variable = {
					name = dragonlord_council_meeting_opposition_votes
					add = scope:dragonlord_council_ballot_voter.var:dragonlord_council_meeting_vote_weight
				}
			}
		}
	}

	if = {
		limit = {
			var:dragonlord_council_meeting_support_votes > var:dragonlord_council_meeting_opposition_votes
		}
		set_variable = { name = dragonlord_council_vote_passed value = yes }
	}
}

# Apply a council-authorized law through its ordinary enactment path first so
# its existing on_pass effects still run.  Some imported law groups retain a
# second layer of vanilla cooldown/cost requirements; if that layer refuses a
# law already authorized by the council, force only the database law switch as
# a fallback.  The post-condition flag also drives the result button text.
dragonlord_council_apply_realm_law_effect = {
	add_realm_law = $LAW$
	if = {
		limit = { NOT = { has_realm_law = $LAW$ } }
		add_realm_law_skip_effects = $LAW$
	}
	if = {
		limit = { has_realm_law = $LAW$ }
		set_variable = { name = dragonlord_council_result_law_enacted value = yes }
		# Record the law actually present after enactment in this event chain.
		save_scope_value_as = { name = dragonlord_result_new_law value = flag:$LAW$ }
	}
}

		limit = { var:dragonlord_council_selected_law = flag:dragonlord_religion_fourteen_flames_law }
		dragonlord_council_apply_realm_law_effect = { LAW = dragonlord_religion_fourteen_flames_law }
	}
	else_if = {
		limit = { var:dragonlord_council_selected_law = flag:dragonlord_religion_valyrian_law }
		dragonlord_council_apply_realm_law_effect = { LAW = dragonlord_religion_valyrian_law }
	}
	else_if = {
		limit = { var:dragonlord_council_selected_law = flag:dragonlord_religion_all_law }
		dragonlord_council_apply_realm_law_effect = { LAW = dragonlord_religion_all_law }
	}
}

# Route each agenda category to existing game state only after a successful
# vote.  War motions grant a one-year, single-use authorization rather than
# starting a war directly.
dragonlord_council_enact_passed_agenda_effect = {
	remove_variable = dragonlord_council_result_law_enacted
	if = {
		limit = { var:dragonlord_council_selected_agenda_category = flag:dragonlord_law_agenda }
		dragonlord_council_capture_previous_law_effect = yes
		add_character_flag = dragonlord_council_enacting_law
		dragonlord_council_enact_selected_law_effect = yes
		remove_character_flag = dragonlord_council_enacting_law
	}
	else_if = {
		limit = { var:dragonlord_council_selected_agenda_category = flag:dragonlord_war_agenda }
		dragonlord_council_grant_war_authorization_effect = yes
	}
	else_if = {
		limit = { var:dragonlord_council_selected_agenda_category = flag:dragonlord_development_agenda }
		dragonlord_council_enact_development_agenda_effect = yes
	}
	else_if = {
		limit = { var:dragonlord_council_selected_agenda_category = flag:dragonlord_finance_agenda }
		add_character_flag = dragonlord_council_enacting_law
		dragonlord_council_enact_treasury_budget_effect = yes
		remove_character_flag = dragonlord_council_enacting_law
	}
}
