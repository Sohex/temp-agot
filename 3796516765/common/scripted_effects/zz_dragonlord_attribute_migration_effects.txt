# One-time save migration for Houses which selected an original administrative
# family attribute before Dragonlord-only counterparts existed. Parameter-rich
# attributes preserve their level. Legacy attributes without identifying
# parameters fall back to Giant Lords I and can then be changed normally.
dragonlord_migrate_legacy_fourteen_attribute_effect = {
	if = {
		limit = { NOT = { has_variable = dragonlord_attribute_migrated } }

		if = {
			limit = { has_house_aspiration_parameter = diplomatic_envoys }
			if = { limit = { has_house_aspiration_parameter = increased_marriage_acceptance_3 } set_house_aspiration = { type = dragonlord_diplomatic_envoys level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = increased_marriage_acceptance_2 } set_house_aspiration = { type = dragonlord_diplomatic_envoys level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_diplomatic_envoys level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = staunch_stewards }
			if = { limit = { has_house_aspiration_parameter = improved_siphon_treasury_scheme_3 } set_house_aspiration = { type = dragonlord_staunch_stewards level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = improved_siphon_treasury_scheme_2 } set_house_aspiration = { type = dragonlord_staunch_stewards level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_staunch_stewards level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = learned_philosophers }
			set_house_aspiration = { type = dragonlord_learned_philosophers level = 1 }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = army_commanders }
			if = { limit = { has_house_aspiration_parameter = increased_request_troops_acceptance_3 } set_house_aspiration = { type = dragonlord_army_commanders level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = increased_request_troops_acceptance_2 } set_house_aspiration = { type = dragonlord_army_commanders level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_army_commanders level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = tactical_besiegers }
			if = { limit = { has_house_aspiration_parameter = occupations_give_bonuses_3 } set_house_aspiration = { type = dragonlord_tactical_besiegers level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = occupations_give_bonuses_2 } set_house_aspiration = { type = dragonlord_tactical_besiegers level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_tactical_besiegers level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = political_meddlers }
			if = { limit = { has_house_aspiration_parameter = increased_investment_cap_bonus_3 } set_house_aspiration = { type = dragonlord_political_meddlers level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = increased_investment_cap_bonus_2 } set_house_aspiration = { type = dragonlord_political_meddlers level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_political_meddlers level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = confident_schemers }
			if = { limit = { has_house_aspiration_parameter = improved_raid_estate_scheme_3 } set_house_aspiration = { type = dragonlord_confident_schemers level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = improved_raid_estate_scheme_2 } set_house_aspiration = { type = dragonlord_confident_schemers level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_confident_schemers level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = charismatic_socialites }
			set_house_aspiration = { type = dragonlord_charismatic_socialites level = 1 }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = respected_despots }
			if = { limit = { has_house_aspiration_parameter = swing_scales_cost_reduction_3 } set_house_aspiration = { type = dragonlord_respected_despots level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = swing_scales_cost_reduction_2 } set_house_aspiration = { type = dragonlord_respected_despots level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_respected_despots level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = lofty_architects }
			if = { limit = { has_house_aspiration_parameter = gain_dynasty_prestige_on_building_construction_3 } set_house_aspiration = { type = dragonlord_lofty_architects level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = gain_dynasty_prestige_on_building_construction_2 } set_house_aspiration = { type = dragonlord_lofty_architects level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_lofty_architects level = 1 } }
		}
		else_if = {
			limit = { has_house_aspiration_parameter = faithful_nobles }
			if = { limit = { has_house_aspiration_parameter = increased_clerical_influence_acceptance_3 } set_house_aspiration = { type = dragonlord_faithful_nobles level = 3 } }
			else_if = { limit = { has_house_aspiration_parameter = increased_clerical_influence_acceptance_2 } set_house_aspiration = { type = dragonlord_faithful_nobles level = 2 } }
			else = { set_house_aspiration = { type = dragonlord_faithful_nobles level = 1 } }
		}
		else_if = {
			limit = {
				NOR = {
					has_house_aspiration_parameter = dragon_lords_taming_chance_1
					has_house_aspiration_parameter = dragon_lords_taming_chance_2
					has_house_aspiration_parameter = dragon_lords_taming_chance_3
				}
			}
			set_house_aspiration = { type = dragon_lords level = 1 }
		}

		set_variable = { name = dragonlord_attribute_migrated value = yes }
	}
}
