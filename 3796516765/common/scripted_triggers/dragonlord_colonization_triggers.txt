# The durable marker belongs to the kingdom so it survives appointments and
# inheritance. Its subjects retain their existing Free City family structure.
dragonlord_is_colonial_city_subject_trigger = {
	OR = {
		has_character_flag = dragonlord_colonial_governor_installing
		any_held_title = { tier = tier_kingdom has_variable = dragonlord_colonial_free_city }
		any_liege_or_above = {
			any_held_title = { tier = tier_kingdom has_variable = dragonlord_colonial_free_city }
		}
	}
}
