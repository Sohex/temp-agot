# Independent First Lords may change realm law only while a passed council
# motion is being enacted. Vassals and other governments retain normal laws.
dragonlord_council_direct_law_change_allowed_trigger = {
	OR = {
		NOT = {
			top_liege = this
			government_has_flag = government_is_dragonlord_oligarchy
		}
		has_character_flag = dragonlord_council_enacting_law
	}
}
