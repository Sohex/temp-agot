# The AGOT CB groups already route ordinary player-declared wars through
# agot_generic_war_blocks_trigger.  Add the Dragonlord constitution there so
# every normal offensive group is closed by default; the four council groups
# below bypass only this new clause after a matching vote.
agot_generic_war_blocks_trigger = {
	OR = {
		faith = { has_doctrine_parameter = cannot_declare_war }
		government_has_flag = government_is_uninteractable
		AND = {
			faith = { has_doctrine_parameter = no_same_faith_wars }
			scope:defender.faith = prev.faith
		}
		trigger_if = {
			limit = { exists = scope:defender }
			scope:defender = { government_has_flag = government_is_uninteractable }
		}
		trigger_if = {
			limit = {
				exists = scope:attacker
				exists = scope:defender
			}
			NOT = {
				agot_pentos_braavos_treaty_prohibited_war_declaration_trigger = {
					ACTOR = scope:attacker
					RECIPIENT = scope:defender
				}
			}
		}
		trigger_if = {
			limit = {
				scope:attacker ?= {
					has_diarchy_active_parameter = diarch_must_approve_war_declaration
				}
			}
			NOT = { has_variable = war_support_diarch }
		}
		trigger_if = {
			limit = {
				norvos_are_the_bearded_priests_the_real_power_trigger = yes
				top_liege = title:k_norvos.holder
			}
			scope:defender ?= { has_title = title:d_norvos_high_priesthood }
		}
		AND = {
			top_liege = this
			government_has_flag = government_is_dragonlord_oligarchy
		}
	}
}

dragonlord_council_war_authorization_matches_target_trigger = {
	has_character_flag = dragonlord_council_war_authorized
	has_variable = dragonlord_council_war_authorization_type
	has_variable = dragonlord_council_war_authorization_target
	exists = scope:defender
	var:dragonlord_council_war_authorization_target ?= scope:defender
}

dragonlord_council_has_colonization_authorization_trigger = {
	dragonlord_council_war_authorization_matches_target_trigger = yes
	var:dragonlord_council_war_authorization_type = flag:dragonlord_colonization_war_agenda
}
