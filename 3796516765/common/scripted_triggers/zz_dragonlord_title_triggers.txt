# Late override of AGOT's kingdom-title creation gate. Dragonlord oligarchies
# retain an imperial crown but cannot recreate the kingdom layer below it.
dragonlord_legacy_can_create_kingdom_title = {
	NAND = {
		primary_title.tier <= tier_duchy
		has_variable = agot_pwl_direct
		var:agot_pwl_direct.primary_title.tier = tier_kingdom
	}
	trigger_if = {
		limit = {
			government_has_flag = government_is_free_city
		}
		primary_title.tier != tier_kingdom
	}
	NOT = {
		government_has_flag = government_is_dragonlord_oligarchy
	}
}

# Character-scope equivalent of the custom elite-family status. Future
# Dragonlord interactions and appointments can use this instead of the native,
# globally capped is_powerful_family trigger.
dragonlord_is_fourteen_dragonlords_trigger = {
	exists = house
	house = {
		has_house_modifier = dragonlord_fourteen_dragonlords_house_modifier
	}
	top_liege = {
		government_has_flag = government_is_dragonlord_oligarchy
	}
}

# House scope. The First Lord's House occupies the chief seat above the
# Fourteen, while the ranked Houses occupy the fourteen council seats. Both
# receive the Freehold's family attributes.
dragonlord_is_council_attribute_house_trigger = {
	OR = {
		has_variable = dragonlord_is_chief_family
		has_variable = dragonlord_is_fourteen_dragonlord
	}
}

# Character-scope check for Dragonlord-family effects. This is intentionally
# broader than the council-seat trigger: ordinary Dragonlords
# share the same attribute list; their status controls the additional council
# benefits.
dragonlord_is_registered_dragonlord_member_trigger = {
	exists = house
	house = {
		has_variable = dragonlord_is_registered_family
	}
	top_liege = {
		government_has_flag = government_is_dragonlord_oligarchy
	}
}

# Character belongs to a House whose dignity title was registered by the
# Dragonlord Family creation effect. Checking the title variable, rather than
# culture, government, or rank, prevents ordinary nobles and colonial
# governors from entering Dragonlord-only systems.
dragonlord_is_registered_family_member_trigger = {
	exists = house
	house = {
		any_house_member = {
			any_held_title = {
				is_noble_family_title = yes
				has_variable = dragonlord_family_title
			}
		}
	}
}
