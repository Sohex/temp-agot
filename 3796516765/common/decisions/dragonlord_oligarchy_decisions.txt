adopt_dragonlord_oligarchy_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dragons.dds"
	}
	decision_group_type = major
	desc = adopt_dragonlord_oligarchy_decision_desc
	selection_tooltip = adopt_dragonlord_oligarchy_decision_tooltip
	confirm_text = adopt_dragonlord_oligarchy_decision_confirm

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		NOT = { government_has_flag = government_is_dragonlord_oligarchy }
		highest_held_title_tier >= tier_empire
		faith = faith:valyrian_pan_freehold
		culture = culture:high_valyrian
	}

	is_valid = {
		has_ep3_dlc_trigger = yes
		highest_held_title_tier >= tier_empire
		faith = faith:valyrian_pan_freehold
		culture = culture:high_valyrian
	}

	effect = {
		hidden_effect = {
			add_character_flag = dragonlord_oligarchy_adopted
			change_to_administrative_holding_effect = yes
			change_government = dragonlord_oligarchy_government
			dragonlord_refresh_bureaucracy_opinion_penalty_effect = yes
			primary_title = {
				set_title_name = dragonlord_valyrian_dragonlord_council
				set_definitive_form = yes
			}
			if = {
				limit = {
					NOT = { has_realm_law = dragonlord_council_election_succession_law }
				}
				add_realm_law_skip_effects = dragonlord_council_election_succession_law
			}
				every_vassal_or_below = {
					limit = {
						highest_held_title_tier >= tier_county
				}
					change_to_administrative_holding_effect = yes
					change_government = dragonlord_oligarchy_government
				}

			dragonlord_enforce_single_high_title_effect = yes

			# High-title destruction can promote former sub-vassal counts into direct
			# vassals. Select the Forty Families only after that hierarchy has
			# settled so every ruler who is direct in the restored Freehold is
			# evaluated by the same rule.
			dragonlord_initialize_noble_laws_effect = yes
			dragonlord_establish_family_titles_effect = yes
			add_character_flag = dragonlord_final_hierarchy_family_selection_migrated
			dragonlord_restore_family_primary_titles_effect = yes

			# At full capacity, the strongest fourteen of the forty Dragonlord
			# Families become the first Fourteen Dragonlords.
			dragonlord_recalculate_fourteen_dragonlords_effect = yes
			dragonlord_recalculate_tamed_dragon_influence_effect = yes
			dragonlord_establish_default_provincial_laws_effect = yes
		}
		custom_tooltip = adopt_dragonlord_oligarchy_decision_effect_tooltip
	}

	ai_potential = { always = no }
	ai_will_do = { base = 0 }
}

# Save repair for realms founded while the custom government was not yet
# preregistered. It disappears permanently once a family title exists.
summon_dragonlord_families_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dragons.dds"
	}
	decision_group_type = major
	desc = summon_dragonlord_families_decision_desc
	selection_tooltip = summon_dragonlord_families_decision_tooltip
	confirm_text = summon_dragonlord_families_decision_confirm

	is_shown = {
		top_liege = this
		government_has_flag = government_is_dragonlord_oligarchy
		NOR = {
			any_held_title = {
				is_noble_family_title = yes
				has_variable = dragonlord_family_title
			}
			any_vassal_or_below = {
				any_held_title = {
					is_noble_family_title = yes
					has_variable = dragonlord_family_title
				}
			}
		}
	}

	is_valid = {
		has_ep3_dlc_trigger = yes
	}

	effect = {
		hidden_effect = {
			primary_title = {
				set_title_name = dragonlord_valyrian_dragonlord_council
				set_definitive_form = yes
			}
			if = {
				limit = {
					NOT = { has_realm_law = dragonlord_council_election_succession_law }
				}
				add_realm_law_skip_effects = dragonlord_council_election_succession_law
			}
			dragonlord_initialize_noble_laws_effect = yes
			dragonlord_establish_family_titles_effect = yes
			dragonlord_recalculate_fourteen_dragonlords_effect = yes
			dragonlord_recalculate_tamed_dragon_influence_effect = yes
			dragonlord_initialize_provincial_laws_effect = yes
		}
		custom_tooltip = summon_dragonlord_families_decision_effect_tooltip
	}

	ai_check_interval = 12
	ai_potential = { always = yes }
	ai_will_do = { base = 100 }
}
