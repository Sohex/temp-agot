# The stock AGOT picker only records the selected title. The Dragonlord estate
# uses a noble-family title as a GUI proxy, so also record whether the clicked
# entry is that proxy. This prevents a conventional Dragonpit in the same
# county as the estate from being treated as the Topless Towers.
dragonlord_choose_dragonpit_set_selected_title = {
	scope = character

	is_shown = {
		var:selected_title ?= scope:selected_title
	}

	effect = {
		set_variable = {
			name = selected_title
			value = scope:selected_title
		}
		if = {
			limit = {
				exists = var:dragonlord_topless_towers_picker_title
				scope:selected_title ?= var:dragonlord_topless_towers_picker_title
			}
			set_variable = { name = dragonlord_topless_towers_choice value = 1 }
		}
		else = {
			remove_variable = dragonlord_topless_towers_choice
		}
	}
}
