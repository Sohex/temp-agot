# Refresh cached localization values at the exact moment the player opens the
# native family list. The unconditional GUI validity keeps this harmless in
# non-Dragonlord administrative realms.
dragonlord_refresh_family_ui_counts = {
	scope = character
	is_shown = { always = yes }
	is_valid = { always = yes }
	effect = {
		top_liege = {
			if = {
				limit = {
					top_liege = this
					government_has_flag = government_is_dragonlord_oligarchy
				}
				dragonlord_refresh_family_ui_counts_effect = yes
			}
		}
	}
}
