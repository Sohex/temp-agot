# Admission requirements for Dragonlord Houses. The initial law in each group
# is synchronized once from the matching game rule; later changes require a
# passed Dragonlord Council motion.

dragonlord_noble_culture_laws = {
	default = dragonlord_culture_high_valyrian_law
	cumulative = no
	flag = realm_law
	can_change_law_group = { dragonlord_council_direct_law_change_allowed_trigger = yes }

	dragonlord_culture_high_valyrian_law = {
		can_pass = {
			dragonlord_council_direct_law_change_allowed_trigger = yes
		}
		can_have = { top_liege = this has_government = dragonlord_oligarchy_government }
		can_keep = { top_liege = this has_government = dragonlord_oligarchy_government }
