term_length_laws = {
	default = term_length_long
	cumulative = no
	flag = admin_law
	can_change_law_group = { dragonlord_council_direct_law_change_allowed_trigger = yes }

	term_length_short = {
		can_have = {
			OR = {
				has_government = dragonlord_oligarchy_government
				AND = {
					is_independent_ruler = yes
					government_allows = administrative
					agot_is_historical_free_city = no
				}
			}
		}
		can_keep = {
			OR = {
				has_government = dragonlord_oligarchy_government
				agot_is_historical_free_city = no
			}
		}
		can_pass = {
			dragonlord_council_direct_law_change_allowed_trigger = yes
			is_independent_ruler = yes
			trigger_if = {
				limit = {
					government_allows = administrative
				}
				custom_tooltip = {
					text = admin_law_cooldown_desc
					NOT = { has_variable = admin_law_cooldown }
				}
			}
		}
		on_pass = {
			## Set cooldown
			set_variable = {
				name = admin_law_cooldown
				years = 25
			}
		}
	}
