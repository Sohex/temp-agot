# Normal offensive groups remain blocked by agot_generic_war_blocks_trigger.
# These groups retain AGOT's ordinary restrictions, then accept the exact
# council authorization type and defender selected by the passed motion.
conquest = {
	allowed_for_character = {
		herders_and_tributary_constraints = yes
		NOT = { dragonlord_council_agot_base_war_blocks_trigger = yes }
		trigger_if = {
			limit = {
				top_liege = this
				government_has_flag = government_is_dragonlord_oligarchy
			}
			custom_tooltip = {
				text = dragonlord_council_war_requires_authorization_tt
				dragonlord_council_has_conquest_authorization_trigger = yes
			}
		}
	}
}
