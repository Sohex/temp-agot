# Only the heads of registered Dragonlord Families may be invited to the
# Dragonlord Council activity. The temporary House list prevents duplicate
# entries when one House controls more than one registered family title.
activity_invite_rule_dragonlord_council_members = {
	effect = {
		every_held_title = {
			limit = {
				is_noble_family_title = yes
				has_variable = dragonlord_family_title
			}
			holder.house ?= {
				if = {
					limit = { NOT = { is_in_list = dragonlord_council_activity_houses } }
					add_to_temporary_list = dragonlord_council_activity_houses
				}
			}
		}

		every_vassal_or_below = {
			every_held_title = {
				limit = {
					is_noble_family_title = yes
					has_variable = dragonlord_family_title
				}
				holder.house ?= {
					if = {
						limit = { NOT = { is_in_list = dragonlord_council_activity_houses } }
						add_to_temporary_list = dragonlord_council_activity_houses
					}
				}
			}
		}

		every_in_list = {
			list = dragonlord_council_activity_houses
			house_head ?= {
				if = {
					limit = {
						is_alive = yes
						this != root
					}
					add_to_list = characters
				}
			}
		}
	}
}
