# Dragonlord oligarchy: an AGOT administrative-government derivative for a restored Valyrian Freehold.
dragonlord_oligarchy_government = {
	government_rules = {
		# Dragonlord branches are founded once by the restoration effect. Members
		# cannot keep splitting them through the normal cadet-house decision.
		create_cadet_branches = no
		rulers_should_have_dynasty = yes
		dynasty_named_realms = no
		administrative = yes
		landless_playable = yes
		use_as_base_on_landed = yes
		use_as_base_on_rank_up = yes
		allow_as_base_for_baronies = no
		inherit_from_dynastic_government = no
		sticky_government = yes
		house_aspirations = yes
		noble_families = yes
		admin_allows_holding_multiple_primary_tier_titles = yes
		treasury = yes
		replace_gold_cost_by_treasury = yes
		allow_accolades = yes
	}

	royal_court = any
	blocked_subject_courts = { meritocratic_government celestial_government japan_administrative_government administrative_government steppe_admin_government }

	domicile_type = estate
	main_administrative_tier = duchy
	# Registered Dragonlord Families may legitimately be county-tier.  Keeping
	# the Byzantine duchy minimum made those families lose their governorships
	# and prevented a landless family head from being granted a county again.
	min_appointment_tier = county
	minimum_provincial_maa_tier = duchy
	fallback = 3

	primary_holding = castle_holding
	valid_holdings = { city_holding temple_citadel_holding ruin_holding settlement_holding }
	required_county_holdings = { castle_holding city_holding }
	vassal_contract_group = admin_vassal

	# The decision marks its founder; all direct and indirect vassals then remain eligible.
	can_get_government = {
		NOT = { dragonlord_is_colonial_city_subject_trigger = yes }
		OR = {
			has_character_flag = dragonlord_oligarchy_adopted
			any_liege_or_above = { has_character_flag = dragonlord_oligarchy_adopted }
		}
	}

	character_modifier = {
		levy_size = -0.7
		feudal_government_tax_contribution_mult = -0.5
		feudal_government_levy_contribution_mult = -0.5
		clan_government_tax_contribution_mult = -0.5
		clan_government_levy_contribution_mult = -0.5
		men_at_arms_cap = -3
		men_at_arms_limit = -5
		knight_limit = -5
		vassal_limit = 100
		accolades = -1
		title_creation_cost_mult = -0.05
		mercenary_hire_cost_mult = 2
		# Use the normal 100% MAA upkeep baseline for Dragonlord rulers.
		monthly_treasury_from_liege_mult = -0.15
	}

	top_liege_character_modifier = {
		monthly_treasury_from_vassals = 0.85
		men_at_arms_recruitment_cost = 1
	}

	flags = {
		government_is_administrative
		government_has_influence
		government_has_treasury
		government_has_title_men_at_arms
		# Do not use government_has_county_tier_noble_families here. Besides
		# relaxing the yearly failsafe, that native flag automatically creates a
		# Noble Family whenever a House head is granted a county. Dragonlord
		# county governors should remain ordinary appointed counts; registered
		# Dragonlord Families are created explicitly by the restoration effect.
		# Native powerful families are globally capped and cannot express a
		# government-specific percentile. This government uses the scripted
		# Fourteen Dragonlords ranking and Dragonlord-only attribute copies.
		government_has_fourteen_dragonlords
		government_forbids_kingdom_titles
		government_is_settled
		government_uses_domicile_but_not_adventurer
		government_uses_domain_limit
		government_uses_admin_province_obligations
		government_is_default
		government_is_dragonlord_oligarchy
	}

	mechanic_type = administrative
	color = { 111 24 24 }
	realm_mask_offset = { 0.0 0.01 }
	realm_mask_scale = { 1 1 }
}
