# Dragonlord-only counterparts of the thirteen administrative family attributes.
# They preserve the original parameters and member bonuses. The native
# powerful-family scope remains inactive for ordinary Houses; the scripted
# council trigger reapplies the same full effects to the Chief and selected
# Houses without using the engine's global five-Powerful-Families list.

@dragonlord_admin_level_1_cost = 250
@dragonlord_admin_level_2_cost = 1000
@dragonlord_admin_level_3_cost = 2500
@dragonlord_admin_cooldown_duration = 10
dragonlord_diplomatic_envoys = {
	is_shown = {
		house_head ?= {
			government_has_flag = government_is_dragonlord_oligarchy
			# The ordinary Dragonlord Houses and the Fourteen share this list.
			# The scripted council scope is active only for Fourteen/Chief Houses.
			house ?= { has_variable = dragonlord_is_registered_family }
		}
	}

	on_changed = {
		scope:house = {
			dragonlord_refresh_fourteen_attribute_modifiers_effect = yes
		}
	}

	on_upgraded = {
		scope:house = {
			dragonlord_refresh_fourteen_attribute_modifiers_effect = yes
		}
	}
	level = {
		cost = {
			influence = @dragonlord_admin_level_1_cost
		}
		

powerful_family_top_liege_modifier = {
					negate_diplomacy_penalty_add = 1
					court_grandeur_baseline_add = 1
				}
		powerful_family_member_modifier = {
			domicile_monthly_prestige_mult = 0.1
		}
		
parameters = {
			dragonlord_attribute_diplomatic_envoys = yes
			dragonlord_attribute_level_1 = yes
			diplomatic_envoys = yes
			increased_request_appointment_acceptance_1 = yes
			increased_marriage_acceptance_1 = yes
		}
		
		ai_score = {
			value = 1
			root ?= {
				dynasty ?= {
					if = {
						limit = { has_dynasty_perk = glory_legacy_1 }
						add = 10
					}
					if = {
						limit = { has_dynasty_perk = ep1_culture_legacy_1 }
						add = 10
					}
				}
				if = {
					limit = { ai_compassion > high_positive_ai_value }
					add = {
						value = ai_compassion
						multiply = 0.05
					}
				}
				if = {
					limit = { ai_sociability > 0 }
					add = {
						value = ai_sociability 
						multiply = 0.05
					}
				}
				if = {
					limit = { ai_honor < 0 }
					add = {
						value = ai_honor 
						multiply = 0.05
					}
				}
			}
		}
	}

	level = {
		cost = {
			influence = @dragonlord_admin_level_2_cost
		}
		

powerful_family_top_liege_modifier = {
					negate_diplomacy_penalty_add = 1
					court_grandeur_baseline_add = 2
				}
		powerful_family_member_modifier = {
			domicile_monthly_prestige_mult = 0.2
		}
		
parameters = {
			dragonlord_attribute_diplomatic_envoys = yes
			dragonlord_attribute_level_2 = yes
			diplomatic_envoys = yes
			increased_request_appointment_acceptance_2 = yes
			increased_marriage_acceptance_2 = yes
		}
		
		ai_score = {
			value = 400
		}
	}

	level = {
		cost = {
			influence = @dragonlord_admin_level_3_cost
		}
		

powerful_family_top_liege_modifier = {
					negate_diplomacy_penalty_add = 1
					court_grandeur_baseline_add = 4
				}
		powerful_family_member_modifier = {
			domicile_monthly_prestige_mult = 0.3
			ignore_negative_culture_opinion = yes
		}
		
parameters = {
			dragonlord_attribute_diplomatic_envoys = yes
			dragonlord_attribute_level_3 = yes
			diplomatic_envoys = yes
			increased_request_appointment_acceptance_3 = yes
			increased_marriage_acceptance_3 = yes
		}
		
		ai_score = {
			value = 500
		}
	}
	
	cooldown = { years = @dragonlord_admin_cooldown_duration }

	illustration = "gfx/interface/illustrations/decisions/fp2_decision_struggle_compromise.dds"
}
