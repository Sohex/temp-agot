@dragonlord_attribute_level_1_cost = 250
@dragonlord_attribute_level_2_cost = 1000
@dragonlord_attribute_level_3_cost = 2500

# Save-compatibility only. Earlier development builds could assign this
# placeholder. It is permanently hidden and the annual recalculation replaces
# it with the normal Dragonlord attribute.
dragonlord_unseated_attribute = {
	is_shown = { always = no }
	level = {
		cost = { influence = 0 }
		parameters = { dragonlord_unseated_attribute = yes }
		ai_score = { value = 1000 }
	}
	cooldown = { days = 1 }
	illustration = "gfx/interface/illustrations/decisions/decision_dragons.dds"
}

# Base Dragonlord attribute for every registered House. The complete
# administrative attribute list is shared with the Fourteen and the Chief.
# Ordinary Houses keep the selection available, while the government-specific
# council trigger activates the full effects only for the Chief and Fourteen.
# The taming chance parameters are consumed by zz_dragonlord_taming_values.txt.
dragon_lords = {
	is_default = yes

	is_shown = {
		house_head ?= {
			government_has_flag = government_is_dragonlord_oligarchy
			house ?= {
				has_variable = dragonlord_is_registered_family
			}
		}
	}

	on_changed = {
		scope:house = {
			dragonlord_refresh_fourteen_attribute_modifiers_effect = yes
		}
	}

	on_upgraded = {
		scope:house = {
			dragonlord_refresh_fourteen_attribute_modifiers_effect = yes
		}
	}
	level = {
		cost = { influence = @dragonlord_attribute_level_1_cost }

		# Ordinary Houses can choose this, but only the Chief and Fourteen
		# Houses receive its member bonuses through the scripted council entry
		# point. Do not duplicate them with a native powerful-family modifier.

		parameters = {
			dragonlord_attribute_dragon_lords = yes
			dragonlord_attribute_level_1 = yes
			dragon_lords_taming_chance_1 = yes
			# Display-only parameters mirroring the scripted member modifier.
			dragon_lords_bond_scheme_duration_1 = yes
			dragon_lords_deepen_bond_scheme_duration_1 = yes
			dragon_lords_monthly_influence_1 = yes
		}

		ai_score = { value = 100 }
	}

	level = {
		cost = { influence = @dragonlord_attribute_level_2_cost }

		parameters = {
			dragonlord_attribute_dragon_lords = yes
			dragonlord_attribute_level_2 = yes
			dragon_lords_taming_chance_2 = yes
			dragon_lords_bond_scheme_duration_2 = yes
			dragon_lords_deepen_bond_scheme_duration_2 = yes
			dragon_lords_monthly_influence_2 = yes
		}

		ai_score = { value = 200 }
	}

	level = {
		cost = { influence = @dragonlord_attribute_level_3_cost }

		parameters = {
			dragonlord_attribute_dragon_lords = yes
			dragonlord_attribute_level_3 = yes
			dragon_lords_taming_chance_3 = yes
			dragon_lords_bond_scheme_duration_3 = yes
			dragon_lords_deepen_bond_scheme_duration_3 = yes
			dragon_lords_monthly_influence_3 = yes
		}

		ai_score = { value = 300 }
	}

	cooldown = { years = 10 }
	illustration = "gfx/interface/illustrations/decisions/decision_dragons.dds"
}
