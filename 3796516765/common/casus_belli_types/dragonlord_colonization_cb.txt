# Uses the existing vassalization settlement, gated by the council's single-use
# authorization. No ordinary AGOT CB is replaced.
dragonlord_colonization_cb = {
	icon = dragon_kingdom_conquest_cb
	group = dragonlord_colonization
	target_titles = none
	ticking_war_score_targets_entire_realm = yes
	attacker_score_from_occupation_scale = 150
	attacker_score_from_battles_scale = 150
	defender_score_from_battles_scale = 150

	allowed_for_character = {
		government_has_flag = government_is_dragonlord_oligarchy
		is_independent_ruler = yes
		highest_held_title_tier >= tier_empire
	}
	allowed_against_character = {
		is_alive = yes
		is_landed = yes
		agot_is_independent_ruler = yes
		highest_held_title_tier = tier_kingdom
		in_diplomatic_range = scope:attacker
		NOT = { has_trait = ruin }
		NOT = { has_government = ruins_government }
		NOT = { has_government = wilderness_government }
		NOT = { government_has_flag = government_is_nw }
		NOT = { any_held_title = { is_holy_order = yes } }
		agot_mw_war_valid_during_megawar = yes
	}
	# The permit is consumed on declaration. It must not be rechecked here.
	should_invalidate = {
		OR = {
			scope:attacker = { NOT = { government_has_flag = government_is_dragonlord_oligarchy } }
			scope:defender = { is_independent_ruler = no }
			scope:defender.highest_held_title_tier != tier_kingdom
			scope:attacker.primary_title.tier <= scope:defender.primary_title.tier
		}
	}
	check_attacker_inheritance_validity = no
	check_defender_inheritance_validity = no
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	# Same base prestige cost as the previously authorized kingdom conquest.
	cost = {
		prestige = {
			value = major_prestige_value
			multiply = common_cb_prestige_cost_multiplier
		}
		piety = {
			value = 0
			if = {
				limit = { scope:defender.faith = scope:attacker.faith }
				add = { value = medium_piety_value desc = CB_SAME_FAITH_COST }
			}
			multiply = common_cb_piety_cost_multiplier
		}
	}
	on_declaration = { on_declared_war = yes }
	on_invalidated_desc = msg_vassalization_war_invalidated_message
	on_invalidated = {}
	on_victory_desc = { desc = dragonlord_colonization_cb_victory_desc }
	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		add_legitimacy_attacker_victory_effect = yes
		scope:attacker = { accolade_attacker_war_end_glory_gain_med_effect = yes }
		dragonlord_colonization_victory_effect = yes
		scope:attacker = { add_prestige_experience = medium_prestige_value }
		scope:defender = { add_prestige = { value = medium_prestige_value multiply = -1 } }
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = medium_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_victory_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
		agot_war_victory_effects = yes
	}
	on_white_peace_desc = { desc = dragonlord_colonization_cb_white_peace_desc }
	on_white_peace = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			add_prestige = { value = minor_prestige_value multiply = -1 }
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:attacker
			FAME_BASE = minor_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_white_peace_effect = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}
	on_defeat_desc = { desc = dragonlord_colonization_cb_defeat_desc }
	on_defeat = {
		scope:attacker = {
			show_pow_release_message_effect = yes
			pay_short_term_gold_reparations_effect = { GOLD_VALUE = 3 }
			add_prestige = { value = major_prestige_value multiply = -1 }
		}
		scope:defender = {
			accolade_defender_war_end_glory_gain_high_effect = yes
			add_prestige_war_defender_effect = { PRESTIGE_VALUE = medium_prestige_value }
		}
		modify_allies_of_participants_fame_values = {
			WINNER = scope:defender
			FAME_BASE = major_prestige_value
			IS_RELIGIOUS_WAR = no
			WINNER_ALLY_FAME_SCALE = 1
			LOSER_ALLY_FAME_SCALE = 1
		}
		add_truce_attacker_defeat_effect = yes
		on_lost_aggression_war_discontent_loss = yes
		laamp_as_mercenary_payout_tooltip_effect = yes
	}
	war_name = DRAGONLORD_COLONIZATION_WAR_NAME
	war_name_base = DRAGONLORD_COLONIZATION_WAR_NAME_BASE
	cb_name = dragonlord_colonization_cb
	interface_priority = 60
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150
	ai_score_mult = { value = 1 }
}
