

agot_change_dragonpit_status = {
	save_scope_as = dragon
	scope:owner = {
		if = {
			limit = {
				agot_has_multiple_active_dragonpits = yes
			}
			trigger_event = agot_choose_dragonpit_event.0001
		}
		else = { # Current MDE/AGOT single-pit flow with LoV locations added
			if = {
				limit = {
					is_landless_adventurer = yes
				}
				primary_title = {
					save_scope_as = dragon_pit_location
				}
			}
			else = {
				random_held_county = {
					limit = {
						OR = {
							any_county_province = {
								has_holding_type = castle_holding
								has_building_or_higher = generic_dragon_pit_01
							}
							any_county_province = {
								OR = {
									has_building_or_higher = dragonpit_01 # Historical Version
									has_building = dragonpit_ruins_03 # Fully Restored Ruin Version
									has_building = agot_dragonmont_01 # Dragonmont Version
								}
							}
							any_county_province = {
								OR = {
									has_building = valyria_volcano_01
									has_building = valyria_volcano_02
									has_building = valyria_volcano_03
									has_building = valyria_volcano_04
									has_building = valyria_volcano_05
									has_building = zanpur_volcano_01
									has_building = yovesh_volcano_01
									has_building = last_dragon_tower_01
									has_building_or_higher = procrazion_01
									has_building = valyria_dead_volcano_01
									has_building = valyria_dead_volcano_02
								}
							}
						}
						has_variable = has_dragonkeeper_order
					}
					save_scope_as = dragon_pit_location
				}
			}
			scope:dragon = {
				if = {
					limit = {
						has_character_flag = in_dragonpit
					}
					agot_remove_from_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}
				else_if = {
					limit = {
						has_character_flag = owned_dragon
					}
					agot_send_to_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}
				else_if = {
					limit = {
						NOT = {
							has_character_flag = owned_dragon
						}
					}
					agot_force_wild_dragon_into_dragonpit_effect = {
						DRAGON = root
						DRAGONPIT_HOLDER = scope:owner
						DRAGONPIT_COUNTY = scope:dragon_pit_location
					}
				}
			}
		}
	}
}
