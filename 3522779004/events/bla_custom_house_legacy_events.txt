agot_bla_custom_house_legacy.0099 = {
	hidden = yes

	trigger = {
		is_ai = no
		exists = dynasty
		OR = {
			dynasty = {
				has_dynasty_perk = bla_custom_coin_legacy_5
			}
			has_character_modifier = bla_long_counting_full_coffers
			has_character_modifier = bla_long_counting_deep_reserves
			has_character_modifier = bla_long_counting_house_of_means
		}
	}

	immediate = {
		if = {
			limit = {
				dynasty = {
					has_dynasty_perk = bla_custom_coin_legacy_5
				}
			}

			bla_update_long_counting_effect = yes
		}
		else = {
			remove_character_modifier = bla_long_counting_full_coffers
			remove_character_modifier = bla_long_counting_deep_reserves
			remove_character_modifier = bla_long_counting_house_of_means
		}

		if = {
			limit = {
				OR = {
					has_character_modifier = bla_whisper_network_eyes_within
					has_character_modifier = bla_whisper_network_among_banners
					has_character_modifier = bla_whisper_network_beyond_borders
				}
			}

			random = {
				chance = 20

				if = {
					limit = {
						has_character_modifier = bla_whisper_network_eyes_within
					}

					trigger_event = {
						id = agot_bla_custom_house_legacy.0121
					}
				}
				else_if = {
					limit = {
						has_character_modifier = bla_whisper_network_among_banners
					}

					trigger_event = {
						id = agot_bla_custom_house_legacy.0122
					}
				}
				else_if = {
					limit = {
						has_character_modifier = bla_whisper_network_beyond_borders
					}

					trigger_event = {
						id = agot_bla_custom_house_legacy.0123
					}
				}
			}
		}
	}
}

# 0001 — House Age (framing + renown + splendor + unlock count)
# ------------------------------------------------------------
agot_bla_custom_house_legacy.0001 = {
	type = character_event
	title = agot_bla_custom_house_legacy.0001.t
	desc  = agot_bla_custom_house_legacy.0001.desc
	theme = family
	override_background = study

	trigger = {
		is_ai = no
		has_trait = bla_legacy_customizer
		exists = dynasty
		NOT = { dynasty = { has_variable = bla_custom_house_legacy_done } }
	}

	immediate = {

		 create_character = {
        		template = maester_character
        		location = root.capital_province
        		dynasty = none
        		save_scope_as = new_maester
    		}
		scope:new_maester = {
				add_character_flag = traveling_maester
		}

		dynasty = {
			# reset scores
			set_variable = { name = bla_score_coin value = 0 }
			set_variable = { name = bla_score_banner value = 0 }
			set_variable = { name = bla_score_steel value = 0 }
			set_variable = { name = bla_score_veil value = 0 }
			set_variable = { name = bla_score_candle value = 0 }

			# clear track flags
			remove_variable = bla_custom_track_coin
			remove_variable = bla_custom_track_banner
			remove_variable = bla_custom_track_steel
			remove_variable = bla_custom_track_veil
			remove_variable = bla_custom_track_candle
			remove_variable = bla_custom_track_mosaic

			# clear house guard
			remove_variable = bla_custom_guard_heavy_infantry
			remove_variable = bla_custom_guard_skirmishers
			remove_variable = bla_custom_guard_heavy_cavalry
			remove_variable = bla_custom_guard_light_cavalry
			remove_variable = bla_custom_guard_pikemen
			remove_variable = bla_custom_guard_archers

			# age + framing
			set_variable = { name = bla_custom_age_unlocks value = 0 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
		}
	}

	right_portrait = {
		character = scope:new_maester
		animation = throne_room_writer
	}

	# NEW: 0 unlocked, +0 renown
	option = {
		name = agot_bla_custom_house_legacy.0001.a1
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 0 }
			set_variable = { name = bla_custom_house_is_new value = 1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~100: 1 unlocked, +250 renown
	option = {
		name = agot_bla_custom_house_legacy.0001.a2
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 1 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 250
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~250: 2 unlocked, +1000 renown, -2 splendor level
	option = {
		name = agot_bla_custom_house_legacy.0001.a3
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 2 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 1000
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~500: 3 unlocked, +2250 renown, -2 splendor level
	option = {
		name = agot_bla_custom_house_legacy.0001.a4
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 3 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 2250
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}

	# ~1000: 4 unlocked, +4000 renown, -3 splendor levels
	option = {
		name = agot_bla_custom_house_legacy.0001.a5
		dynasty = {
			set_variable = { name = bla_custom_age_unlocks value = 4 }
			set_variable = { name = bla_custom_house_is_new value = 0 }
			add_dynasty_prestige = 4000
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
			if = { limit = { dynasty_prestige_level > 0 } add_dynasty_prestige_level = -1 }
		}
		trigger_event = { id = agot_bla_custom_house_legacy.0015 days = 1 }
	}
}

	type = character_event
	title = agot_bla_custom_house_legacy.0007.t

	override_background = {
		reference = study
	}

		character = scope:new_maester
		animation = throne_room_writer
	}


	option = {
		name = agot_bla_custom_house_legacy.0007.a

			dynasty = {

				remove_variable = bla_custom_track_coin

					limit = {
						var:bla_score_steel > var:bla_score_coin

						var:bla_score_steel > var:bla_score_candle
					}

					set_variable = {
						name = bla_custom_track_steel
						value = 1
					}
				}
				else_if = {
					limit = {
						var:bla_score_coin > var:bla_score_steel
						var:bla_score_coin > var:bla_score_banner
						var:bla_score_coin > var:bla_score_veil
						var:bla_score_coin > var:bla_score_candle
					}

					set_variable = {
						name = bla_custom_track_coin
						value = 1
					}
				}
				else_if = {
					limit = {
						var:bla_score_banner > var:bla_score_steel
						var:bla_score_banner > var:bla_score_coin
						var:bla_score_banner > var:bla_score_veil
						var:bla_score_banner > var:bla_score_candle
					}

					set_variable = {
						name = bla_custom_track_banner
						value = 1
					}
				}
				else_if = {
					limit = {
						var:bla_score_veil > var:bla_score_steel
						var:bla_score_veil > var:bla_score_coin
						var:bla_score_veil > var:bla_score_banner
						var:bla_score_veil > var:bla_score_candle
					}

					set_variable = {
						name = bla_custom_track_veil
						value = 1
					}
				}
				else_if = {
					limit = {
						var:bla_score_candle > var:bla_score_steel
						var:bla_score_candle > var:bla_score_coin
						var:bla_score_candle > var:bla_score_banner
						var:bla_score_candle > var:bla_score_veil
					}

					set_variable = {
						name = bla_custom_track_candle
						value = 1
					}
				}
				else = {
					set_variable = {
						name = bla_custom_track_mosaic
						value = 1
					}
				}
				
				if = {
					limit = { has_variable = bla_custom_track_mosaic }

					limit = { has_variable = bla_custom_track_coin }

					if = { limit = { var:bla_custom_age_unlocks >= 2 } add_dynasty_perk = bla_custom_coin_legacy_2 }
