##########################################
# Butterwell: Rebuild Whitewalls
##########################################


### EVENT 1 — Begin Clearing Whitewalls
agot_butterwell_whitewalls.0001 = {
    type = character_event
    title = agot_butterwell_whitewalls.0001.t
    desc = agot_butterwell_whitewalls.0001.desc
    theme = stewardship
    left_portrait = { character = root }

    option = {
        name = agot_butterwell_whitewalls.0001.a
        custom_tooltip = agot_butterwell_whitewalls.0001.a.tt

        hidden_effect = {
            add_character_flag = clearing_whitewalls_foundation
            trigger_event = { id = agot_butterwell_whitewalls.0002 days = { 365 545 }}
        }
    }
}

### EVENT 2 — Begin Laying Foundations
agot_butterwell_whitewalls.0002 = {
    type = character_event
    title = agot_butterwell_whitewalls.0002.t
    desc = agot_butterwell_whitewalls.0002.desc
    theme = stewardship
    left_portrait = { character = root }

    trigger = {
        has_character_flag = clearing_whitewalls_foundation
    }

    immediate = {
        remove_character_flag = clearing_whitewalls_foundation
    }

    option = {
        name = agot_butterwell_whitewalls.0002.a
        custom_tooltip = agot_butterwell_whitewalls.0002.a.tt

        hidden_effect = {
            add_character_flag = raising_whitewalls_walls
            trigger_event = { id = agot_butterwell_whitewalls.0003 days = { 730 915 } }
        }
    }
}

### EVENT 3 — Rebuild or Reimagine Whitewalls
agot_butterwell_whitewalls.0003 = {
    type = character_event
    title = agot_butterwell_whitewalls.0003.t
    desc = agot_butterwell_whitewalls.0003.desc
    theme = stewardship
    left_portrait = { character = root }

    trigger = {
        has_character_flag = raising_whitewalls_walls
    }

    immediate = {
        remove_character_flag = raising_whitewalls_walls
    }

    option = {
        name = agot_butterwell_whitewalls.0003.a
        custom_tooltip = agot_butterwell_whitewalls.0003.a.tt

        hidden_effect = {
            province:1950 = {
                set_holding_type = castle_holding
                add_building = castle_03
                add_special_building = agot_butterwell_whitewalls_castle_01
            }
        }
    }

    option = {
        name = agot_butterwell_whitewalls.0003.b
        custom_tooltip = agot_butterwell_whitewalls.0003.b.tt
        trigger = { gold >= 1500 }

     	remove_short_term_gold = 1500

        hidden_effect = {
            province:1950 = {
                set_holding_type = castle_holding
                add_building = castle_03
                add_special_building = agot_butterwell_whitewalls_grand_castle_01
            }
        }
    }
}
