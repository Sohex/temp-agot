agot_redwyne_festival.0001 = {
	type = character_event
	title = agot_redwyne_festival.0001.t
	desc = agot_redwyne_festival.0001.desc
	theme = party

	left_portrait = {
		character = root
		animation = happiness
	}

	immediate = {
		add_prestige = 300
		add_character_modifier = {
            		modifier = hosted_redwyne_festival
            		years = 5
        	}
	}
	option = {
		name = agot_redwyne_festival.0001.a
		custom_tooltip = agot_redwyne_festival.0001.a.tt
		trigger_event = { id = agot_redwyne_festival.0002 days = 30 }
	}
}

		animation = personality_content
	}

		random_list = {
			65 = {
				trigger_event = agot_redwyne_festival.0003
			}
			24 = {
				trigger_event = agot_redwyne_festival.0004
			}
			10 = {
				trigger = { NOT = { has_global_variable = redwyne_cask_found } }
				trigger_event = agot_redwyne_festival.0005
			}
			1 = {
				trigger = { NOT = { has_global_variable = redwyne_egg_found } }
				trigger_event = agot_redwyne_festival.0006	
			}
		}
	}
	option = {
		name = agot_redwyne_festival.0002.a
	}
}
