namespace = agot_strong_claim

##########################################
# House Strong Claim Event
##########################################


agot_strong_claim.0001 = {
	type = character_event
	trigger = {
		game_start_date >= 8170.1.1
		NOT = {
			dynasty = dynasty:dynn_Mudd
			dynasty = dynasty:dynn_Strong
			primary_title = title:d_harrenhal
			primary_title =	title:c_gaines
		}
	}

	immediate = {
		# Scope the correct Strong based on start date with is_alive check
		if = {
			limit = { character:Strong_41 = { is_alive = yes } }
			character:Strong_41 = { save_scope_as = strong_candidate }
		}
		else_if = {
			limit = { character:Strong_42 = { is_alive = yes } }
			character:Strong_42 = { save_scope_as = strong_candidate }
		}
		else_if = {
			limit = { character:Strong_43 = { is_alive = yes } }
			character:Strong_43 = { save_scope_as = strong_candidate }
		}
		else_if = {
			limit = { character:Strong_44 = { is_alive = yes } }
			character:Strong_44 = { save_scope_as = strong_candidate }
		}
		else_if = {
			limit = { character:Strong_63 = { is_alive = yes } }
			character:Strong_63 = { save_scope_as = strong_candidate }
		}

		title:d_harrenhal.holder = {
			save_scope_as = harrenhal_lord
		}
	}

	title = agot_strong_claim.0001.t
	desc = agot_strong_claim.0001.desc
	theme = realm

	left_portrait = {
		character = scope:harrenhal_lord
		animation = personality_rational
	}

	right_portrait = {
		character = scope:strong_candidate
		animation = personality_honorable
	}

	option = {
		name = agot_strong_claim.0001.a
		scope:strong_candidate = {
			get_title = title:c_gaines
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			add_prestige = 1000
			add_gold = 500
			add_legitimacy = 350
			add_trait = loyal
			trigger_event = {
			id = agot_strong_claim.0002
			days = 60
			}
		}
	}

	option = {
		name = agot_strong_claim.0001.b
		# Deny the petition
	}

	option = {
		name = agot_strong_claim.0001.c
		scope:strong_candidate = {
			get_title = title:c_gaines
			create_title_and_vassal_change = {
				type = granted
				save_scope_as = change
				add_claim_on_loss = no
			}
			add_prestige = 1000
			add_gold = 500
			add_legitimacy = 350
			add_trait = loyal
			trigger_event = {
			id = agot_strong_claim.0002
			days = 60
			}
		}
		set_player_character = scope:strong_candidate
	}
}

agot_strong_claim.0002 = {
	type = character_event
	trigger = {
		always = yes
	}

	title = agot_strong_claim.0002.t
	desc = agot_strong_claim.0002.desc
	theme = martial

	left_portrait = {
		character = root
		animation = marshal
	}

	immediate = {
		spawn_army = { 
		name = strong_retinue
		men_at_arms = { 
		type = light_footmen stacks = 1 
		}
		men_at_arms = { 
		type = bowmen stacks = 2 
		}
		men_at_arms = { 
		type = armored_footmen stacks = 1 
		}
		men_at_arms = { 
		type = house_guard stacks = 1 
		}
		location = capital_province 
		}
		add_prestige = 100
	}

	option = {
		name = agot_strong_claim.0002.a
		custom_tooltip = agot_strong_claim.0002.a.tt
	}
}
