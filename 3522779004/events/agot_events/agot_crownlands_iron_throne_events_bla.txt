agot_crownlands_bla.0099 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		is_ruler = yes
		is_landed = yes
		is_available_adult = yes

		any_held_county = {
			title_province = {
				OR = {
					geographical_region = world_westeros_the_crownlands_riverlander
					geographical_region = world_westeros_the_crownlands_crackclaw
					geographical_region = world_westeros_the_crownlands_stormlander
					geographical_region = world_westeros_the_crownlands_valyrian
				}
			}
		}

		NOT = {
			has_character_flag = crownlands_any_event_recently_bla
		}
	}

	immediate = {
		add_character_flag = {
			flag = crownlands_any_event_recently_bla
			years = 3
		}

		random_list = {
			# Generic Crownlands events

			120 = { }
		}
	}
}
