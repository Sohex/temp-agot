			years = 5
		}

		if = {
			limit = {

			}
			add_stress = -10
		}
	}

	option = {
		name = agot_riverlands_bla.0701.b

		trigger = {
			OR = {
				has_trait = gallant
				has_trait = lifestyle_blademaster
				prowess >= 12
			}
		}

		add_prestige = 200
		add_martial_lifestyle_xp = 500

		random_list = {
			60 = {
				random_knight = {
					limit = {
						is_alive = yes
					}

					add_opinion = {
						target = root
						modifier = respect_opinion
						opinion = 20
					}
				}

				if = {
					limit = {
						has_trait = lifestyle_blademaster
					}

					add_trait_xp = {
						trait = lifestyle_blademaster
						value = 50
					}
				}
				else = {
					add_trait = lifestyle_blademaster
				}
			}

			40 = {
				add_trait = wounded_1
			}
		}
	}


	option = {
		name = agot_riverlands_bla.0701.c

		trigger = {
			OR = {
				has_trait = deceitful
				has_trait = arbitrary
				intrigue >= 10
			}
		}

		add_intrigue_lifestyle_xp = 300

		random_list = {

					target = scope:mooton_hedge_knight_bla
					type = favor_hook
				}
			}

			40 = {
				add_short_term_gold = 75
			}
		}
	}

	option = {
		name = agot_riverlands_bla.0701.d
