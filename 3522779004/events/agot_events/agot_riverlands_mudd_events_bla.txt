agot_riverlands_bla.1205 = {
	type = character_event
	title = agot_riverlands_bla.1205.t
	desc = agot_riverlands_bla.1205.desc
	theme = martial

	override_background = {
		reference = armory
	}

	left_portrait = {
		character = root
		animation = admiration
	}

	trigger = {
		is_ai = no
		is_landed = yes
		is_available_adult = yes
		dynasty = dynasty:dynn_Mudd

		has_character_flag = mudd_found_spearhead_bla

		NOT = { has_character_flag = mudd_any_event_recently_bla }
		NOT = { has_character_flag = mudd_rushbreaker_event_done_bla }
		NOT = { has_character_flag = mudd_reforged_rushbreaker_bla }
	}

	immediate = {
		add_character_flag = {
			flag = mudd_any_event_recently_bla
			years = 3
		}

		add_character_flag = {
			flag = mudd_recent_reforge_rushbreaker_bla
			years = 10
		}
		add_character_flag = mudd_rushbreaker_event_done_bla
	}

	option = {
		name = agot_riverlands_bla.1205.a

		trigger = {
			gold >= 250
		}

		remove_short_term_gold = 250
		add_prestige = 200
		add_character_flag = mudd_reforged_rushbreaker_bla

		
		agot_create_artifact_mudd_bronze_spear_effect = { OWNER = root }

		remove_character_modifier = mudd_bronze_spearhead_bla

		add_character_modifier = {
			modifier = mudd_rushbreaker_bla
			years = 15
		}
	}

	option = {
		name = agot_riverlands_bla.1205.b

		add_prestige = 150

		remove_character_modifier = mudd_bronze_spearhead_bla

		agot_create_artifact_mudd_bronze_spearhead_effect = { OWNER = root }

		add_character_modifier = {
			modifier = mudd_spearhead_display_bla
			years = 15
		}
	}

	option = {
		name = agot_riverlands_bla.1205.c

		add_learning_lifestyle_xp = 750
		add_piety = 150

		dynasty = {
			add_dynasty_prestige = 400
		}

		remove_character_modifier = mudd_bronze_spearhead_bla
		remove_character_flag = mudd_found_spearhead_bla
	}
}
