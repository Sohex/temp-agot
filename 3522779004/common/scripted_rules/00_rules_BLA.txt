# root: the owner of the army

	OR = {
		culture = {
			OR = {
				has_innovation = innovation_longboats
				has_cultural_parameter = can_travel_along_rivers
				has_innovation = innovation_rhoynar_sailing
			}
		}

		government_has_flag = government_enables_river_travel

		dynasty ?= {
			has_dynasty_modifier = fp1_legacy_of_piracy_modifier
		}

		dynasty ?= {
			has_dynasty_perk = riverlander_legacy_2
		}

		dynasty ?= {
			has_dynasty_perk = manderly_legacy_4
		}
	}
}
