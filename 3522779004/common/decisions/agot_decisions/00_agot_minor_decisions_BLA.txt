bla_call_upon_house_reserves_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_realm.dds"
	}

	is_shown = {
		is_ai = no
		exists = dynasty
		dynasty = {
			has_dynasty_perk = bla_custom_coin_legacy_5
		}
	}

	is_valid = {
		is_landed = yes

		NOT = {
			has_character_modifier = bla_long_counting_vaults_emptied
		}

		OR = {
			has_character_modifier = bla_long_counting_full_coffers
			has_character_modifier = bla_long_counting_deep_reserves
			has_character_modifier = bla_long_counting_house_of_means
		}
	}

	effect = {
		if = {
			limit = {
				has_character_modifier = bla_long_counting_house_of_means
			}

			add_gold = 1000
			remove_character_modifier = bla_long_counting_house_of_means
		}
		else_if = {
			limit = {
				has_character_modifier = bla_long_counting_deep_reserves
			}

			add_gold = 500
			remove_character_modifier = bla_long_counting_deep_reserves
		}
		else_if = {
			limit = {
				has_character_modifier = bla_long_counting_full_coffers
			}

			add_gold = 250
			remove_character_modifier = bla_long_counting_full_coffers
		}

		add_character_modifier = {
			modifier = bla_long_counting_vaults_emptied
			years = 5
		}
	}

	ai_check_interval = 0
}
