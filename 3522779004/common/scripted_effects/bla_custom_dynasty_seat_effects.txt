# BLA Custom Dynasty Seat Effects

bla_apply_custom_dynasty_seat_effect = {
	if = {
		limit = {
			is_landed = yes
			exists = dynasty
			capital_province = { has_holding = yes }
		}

		# COIN
		if = {
			limit = {
				dynasty = { has_variable = bla_custom_track_coin }
				NOT = { capital_province = { has_building_or_higher = bla_coin_dynasty_seat_01 } }
			}

			if = {
				limit = { dynasty = { var:bla_custom_age_unlocks >= 4 } }
				capital_province = { add_special_building = bla_coin_dynasty_seat_05 }
			}
			else_if = {
				limit = { dynasty = { var:bla_custom_age_unlocks >= 3 } }
				capital_province = { add_special_building = bla_coin_dynasty_seat_04 }
			}
			else_if = {
				limit = { dynasty = { var:bla_custom_age_unlocks >= 2 } }
				capital_province = { add_special_building = bla_coin_dynasty_seat_03 }
			}
			else_if = {
				limit = { dynasty = { var:bla_custom_age_unlocks >= 1 } }
				capital_province = { add_special_building = bla_coin_dynasty_seat_02 }
			}
