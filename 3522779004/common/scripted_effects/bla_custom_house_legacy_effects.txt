bla_update_long_counting_effect = {

	# House reserves have recently been exhausted.
	if = {
		limit = {
			has_character_modifier = bla_long_counting_vaults_emptied
		}

		remove_character_modifier = bla_long_counting_full_coffers
		remove_character_modifier = bla_long_counting_deep_reserves
		remove_character_modifier = bla_long_counting_house_of_means
	}

	# A House of Means
	else_if = {
		limit = {
			gold >= 1500
		}

		remove_character_modifier = bla_long_counting_full_coffers
		remove_character_modifier = bla_long_counting_deep_reserves

		if = {
			limit = {
				NOT = {
					has_character_modifier = bla_long_counting_house_of_means
				}
			}

			add_character_modifier = {
				modifier = bla_long_counting_house_of_means
			}
		}
	}

	# Deep Reserves
	else_if = {
		limit = {
			gold >= 750
		}

		remove_character_modifier = bla_long_counting_full_coffers
		remove_character_modifier = bla_long_counting_house_of_means

		if = {
			limit = {
				NOT = {
					has_character_modifier = bla_long_counting_deep_reserves
				}
			}

			add_character_modifier = {
				modifier = bla_long_counting_deep_reserves
			}
		}
	}

	# Full Coffers
	else_if = {
		limit = {
			gold >= 250
		}

		remove_character_modifier = bla_long_counting_deep_reserves
		remove_character_modifier = bla_long_counting_house_of_means

		if = {
			limit = {
				NOT = {
					has_character_modifier = bla_long_counting_full_coffers
				}
			}

			add_character_modifier = {
				modifier = bla_long_counting_full_coffers
			}
		}
	}

	# Coffers below the minimum threshold.
	else = {
		remove_character_modifier = bla_long_counting_full_coffers
		remove_character_modifier = bla_long_counting_deep_reserves
		remove_character_modifier = bla_long_counting_house_of_means
	}
}
