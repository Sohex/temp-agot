namespace = agot_wildling_events

# Generic Milk Name Removal
agot_wildling_events.0001 = {
	type = character_event
	title = agot_wildling_events.0001.t
	desc = {
		first_valid = {
			#If we're renaming ourself
			triggered_desc = {
				trigger = {
					root = scope:child
				}
				desc = agot_wildling_events.0001.desc.me
			}
			#If we're renaming someone else
			triggered_desc = {
				desc = agot_wildling_events.0001.desc
			}
		}

	}
	theme = family
	override_background = { reference = relaxing_room }
	right_portrait = {
		character = scope:child
		animation = happiness
	}
	#NAMING WIDGET
	widgets = {
		widget = {
			gui = "event_window_widget_name_child"
			container = "dynamic_birth_name"
			controller = name_character
			setup_scope = {
				scope:child = { save_scope_as = name_character_target }
			}
		}
	}
	#Go ahead
	option = {
		name = agot_wildling_events.0001.a
	}
}
