scripted_effect new_it_ruler_decides_what_to_do_with_bastard_effect = {
	# Fired from within scope:story
	if = {
		limit = {
			var:it_ruler != var:original_it_ruler
			NOT = { has_variable = new_it_ruler_given_decision }
		}
		set_variable = new_it_ruler_given_decision
	}
}

scripted_trigger is_human_ai_ruler = {
	is_ai = yes
	is_human = yes
	NOT = { this = title:h_the_iron_throne.holder }
}

agot_events_bastard.0900 = { # agot_expose_true_parentage_decision - Hidden handler
	hidden = yes
	trigger = {
		scope:heritage_secret.secret_target = {
			NOT = { has_character_flag = awaiting_bastardry_result } # to prevent double calls if multiple people reveal within a few days of each other
		}
	}
	immediate = {
		scope:heritage_secret = {
			secret_target = {
				save_scope_as = reveal_scope
				mother ?= { save_scope_as = real_mother } # Will be overwritten if not true
				real_mother ?= { save_scope_as = real_mother }
				real_father = { save_scope_as = real_father }
			}
		}
		scope:reveal_scope = {
			add_character_flag = royal_bastard
			add_character_flag = awaiting_bastardry_result
			if = {
				limit = {
					is_ruler = no
					is_courtier = no
				}
				if = {
					limit = {
						scope:revealer = { is_ai = yes }
					}
					set_employer = scope:revealer
				}
				else_if = {
					limit = {
						location.county.holder ?= {
							is_human_ai_ruler = yes
						}
					}
					location.county.holder = { save_scope_as = ruler }
				}
				else_if = {
					limit = {
						location.county.holder = title:h_the_iron_throne.holder
						any_ruler = {
							top_liege = title:h_the_iron_throne.holder
							is_human_ai_ruler = yes
						}
					}
					random_ruler = {
						limit = {
							top_liege = title:h_the_iron_throne.holder
							is_human_ai_ruler = yes
						}
						save_scope_as = ruler
					}
					set_employer = scope:ruler
				}
				else_if = {
					limit = {
						NOT = { exists = location.county }
						any_ruler = {
							is_human_ai_ruler = yes
						}
					}
					random_ruler = {
						limit = {
							is_human_ai_ruler = yes
						}
						save_scope_as = ruler
					}
					set_employer = scope:ruler
				}
				if = {
					limit = { exists = scope:ruler }
					set_employer = scope:ruler
					set_location_to_default = yes
				}
			}
			if = { # Is parent
				limit = {
					OR = {
						father = root
						mother = root
					}
				}
				root = {
					save_scope_as = parent
					trigger_event = agot_events_bastard.0901 # 'Parent'
				}
			}
			else_if = { # Is employer
				limit = {
					employer = root
				}
				root = {
					save_scope_as = employer
					trigger_event = agot_events_bastard.0912 # Employer
				}
			}
			else = { # Is someone else
				root = {
					save_scope_as = sharer
					trigger_event = agot_events_bastard.0903 # Non-parent
				}
			}
			trigger_event = { # Maintainance if this lasts weirdly long
				id = agot_events_bastard.1019
				days = 365
			}
		}
	}
}

scripted_trigger legitimizing_bastard_changes_succession = {
	trigger_if = {
		limit = {
			any_sibling = {
				OR = {
					this = scope:heir
					any_child = { this = scope:heir }
				}
			}
		}
		always = yes
	}
	trigger_if = {
		limit = {
			any_child = {
				OR = {
					this = scope:heir
					any_child = { this = scope:heir }
				}
			}
		}
		trigger_if = {
			limit = {
				any_child = {
					any_child = { this = scope:heir }
				}
			}
			trigger_if = {
				limit = {
					OR = {
						has_realm_law = male_preference_law
						has_realm_law = male_only_law
					}
					scope:potential_new_heir = { any_sibling = { this = scope:heir.father } }
					scope:potential_new_heir.age > scope:heir.father.age
				}
				always = yes
			}
			trigger_if = {
				limit = {
					OR = {
						has_realm_law = female_preference_law
						has_realm_law = female_only_law
					}
					scope:potential_new_heir = { any_sibling = { this = scope:heir.mother } }
					scope:potential_new_heir.age > scope:heir.mother.age
				}
				always = yes
			}
			trigger_if = {
				limit = {
					has_realm_law = equal_law
					OR = {
						scope:potential_new_heir = { any_sibling = { this = scope:heir.mother } }
						scope:potential_new_heir = { any_sibling = { this = scope:heir.father } }
					}
				}
				trigger_if = {
					limit = { scope:potential_new_heir = { any_sibling = { this = scope:heir.mother } } }
					scope:potential_new_heir.age > scope:heir.mother.age
				}
				trigger_if = {
					limit = { scope:potential_new_heir = { any_sibling = { this = scope:heir.father } } }
					scope:potential_new_heir.age > scope:heir.father.age
				}
				always = yes
			}
		}
		trigger_else = {
			trigger_if = {
				limit = { scope:potential_new_heir.age > scope:heir.age }
				always = yes
			}
		}
	}
}
