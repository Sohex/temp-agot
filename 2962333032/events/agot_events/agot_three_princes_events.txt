agot_three_princes.1002 = { # Harvest Prince Assigned Hidden
	hidden = yes

		exists = title:k_lorath.holder
	}

	immediate = {
		agot_lorath_harvest_prince_election_assign_effect = yes
		title:k_lorath.holder = {
			trigger_event = {
				id = agot_three_princes.1003
				days = 1
			}
			every_vassal_or_below = {
				limit = {
					is_ai = no

						has_title = title:c_west_catch

					}
				}
				trigger_event = {
					id = agot_three_princes.1003

				}
			}
		}
	}

agot_three_princes.9000 = { # Three Princes Maintenance
	hidden = yes

	trigger = {
		exists = title:k_lorath.holder
		OR = {
			has_title = title:d_harvest_principality
			has_title = title:d_fisher_principality
			has_title = title:d_principality_of_the_streets
		}
	}

	immediate = {
		if = {
			limit = { has_title = title:d_harvest_principality }
			agot_lorath_harvest_prince_death_effect = yes
		}
		if = {
			limit = { has_title = title:d_fisher_principality }
			agot_lorath_fisher_prince_death_effect = yes
		}
		if = {
			limit = { has_title = title:d_principality_of_the_streets }
			agot_lorath_prince_of_streets_death_effect = yes
		}
	}
}

agot_three_princes.9001 = { # Harvest Prince Vacated Hidden
	hidden = yes

	trigger = {
		exists = title:k_lorath.holder
	}

	immediate = {
		hidden_effect = {
			agot_lorath_harvest_prince_election_shortlist_effect = yes

			ordered_in_list = {
				list = shortlisted_harvest_prince_candidates
				order_by = agot_lorath_harvest_prince_election_score
				position = 0
				check_range_bounds = no
				save_scope_as = harvest_prince_candidate_one
			}
			ordered_in_list = {
				list = shortlisted_harvest_prince_candidates
				order_by = agot_lorath_harvest_prince_election_score
				position = 1
				check_range_bounds = no
				save_scope_as = harvest_prince_candidate_two
			}
			ordered_in_list = {
				list = shortlisted_harvest_prince_candidates
				order_by = agot_lorath_harvest_prince_election_score
				position = 2
				check_range_bounds = no
				save_scope_as = harvest_prince_candidate_three
			}
			title:c_ruins.holder = {
				trigger_event = {
					id = agot_three_princes.1002
					days = 9
				}
			}
		}
		title:k_lorath.holder = {
			trigger_event = agot_three_princes.1000
			trigger_event = {
				id = agot_three_princes.1001
				days = 8
			}
			every_vassal_or_below = {
				limit = {
					is_ai = no
					agot_is_valid_harvest_prince_elector = yes
				}
				trigger_event = agot_three_princes.1000
				trigger_event = {
					id = agot_three_princes.1001
					days = 8
				}
			}
		}
	}
}
