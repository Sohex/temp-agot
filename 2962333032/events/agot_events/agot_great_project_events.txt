namespace = agot_great_project_events

agot_great_project_events.1000 = {
	type = character_event
	window = big_event_window
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { agot_is_chroyane_special_building_great_project = yes }
				desc = agot_great_project_events.1000.t.special
			}
			triggered_desc = {
				trigger = { scope:great_project ?= { great_project_type = agot_ruins_chroyane } }
				desc = agot_great_project_events.1000.t.chroyane
			}
			desc = agot_great_project_events.1000.t.ruins
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { agot_is_chroyane_special_building_great_project = yes }
				desc = agot_great_project_events.1000.desc.special
			}
			triggered_desc = {
				trigger = { scope:great_project ?= { great_project_type = agot_ruins_chroyane } }
				desc = agot_great_project_events.1000.desc.chroyane
			}
			desc = agot_great_project_events.1000.desc.ruins
		}
	}
	theme = default
	left_portrait = {
		character = scope:onlooker
		animation = admiration
	}
	center_portrait = {
		character = root
		animation = personality_zealous
	}
	right_portrait = {
		character = scope:onlooker_2
		camera = camera_event_scheme_vs_center_positioned_left
		animation = ecstasy
	}
	override_effect_2d = { reference = legend_glow }
	immediate = {
		agot_chroyane_gp_owner_reward_effect = yes
		scope:province ?= { save_scope_as = project_location }
		#Grab some characters who are celebrating and excited
		random_courtier = {
			limit = {
				is_available_adult = yes
				this != root
			}
			save_scope_as = onlooker
		}
		if = {
			limit = {
				NOT = { exists = scope:onlooker }
			}
			random_pool_character = {
				province = root.capital_province
				limit = { is_available_adult = yes }
				save_scope_as = onlooker
			}
		}
		random_courtier = {
			limit = {
				is_available_adult = yes
				this != root
				this != scope:onlooker
			}
			save_scope_as = onlooker_2
		}
		if = {
			limit = {
				NOT = { exists = scope:onlooker_2 }
			}
			random_pool_character = {
				province = root.capital_province
				limit = { is_available_adult = yes }
				save_scope_as = onlooker_2
			}
		}
	}

	option = {
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						trigger = { agot_is_chroyane_special_building_great_project = yes }
						desc = agot_great_project_events.1000.a.special
					}
					triggered_desc = {
						trigger = { scope:great_project ?= { great_project_type = agot_ruins_chroyane } }
						desc = agot_great_project_events.1000.a.chroyane
					}
					desc = agot_great_project_events.1000.a.ruins
				}
			}
		}
	}

	after = {
		# Did you contribute something?
		agot_chroyane_gp_contribution_rewards_effect = yes
	}
}

agot_great_project_events.1010 = {
	type = character_event

				desc = agot_great_project_events.1010.t.chroyane
			}
			desc = agot_great_project_events.1010.t.ruins
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { agot_is_chroyane_special_building_great_project = yes }
				desc = agot_great_project_events.1010.desc.special
			}
			triggered_desc = {
				trigger = { scope:great_project ?= { great_project_type = agot_ruins_chroyane } }
				desc = agot_great_project_events.1010.desc.chroyane
			}
			desc = agot_great_project_events.1010.desc.ruins
		}
	}
	theme = vassal
	left_portrait = {
		character = root
		triggered_animation = {
			trigger = { ai_honor <= 0 }
			animation = scheme
		}
		animation = personality_compassionate
	}
	right_portrait = {
		character = scope:envoy
		animation = obsequious_bow
	}
	lower_right_portrait = {
		character = scope:overlord_scope
	}
	immediate = {
		scope:owner ?= {
			save_scope_as = overlord_scope
			save_scope_as = value_target
		}
		scope:province ?= { save_scope_as = project_location }
		create_character = {
			template = travel_messenger
			location = root.location
			culture = scope:overlord_scope.location.culture
			faith = scope:overlord_scope.location.faith
			gender_female_chance = scope_value_target_soldier_female_chance
			save_scope_as = envoy
		}
	}

	option = {
		name = agot_great_project_events.1010.a
		if = {
			limit = { scope:overlord_scope ?= { is_ai = yes } }
			agot_chroyane_gp_contributor_reward_effect = { OVERLORD = scope:overlord_scope }
		}
	}

	after = {
		scope:envoy ?= {
			if = { # Cleanup
				limit = {
					root = { is_ai = yes }
					is_courtier = no
				}
				silent_disappearance_effect = yes
			}
		}

		# Did you contribute something?
		agot_chroyane_gp_contribution_rewards_effect = yes
	}
}
