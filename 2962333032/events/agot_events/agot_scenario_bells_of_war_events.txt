scripted_effect tourney_winner_standard_payout_effect = {
	if = {
		limit = {
			OR = {
				scope:champion.dynasty = root.dynasty
				scope:champion.age >= 15
			}
		}
		agot_tournament_reward_knighthood_prize = {
			WINNER = scope:champion
			HOST = root
		}
	}
	else = {
		scope:champion = {
			add_trait_xp = {
				trait = squire
				track = knight
				value = 25
			}
		}
	}
	scope:champion = {
		add_gold = 100
		add_prestige = 500
	}
	scope:loser = {
		add_gold = 50
	}
}

scripted_effect grant_blackfyre_prize_effect = {
	scope:champion = {
		set_variable = {
			name = blackfyre_formation_BOW_BM_bypass
			value = yes
			days = 1
		}
	}
	if = {
		limit = {
			any_character_artifact = {
				has_artifact_modifier = vs_blackfyre_modifier
			}
		}
		every_character_artifact = {
			limit = {
				has_artifact_modifier = vs_blackfyre_modifier
			}
			set_variable = {
				name = suppress_artifact_notifications
				value = yes
				days = 1
			}
			set_owner = scope:champion
		}
	}
}

scripted_effect acknowledge_targ_bastard_effect = {
	if = {
		limit = {
			scope:champion.real_father = root
			scope:champion.father != root
		}
		scope:champion = {
			set_father = root
		}
	}
}

					add = agot_historic_outcome_2

					}
				}
			}

				modifier = {
					add = agot_nonhistoric_outcome_2
				}
			}
		}
		if = {
			limit = {
				NOT = {
					exists = scope:champion
				}
			}
			random_list = {
				1 = {
					scope:top_bracket_finalist = {
						save_scope_as = champion
