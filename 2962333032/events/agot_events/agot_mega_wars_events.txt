agot_mega_wars.0003 = {
	type = character_event
	title = agot_mega_wars.0002.t
	desc = {
		first_valid = {
			# Internal war
			triggered_desc = { # you are part of the rebelling faction
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_internal
					agot_mw_root_is_in_mw_faction_trigger = { TARGET = scope:this_scope }
				}
				desc = agot_mega_wars.0003.desc.a.10
			}
			triggered_desc = { # you are a diarch of a participating ruler
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_internal
					exists = scope:mw_diarch_target
					is_diarch_of_target = scope:mw_diarch_target
					scope:mw_crown_story = { is_target_in_variable_list = { name = mw_participants target = root } }
				}
				desc = agot_mega_wars.0003.desc.a.11
			}
			triggered_desc = { # you are a diarch of a participating ruler
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_internal
					exists = scope:mw_diarch_target
					is_diarch_of_target = scope:mw_diarch_target
				}
				desc = agot_mega_wars.0003.desc.a.12
			}
			triggered_desc = { # you are a child Targaryen ruler of Dragonstone
				trigger = {
					agot_mw_dragonstone_is_loyal_to_crown_trigger = yes
					var:mw_is_loyal_to ?= scope:mw_crown
					has_variable = mw_block_other_stances
				}
				desc = agot_mega_wars.0003.desc.a.13
			}
			triggered_desc = { # Claimant war was started and you are the beneficient!
				trigger = {
					scope:war_claimant ?= root
				}
				desc = agot_mega_wars.0003.desc.a.14
			}
			triggered_desc = { # Standard desc
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_internal
				}
				desc = agot_mega_wars.0003.desc.a.15
			}

			# External war
			triggered_desc = { # you are a diarch of a participating ruler
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_external
					exists = scope:mw_diarch_target
					is_diarch_of_target = scope:mw_diarch_target
				}
				desc = agot_mega_wars.0003.desc.a.20
			}
			triggered_desc = { # Standard desc
				trigger = {
					scope:mw_rebel_story.var:mw_source ?= flag:mw_external
				}
				desc = agot_mega_wars.0003.desc.a.21
			}
		}

		triggered_desc = {
			trigger = {
				OR = {
					scope:mw_ruler_scope.var:mw_is_loyal_to ?= scope:mw_crown
					scope:mw_ruler_scope.var:mw_is_loyal_to ?= scope:mw_rebel_leader
				}
			}
			desc = agot_mega_wars.0003.desc.b.1
		}
		first_valid = {
			triggered_desc = { # liege is loyalist
				trigger = {
					scope:mw_ruler_scope ?= { is_at_war_with = liege }
				}
				desc = agot_mega_wars.0003.desc.c.1
			}
			triggered_desc = { # liege is loyalist
				trigger = {
					NOT = { liege = scope:mw_crown }
					agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_crown TYPE = crown LIST_NAME = mw_loyalist_list TARGET = root.liege }
				}
				desc = agot_mega_wars.0003.desc.c.2
			}
			triggered_desc = { # liege is neutral
				trigger = {
					NOR = {
						liege = scope:mw_crown
						liege = scope:mw_rebel_leader
					}
					agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_crown TYPE = crown LIST_NAME = mw_neutral_list TARGET = root.liege }
				}
				desc = agot_mega_wars.0003.desc.c.3
			}
			triggered_desc = { # liege is rebel leader
				trigger = {
					NOT = { liege = scope:mw_rebel_leader }
					agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_rebel_leader TYPE = rebel LIST_NAME = mw_rebel_supporter_list TARGET = root.liege }
				}
				desc = agot_mega_wars.0003.desc.c.4
			}
			triggered_desc = { # liege is independence leader
				trigger = {
					NOR = {
						liege = scope:mw_crown
						liege = scope:mw_rebel_leader
					}
					agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_crown TYPE = crown LIST_NAME = mw_independence_rebel_leader_list TARGET = root.liege }
				}
				desc = agot_mega_wars.0003.desc.c.5
			}
		}
	}

	theme = war
	override_effect_2d = { reference = smoke }
	left_portrait = {
		character = scope:mw_rebel_leader
		animation = aggressive_sword
	}
	right_portrait = {
		character = scope:mw_crown
		animation = aggressive_sword
	}

	trigger = {
		OR = {
			highest_held_title_tier > tier_barony
			trigger_if = {
				limit = { exists = scope:mw_diarch_target }
				is_diarch_of_target = scope:mw_diarch_target
			}
		}
		trigger_if = {
			limit = { NOT = { has_variable = mw_is_loyal_to } }
			NOR = {
				#agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_crown TYPE = crown LIST_NAME = mw_loyalist_list TARGET = root }
				#agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_crown TYPE = crown LIST_NAME = mw_neutral_list TARGET = root }
				agot_mw_in_LIST_of_trigger = { STORY_OWNER = scope:mw_rebel_leader TYPE = rebel LIST_NAME = mw_rebel_supporter_list TARGET = root }
			}
		}
		trigger_else = {
			agot_mw_rebel_leader_trigger = no
		}
		NOT = { scope:mw_crown_story = { is_target_in_variable_list = { name = mw_independence_rebel_leader_list target = root } } }
	}

	immediate = {
		add_character_flag = { flag = mw_had_event_0003 days = 1 }

		if = {
			limit = { has_variable = mw_is_loyal_to }

			var:mw_is_loyal_to = { save_scope_as = mw_is_loyal_to }

			scope:mw_crown_story = {
				remove_list_variable = { name = mw_loyalist_list target = prev }
				remove_list_variable = { name = mw_neutral_list target = prev }
			}
			scope:mw_rebel_story = {
				remove_list_variable = { name = mw_rebel_supporter_list target = prev }
			}
		}

		# Determine the ruler that has the title to be tested by agot_mw_choose_stance_modifier
		if = {
			limit = {
				exists = scope:mw_diarch_target
				is_diarch_of_target = scope:mw_diarch_target
			}
			scope:mw_diarch_target = { save_scope_as = mw_ruler_scope }
		}
		else = {
			save_scope_as = mw_ruler_scope
		}

		if = { # if you already are in rebel list of another ruler, fetch that scope!
			limit = {
				any_ruler = {
					NOR = {
						this = scope:mw_crown
						this = scope:mw_rebel_leader
					}
					any_owned_story = {
						story_type = story_agot_mw_rebel
						is_target_in_variable_list = { name = mw_rebel_supporter_list target = scope:mw_ruler_scope }
					}
				}
			}
			random_ruler = {
				limit = {
					NOR = {
						this = scope:mw_crown
						this = scope:mw_rebel_leader
					}
					any_owned_story = {
						story_type = story_agot_mw_rebel
						is_target_in_variable_list = { name = mw_rebel_supporter_list target = scope:mw_ruler_scope }
					}
				}
				save_scope_as = mw_old_stance_leader
			}
		}

		scope:mw_ruler_scope.var:mw_is_loyal_to ?= { save_scope_as = mw_loyalty_target_1 }


		scope:mw_ruler_scope.var:mw_is_loyal_to ?= { save_scope_as = mw_loyalty_target_1 }


		# Saving these values as scopes saves performance: The choice modifier calls it oftimes and since it is a complex calculation * many times per character * shit ton of characters = performance issues
		scope:mw_ruler_scope ?= {
			save_scope_value_as = { name = s_relations_to_crown value = mw_relations_to_crown_value }
			save_scope_value_as = { name = s_relations_to_rebel_leader value = mw_relations_to_rebel_leader_value }
			scope:mw_crown = { save_temporary_scope_as = target_scope }
			save_scope_value_as = { name = s_close_family_crown value = mw_close_family_value }
			scope:mw_rebel_leader = { save_temporary_scope_as = target_scope }
			save_scope_value_as = { name = s_close_family_rebel_leader value = mw_close_family_value }
			if = {
				limit = { highest_held_title_tier = tier_county }
				save_scope_value_as = { name = s_leverage_unlock value = mw_leverage_unlock_value }
			}
			if = {
				limit = { agot_mw_has_relation_to_TARGET_trigger = { TARGET = scope:mw_crown } }
				save_scope_value_as = { name = s_relation_t_crown value = yes }
			}
			else = {
				save_scope_value_as = { name = s_relation_t_crown value = no }
			}
			if = {
				limit = { agot_mw_has_relation_to_TARGET_trigger = { TARGET = scope:mw_rebel_leader } }
				save_scope_value_as = { name = s_relation_t_rebel_leader value = yes }
			}
			else = {
				save_scope_value_as = { name = s_relation_t_rebel_leader value = no }
			}
		}
		####

		### Check whether one of the stances is positive, else create fallback --> neutral; this prevents a stance being chosen randomly
		scope:mw_ruler_scope ?= { add_character_flag = { flag = mw_fallback_checker days = 1 } }
		random = { # loyalist
			chance = 0
			scope:mw_ruler_scope ?= { add_character_flag =  { flag = mw_loyalist days = 1 } }
			agot_mw_choose_stance_modifier = { TARGET = scope:mw_crown.primary_title }
			modifier = {
				factor = 1000
			}
			modifier = {
				NOT = { agot_mw_show_stance_trigger = { CHECK = loyalist } }
				factor = 0
			}
		}
		random = { # neutral
			chance = 0
			scope:mw_ruler_scope ?= { add_character_flag =  { flag = mw_neutral days = 1 } }
			agot_mw_choose_stance_modifier = { TARGET = scope:mw_ruler_scope.primary_title }
			modifier = {
				factor = 1000
			}
			modifier = {
				NOT = { agot_mw_show_stance_trigger = { CHECK = neutral } }
				factor = 0
			}
		}
		random = { # rebel
			chance = 0
			scope:mw_ruler_scope ?= { add_character_flag = { flag = mw_rebel days = 1 } }
			agot_mw_choose_stance_modifier = { TARGET = scope:mw_rebel_leader.primary_title }
			modifier = {
				factor = 1000
			}
			modifier = {
				NOT = { agot_mw_show_stance_trigger = { CHECK = rebel } }
				factor = 0
			}
		}
		random = { # declare independence
			chance = 0
			scope:mw_ruler_scope ?= { add_character_flag = { flag = mw_independence days = 1 } }
			agot_mw_choose_stance_modifier = { TARGET = root.capital_barony }
			modifier = {
				factor = 1000
			}
			modifier = {
				NOT = { agot_mw_show_stance_trigger = { CHECK = independence } }
				factor = 0
			}
		}
		scope:mw_ruler_scope ?= { remove_character_flag = mw_fallback_checker }
		###
	}

	option = { # Stay loyal
		name = { # Standard text
			trigger = { agot_mw_dragonstone_is_loyal_to_crown_trigger = no }
			text = agot_mega_wars.0003.a
		}
		# Specific texts
		name = {
			trigger = {
				agot_mw_dragonstone_is_loyal_to_crown_trigger = yes
				var:mw_is_loyal_to ?= scope:mw_crown
				has_variable = mw_block_other_stances
				has_character_flag = dragonstone_positive_opinion
			}
			text = agot_mega_wars.0003.a.dragonstone_positive
		}
		name = {
			trigger = {
				agot_mw_dragonstone_is_loyal_to_crown_trigger = yes
				var:mw_is_loyal_to ?= scope:mw_crown
				has_variable = mw_block_other_stances
				NOT = { has_character_flag = dragonstone_positive_opinion }
			}
			text = agot_mega_wars.0003.a.dragonstone_negative
		}
		#

		trigger = {
			agot_mw_show_stance_trigger = { CHECK = loyalist } # show loyalist stance option?
		}

		####################################################################################################

		agot_mw_betray_loyality_var_effect = { TARGET = scope:mw_crown }

		if = { # in case a diarch is getting this event: this sets the stance for the diarch's target
			limit = {
				exists = scope:mw_diarch_target
				is_diarch_of_target = scope:mw_diarch_target # root is diarch of diarch_target
				scope:mw_crown_story = { is_target_in_variable_list = { name = mw_participants target = scope:mw_diarch_target } }
			}

			agot_mw_join_loyalists_effect = { RULER = scope:mw_diarch_target CROWN = scope:mw_crown }
		}
		if = { # this sets the stance for the diarch or for the non-diarch ruler if he is not in an active diarchy
			limit = {
				OR = {
					AND = { # the diarch scope exists
						exists = scope:mw_diarch_target
						OR = { # either root is diarch of target --> make root join same side; or root is not diarch of target --> ruler is free
							is_diarch_of_target = scope:mw_diarch_target # in case a diarch is getting this event
							scope:mw_diarch_target != root # in case the proper ruler gets the event
						}
					}
					NOT = { exists = scope:mw_diarch_target }
				}
			}

			agot_mw_join_loyalists_effect = { RULER = root CROWN = scope:mw_crown }
		}

		if = {
			limit = {
				exists = var:mw_is_faction_member_of
				var:mw_is_faction_member_of = scope:mw_rebel_leader
			}
			custom_tooltip = agot_mega_wars.0003.a.tt
		}

		hidden_effect = {
			scope:mw_crown_story ?= {
				if = {
					limit = {
						var:mw_mode ?= flag:mw_lite
						trigger_if = {
							limit = {
								NOR = {
									scope:mw_ruler_scope.liege = scope:mw_crown
									scope:mw_ruler_scope.liege = scope:mw_rebel_leader
								}
							}
							NOT = { is_target_in_variable_list = { name = mw_loyalist_list target = scope:mw_ruler_scope.liege } }
						}
						trigger_else = { always = no }
					}

					every_in_list = {
						variable = mw_wars

						if = {
							limit = { is_attacker = scope:mw_crown }

							add_attacker = root
						}
						if = {
							limit = { is_defender = scope:mw_crown }

							add_defender = root
						}
					}
				}
			}
		}
		####################################################################################################

		ai_chance = {
			base = 0

			agot_mw_choose_stance_modifier = { TARGET = scope:mw_crown.primary_title }
		}
	}

	option = { # Stay neutral
		name = {
			text = {
				first_valid = {
					triggered_desc = {
						desc = agot_mega_wars.0003.ba
						trigger = {
							exists = scope:mw_old_stance_leader
							agot_mw_is_in_another_rebel_list_bilateral_trigger = { RULER = scope:mw_ruler_scope TARGET = scope:mw_old_stance_leader }
						}
					}
					triggered_desc = {
						desc = agot_mega_wars.0003.bb
						trigger = {
							has_variable = mw_claimant_had_0020
						}
					}
					desc = agot_mega_wars.0003.bc
				}
			}
		}

		trigger = {
			agot_mw_show_stance_trigger = { CHECK = neutral } # show neutral stance option?
		}

		####################################################################################################

		if =  { # Normal 'neutral' stance
			limit = {
				NOT = {
					exists = scope:mw_old_stance_leader # are you already in the list of a rebel leader?
					agot_mw_is_in_another_rebel_list_bilateral_trigger = { RULER = scope:mw_ruler_scope TARGET = scope:mw_old_stance_leader } # you are not in the rebel list of mw_rebel_leader
				}
			}

			if = { # in case a diarch is getting this event: this sets the stance for the diarch's target
				limit = {
					exists = scope:mw_diarch_target
					is_diarch_of_target = scope:mw_diarch_target # root is diarch of diarch_target
					scope:mw_crown_story = { is_target_in_variable_list = { name = mw_participants target = scope:mw_diarch_target } }
				}

				agot_mw_stay_neutral_effect = { RULER = scope:mw_diarch_target CROWN = scope:mw_crown }
			}
			if = { # this sets the stance for the diarch or for the non-diarch ruler if he is not in an active diarchy
				limit = {
					OR = {
						AND = { # the diarch scope exists
							exists = scope:mw_diarch_target
							OR = { # either root is diarch of target --> make root join same side; or root is not diarch of target --> ruler is free
								is_diarch_of_target = scope:mw_diarch_target # in case a diarch is getting this event
								scope:mw_diarch_target != root # in case the proper ruler gets the event
							}
						}
						NOT = { exists = scope:mw_diarch_target }
					}
				}

				agot_mw_stay_neutral_effect = { RULER = root CROWN = scope:mw_crown }
			}
		}
		else = { # you already are in a mw
			custom_tooltip = agot_mega_wars.0003.b.tt
		}

		####################################################################################################

		ai_chance = {
			base = 0

			agot_mw_choose_stance_modifier = { TARGET = scope:mw_ruler_scope.primary_title }
		}
	}

	option = { # Join enemies
		name = agot_mega_wars.0003.c

		trigger = {
			agot_mw_show_stance_trigger = { CHECK = rebel } # show rebel stance option?
		}

		####################################################################################################

		agot_mw_betray_loyality_var_effect = { TARGET = scope:mw_rebel_leader }

		if = { # in case a diarch is getting this event: this sets the stance for the diarch's target
			limit = {
				exists = scope:mw_diarch_target
				is_diarch_of_target = scope:mw_diarch_target # root is diarch of diarch_target
				scope:mw_crown_story = { is_target_in_variable_list = { name = mw_participants target = scope:mw_diarch_target } }
			}

			agot_mw_join_rebels_effect = { RULER = scope:mw_diarch_target REBEL_LEADER = scope:mw_rebel_leader }
		}
		if = { # this sets the stance for the diarch or for the non-diarch ruler if he is not in an active diarchy
			limit = {
				OR = {
					AND = { # the diarch scope exists
						exists = scope:mw_diarch_target
						OR = { # either root is diarch of target --> make root join same side; or root is not diarch of target --> ruler is free. join side
							is_diarch_of_target = scope:mw_diarch_target # in case a diarch is getting this event
							scope:mw_diarch_target != root # in case the proper ruler gets the event
						}
					}
					NOT = { exists = scope:mw_diarch_target }
				}
			}

			agot_mw_join_rebels_effect = { RULER = root REBEL_LEADER = scope:mw_rebel_leader }
		}

		if = {
			limit = {
				scope:war = { using_cb = agot_independence_war }
				highest_held_title_tier > tier_duchy # Kingdoms and Empires
			}

			custom_tooltip = agot_mega_wars.0003.c.tt.independence
		}

		hidden_effect = {
			scope:mw_crown_story ?= {
				if = {
					limit = {
						var:mw_mode ?= flag:mw_lite
						exists = scope:mw_ruler_scope.liege
						scope:mw_ruler_scope.liege != scope:mw_rebel_leader
						trigger_if = {
							limit = {
								NOR = {
									scope:mw_ruler_scope.liege = scope:mw_crown
									scope:mw_ruler_scope.liege = scope:mw_rebel_leader
								}
							}
							is_target_in_variable_list = { name = mw_loyalist_list target = scope:mw_ruler_scope.liege }
						}
						trigger_else = { always = yes }
					}

					every_in_list = {
						variable = mw_wars

						if = {
							limit = { is_attacker = scope:mw_rebel_leader }

							add_attacker = root
						}
						if = {
							limit = { is_defender = scope:mw_rebel_leader }

							add_defender = root
						}
					}
				}
			}
		}

		####################################################################################################

		ai_chance = {
			base = 0

			agot_mw_choose_stance_modifier = { TARGET = scope:mw_rebel_leader.primary_title }
		}
	}

	option = { # Declare independence | This option may only show up if the started war is not an independence war!
		name = {
			text = {
				first_valid = {
					triggered_desc = { # If your liege started the independence war, show that
