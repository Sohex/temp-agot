scripted_trigger available_for_maidens_day = {
	agot_believes_in_the_seven = yes
	is_female = yes
	is_married = no
	is_pregnant = no
	age >= 8
	NOT = { any_child ?= { even_if_dead = yes } }
	NOT = { any_former_spouse ?= { even_if_dead = yes } }
	NOT = { any_former_concubine ?= { even_if_dead = yes } }
	NOT = { any_former_concubinist ?= { even_if_dead = yes } }
	trigger_if = {
		limit = { has_trait = fornicator }
		always = no
	}
}
