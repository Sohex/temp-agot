admiral_maritime = {
	court_position_types = { admiral_court_position }

	on_start = {
		custom_tooltip = task_admiral_maritime_effect_desc
	}

	scaling_employer_modifiers = {
		terrible = {
			supply_limit_mult = 0.02
			domain_tax_mult = 0.01
		}
		poor = {
			supply_limit_mult = 0.04
			domain_tax_mult = 0.02
		}
		average = {
			supply_limit_mult = 0.06
			domain_tax_mult = 0.03
		}
		good = {
			supply_limit_mult = 0.08
			domain_tax_mult = 0.04
		}
		excellent = {
			supply_limit_mult = 0.1
			domain_tax_mult = 0.05
		}
	}

	ai_will_do = {
		value = 1 # Always a good backup

		if = { # Very greedy rulers prefer this over increasing Acceptance, in most cases
			limit = {
				liege_or_court_owner.ai_greed > 0
			}
			add = {
				value = liege_or_court_owner.ai_greed
				divide = 5
			}
		}

		if = {
			# Wealth Focused characters like this
			limit = {
				liege_or_court_owner = {
					has_focus = stewardship_wealth_focus
				}
			}
			add = 15
		}

		if = {
			limit = {
				liege_or_court_owner.primary_title.tier >= tier_duchy
				liege_or_court_owner.gold < 100
			}
			add = 500
		}
		else_if = {
			limit = {
				liege_or_court_owner.gold < 10
			}
			add = 500
		}
	}
}
