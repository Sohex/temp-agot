set_first_mate = {
	scope:activity = {
		random_attending_character = {
			limit = {
				is_alive = yes
				is_ai = yes
				OR = {
					has_relation_friend = root
					has_relation_best_friend = root
				}
				NOT = { this = scope:host }
			}
			alternative_limit = {
				is_alive = yes
				is_ai = yes
				has_trait = lifestyle_mariner
				NOT = { this = scope:host }
			}
			alternative_limit = {
				is_alive = yes
				is_ai = yes
				is_powerful_vassal_of = root
				NOT = { this = scope:host }
			}
			alternative_limit = {
				is_alive = yes
				is_ai = yes
				stewardship > 10
				NOT = { this = scope:host }
			}
			alternative_limit = {
				is_alive = yes
				is_ai = yes
				NOT = { this = scope:host }
			}
			save_scope_as = first_mate
		}
	}
}

agot_sunken_treasure_map_effect = {
	if = {
		limit = {
			NOT = { exists = scope:sunken_ship_province }
		}
		random_province = {
			limit = { is_coastal = yes }
			save_scope_as = sunken_ship_province
		}
	}
	$OWNER$ = {
		save_scope_as = owner
	}
	set_artifact_rarity_illustrious = yes
	create_artifact = {
		name = treasure_map_name
		visuals = scroll
		description = treasure_map_description
		type = miscellaneous
		wealth = scope:wealth
		quality = scope:quality
		modifier = sunken_ship_map_artifact_modifier
		save_scope_as = newly_created_artifact
		decaying = no
	}
	scope:newly_created_artifact ?= {
		set_variable = {
			name = sunken_ship
			value = scope:sunken_ship_province
		}
	}
}
