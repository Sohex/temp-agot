agot_strip_lordship_as_punishment_effect = {
	$STRIPPER$ = {
		create_character_memory = {
			type = agot_stripped_lordship
			participants = {
				LORD = scope:recipient
			}
		}
	}
	$LORD$ = {
		add_character_flag = {
			flag = agot_newly_created_landed_knight
			days = 1
		}
		scope:recipient = {
			every_held_title = {
				limit = { tier = tier_duchy }
				save_scope_as = destroyable_title
				scope:recipient = { destroy_title = scope:destroyable_title }
			}
		}
		vassal_contract_set_obligation_level = { type = special_contract level = 6 }
		add_prestige = major_prestige_loss
		create_character_memory = {
			type = agot_was_stripped_of_lordship
			participants = {
				STRIPPER = scope:actor
			}
		}
		add_opinion = {
			target = $STRIPPER$
			modifier = stripped_my_lordship
			opinion = -100
		}
		add_to_global_variable_list = {
			name = historical_landed_knights_list
			target = scope:recipient
		}
		scope:recipient.house ?= {
			change_house_relation_effect = {
				HOUSE = scope:actor.house
				VALUE = house_relation_damage_medium_value
				REASON = stripped_lordship_reason
				CHAR = scope:actor
				TARGET_CHAR = scope:recipient
				TITLE = scope:dummy_gender
			}
		}
		additional_flavor_check_effect = yes
	}
}
