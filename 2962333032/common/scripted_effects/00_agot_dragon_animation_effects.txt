### Set this in Event Immediate's to set flags
### Current Flags are dragon_roar, dragon_roar_fire, dragon_flying, dragon_flying_fire, dragon_idle, dragon_hover, dragon_hover_fire, dragon_baby_fire.
### Set RIDING_DRAGON to yes if the character is riding a dragon.
agot_set_dragon_animation_flag_effect = {
	if = {
		limit = {
			NOT = {
				has_character_flag = $FLAG$
			}
		}
		add_character_flag = $FLAG$
	}
	if = {
		limit = {
			$RIDING_DRAGON$ = yes
		}
		add_character_flag = currently_riding_dragon
	}
}

agot_clear_dragon_animation_flags = {
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_hover
		}
		remove_character_flag = dragon_hover
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_hover_fire
		}
		remove_character_flag = dragon_hover_fire
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_roar
		}
		remove_character_flag = dragon_roar
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_roar_fire
		}
		remove_character_flag = dragon_roar_fire
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_flying
		}
		remove_character_flag = dragon_flying
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_flying_fire
		}
		remove_character_flag = dragon_flying_fire
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_idle
		}
		remove_character_flag = dragon_idle
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = dragon_baby_fire
		}
		remove_character_flag = dragon_baby_fire
	}
	if = {
		limit = {
			is_alive = yes
			has_character_flag = currently_riding_dragon
		}
		remove_character_flag = currently_riding_dragon
	}
}

### Dragon customizer: show the fire breath animation while the player is
### adjusting the dragonfire colour or smoke genes. Only acts when the flag
### is not already set, so repeated arrow clicks do not restart the animation.
agot_dragon_customizer_show_fire_effect = {
	if = {
		limit = {
			exists = root.var:editing_dragon
			NOT = { root.var:editing_dragon = { has_character_flag = dragon_roar_fire } }
		}
		root.var:editing_dragon = {
			agot_clear_dragon_animation_flags = yes
			agot_set_dragon_animation_flag_effect = { FLAG = dragon_roar_fire RIDING_DRAGON = no }
		}
	}
}

### Dragon customizer: drop back to the idle animation when the player moves
### off the dragonfire genes. Only acts while the fire flag is set, so
### repeated clicks on other controls do not restart the idle animation.
agot_dragon_customizer_clear_fire_effect = {
	if = {
		limit = {
			exists = root.var:editing_dragon
			root.var:editing_dragon = { has_character_flag = dragon_roar_fire }
		}
		root.var:editing_dragon = {
			agot_clear_dragon_animation_flags = yes
		}
	}
}
