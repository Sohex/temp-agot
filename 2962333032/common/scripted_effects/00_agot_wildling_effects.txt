agot_generate_duchy_tribe = {
	$DUCHY$ = { save_scope_as = wilderness_duchy }
	scope:wilderness_duchy = {
		random_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				holder = { has_character_flag = enhanced_freefolk_flag }
			}
			save_scope_as = siren_distribution_title
			create_character = {
				location = character:Local_Rulers.capital_province
				random_traits = yes
				culture = scope:siren_distribution_title.title_province.culture
				faith = scope:siren_distribution_title.title_province.faith
				age = { 16 55 }
				gender = male
				dynasty = generate
				after_creation = {
					save_scope_as = tribal_chief
					get_title = scope:siren_distribution_title
					add_gold = { minor_gold_value_check medium_gold_value_check }
					add_prestige = { minor_prestige_gain medium_prestige_gain }
					add_piety = { minor_piety_gain medium_piety_gain }
				}
			}
			scope:tribal_chief = {
				get_title = scope:wilderness_duchy
			}
		}
		every_in_de_jure_hierarchy = {
			limit = {
				tier = tier_county
				holder = { has_character_flag = enhanced_freefolk_flag }
			}
			save_scope_as = siren_distribution_title
			create_character = {
				location = character:Local_Rulers.capital_province
				random_traits = yes
				culture = scope:siren_distribution_title.title_province.culture
				faith = scope:siren_distribution_title.title_province.faith
				age = { 16 55 }
				gender = male
				dynasty = generate
				after_creation = {
					get_title = scope:siren_distribution_title
					add_gold = { minor_gold_value_check medium_gold_value_check }
					add_prestige = { minor_prestige_gain medium_prestige_gain }
					add_piety = { minor_piety_gain medium_piety_gain }
					create_title_and_vassal_change = {
						type = swear_fealty
						save_scope_as = change
						add_claim_on_loss = no
					}
					change_liege = {
						liege = scope:tribal_chief
						change = scope:change
					}
					resolve_title_and_vassal_change = scope:change
				}
			}
		}
	}
}

agot_update_wildling_resistance = {
	if = {
		limit = {
			holder = {
				government_has_flag = government_is_uninteractable
			}
		}
		if = {
			limit = { has_county_modifier = btw_wildlings_resistance }
			remove_county_modifier = btw_wildlings_resistance
		}
	}
	else_if = {
		limit = { #wildling county, not wildling holder
			culture = { agot_is_wildling_culture = yes }
			holder = {
				NOT = {
					culture = { agot_is_wildling_culture = yes }
				}
			}
			NOT = {
				has_county_modifier = btw_wildlings_resistance
			}
		}
		add_county_modifier = btw_wildlings_resistance
	}
	else_if = {
		limit = { #wildling county, wildling holder, no resistance
			culture = { agot_is_wildling_culture = yes }
			holder = {
				culture = { agot_is_wildling_culture = yes }
			}
			has_county_modifier = btw_wildlings_resistance
		}
		remove_county_modifier = btw_wildlings_resistance
	}
	else_if = {
		limit = { #not wildling county, not wildling holder, no resistance
			NOT = {
				culture = { agot_is_wildling_culture = yes }
			}
			holder = {
				NOT = {
					culture = { agot_is_wildling_culture = yes }
				}
			}
			has_county_modifier = btw_wildlings_resistance
		}
		remove_county_modifier = btw_wildlings_resistance
	}
	else_if = {
		limit = { #not wildling county, wildling holder, no resistance
			culture = { agot_is_wildling_culture = yes }
			holder = {
				culture = { agot_is_wildling_culture = yes }
			}
			has_county_modifier = btw_wildlings_resistance
		}
		remove_county_modifier = btw_wildlings_resistance
	}
}

agot_remove_permafrost_modifiers = {

agot_replace_milk_name_effect = {
	save_scope_as = child
	if = {
		limit = {
			has_character_flag = has_animal_milk_name
		}
		agot_assign_base_animal_name_effect = yes
		if = {
			limit = {
				OR = {
					is_ai = no
					court_owner ?= {
						is_ai = no
						is_close_or_extended_family_of = prev
					}
				}
			}
			if = {
				limit = { is_ai = no }
				save_scope_as = namer
			}
			else = {
				court_owner = { save_scope_as = namer }
			}
			save_scope_as = child
			scope:namer = { trigger_event = agot_wildling_events.0100 }
		}
	}
	else_if = {
		limit = {
			has_character_flag = has_generic_milk_name
			OR = {
				is_ai = no
				court_owner ?= {
					is_ai = no
					is_close_or_extended_family_of = prev
				}
			}
		}
		if = {
			limit = { is_ai = no }
			save_scope_as = namer
		}
		else = {
			court_owner = { save_scope_as = namer }
		}
		save_scope_as = child
		scope:namer = { trigger_event = agot_wildling_events.0001 }
	}
	else_if = {
		limit = {
			has_character_flag = has_generic_milk_name
		}
		hidden_effect = {
			create_character = {
				location = title:c_ruins.holder.location
				culture = scope:child.culture
				faith = scope:child.faith
				gender_female_chance = {
					if = {
						limit = {
							scope:child = { is_female = yes }
						}
						add = 100
					}
					else = {
						add = 0
					}
				}
				age = 32
				save_scope_as = name_thief_char
			}
		}

		change_first_name = { template_character = scope:name_thief_char }
		remove_character_flag = has_generic_milk_name

		hidden_effect = {
			scope:name_thief_char = {
				death = {
					death_reason = death_vanished
				}
			}
		}
	}
}
