		limit = {
			NOR = {
				is_foreign_court_guest_of = character:Targaryen_88
				employer ?= character:Targaryen_88
			}
		}
		visit_court_of = character:Targaryen_88
	}
	add_character_modifier = agot_BOW_tourney_for_squires
	add_character_flag = agot_in_bm_psuedo_activity
	add_to_global_variable_list = { # every_courtier_or_guest does not work in on_death
		name = BOW_psuedo_activity_death_manager
		target = this
	}
}
