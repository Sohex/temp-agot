volantene_rhoynar_confederation_war = {
	icon = claim_cb
	group = event
	ai_only_against_liege = no
	allow_hostages = no
	white_peace_possible = no
	allowed_for_character = {
	}
	allowed_against_character = {
	}
	valid_to_start = {
		always = no
	}
	should_invalidate = {
		OR = {
			NOT = {
				exists = title:d_gar_sagon.holder
			}
			scope:attacker != title:d_gar_sagon.holder
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = volantene_rhoynar_confederation_war_victory_desc
	}

	on_victory = {
		scope:defender = {
			add_prestige = -1000
		}
		scope:attacker = { show_pow_release_message_effect = yes }
		hidden_effect = {
			scope:attacker = {
				add_truce_both_ways = {
					character = scope:defender
					days = 1825
					war = root.war
					result = victory
				}
			}
		}
		scope:attacker = {
			trigger_event = agot_bells_of_war.0051
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		scope:war ?= {
			if = {
				limit = {
					any_war_defender = {
						this = title:c_perforated_mangrove.holder
					}
				}
				title:c_perforated_mangrove = {
					holder = {
						set_variable = {
							name = deposed_primary_title
							value = prev
							months = 1
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
			if = {
				limit = {
					any_war_defender = {
						this = title:c_freedmens_respite.holder
					}
				}
				title:c_freedmens_respite = {
					holder = {
						set_variable = {
							name = deposed_primary_title
							value = prev
							months = 1
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
			if = {
				limit = {
					any_war_defender = {
						this = title:c_lomoyrea.holder
					}
				}
				title:c_lomoyrea = {
					holder = {
						set_variable = {
							name = deposed_primary_title
							value = prev
							months = 1
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
			if = {
				limit = {
					any_war_defender = {
						this = title:c_soytialne.holder
					}
				}
				title:c_soytialne = {
					holder = {
						set_variable = {
							name = deposed_primary_title
							value = prev
							months = 1
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
			if = {
				limit = {
					any_war_defender = {
						this = title:c_pihu_sar.holder
					}
				}
				title:c_pihu_sar = {
					holder = {
						set_variable = {
							name = deposed_primary_title
							value = prev
							months = 1
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		if = {
			limit = {
				has_ep3_dlc_trigger = yes
			}
			custom_tooltip = volantene_rhoynar_confederation_war_adv_tt
			scope:war ?= {
				every_war_defender = {
					limit = {
						is_landed = no
						is_ai = no
					}
					var:deposed_primary_title = {
						add_to_list = target_titles
						save_scope_as = conquered
					}
					# Required scoping for vanilla laamp events
					save_scope_as = title_loser
					save_scope_as = defender
					save_scope_as = adventurer
					scope:attacker =  {
						save_scope_as = new_holder
						save_scope_as = winner
						save_scope_as = attacker
					}
					trigger_event = ep3_laamps.0002
				}
				every_war_defender = {
					limit = {
						is_landed = no
						is_ai = yes
					}
					trigger_event = agot_bells_of_war.9999
				}
			}
		}
		agot_war_victory_effects = yes
	}

	on_defeat_desc = {
		desc = volantene_rhoynar_confederation_war_on_defeat_desc
	}

	on_white_peace_desc = {
		desc = debug_war_white_peace_desc
	}

	on_defeat = {
		scope:attacker = {
			add_prestige = -1000
			show_pow_release_message_effect = yes
		}
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}

		title:c_tianos = {
			change_title_holder = {
				holder = scope:defender
				change = scope:change
			}
		}
		title:d_gar_sagon = {
			change_title_holder = {
				holder = scope:defender
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:defender = {
			add_dread = medium_dread_gain
			trigger_event = agot_bells_of_war.0050
		}
	}

	on_invalidated_desc = msg_invalidate_war_title

	on_invalidated = {}

	check_attacker_inheritance_validity = no
	check_defender_inheritance_validity = no

	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit

	transfer_behavior = transfer

	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "volantene_rhoynar_confederation_war"

	interface_priority = 80

	attacker_wargoal_percentage = 0.8

	attacker_score_from_occupation_scale = 75
	defender_score_from_occupation_scale = 75
	max_attacker_score_from_battles = 100
	max_defender_score_from_battles = 100
}
