	is_shown = {
		NOT = { scope:special_option ?= flag:agot_sailing_type_valyria }
	}

			limit = { has_trait = lifestyle_mariner }
			NOT = {
				has_trait_xp = {
					trait = lifestyle_mariner
					value >= 100
				}
			}
		}
	}
	ai_will_do = {
		value = 0
		add = {
			value = root.ai_energy
		}
		if = {
			limit = {
				culture = { has_cultural_tradition = tradition_seafaring }
			}
			add = 40
		}
	}

	scripted_animation = {
		animation = survey
	}
}
