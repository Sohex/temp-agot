# Increment Options
agot_dc_increment_value_set = {
	scope = character
	effect = {
		agot_dragon_customizer_clear_fire_effect = yes
		set_variable = {
			name = value_to_change
			value = 0.01
		}
	}
}

# gene_dragon_primary_color_value +/-, reset
agot_dragon_skin_color_plus = {
	scope = character
	saved_scopes = { dragon }
	is_valid = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
			trigger_if = {
				limit = {
					root.var:value_to_change = 0.01
					var:gene_dragon_primary_color_value < 1
				}
				var:gene_dragon_primary_color_value.agotadd001 <= 1.00
			}
			trigger_else_if = {
				limit = {
					root.var:value_to_change = 0.05
					var:gene_dragon_primary_color_value < 1
				}
				var:gene_dragon_primary_color_value <= 0.95
			}
			trigger_else = {
				always = no
			}
		}
	}

	effect = {
		agot_dragon_customizer_clear_fire_effect = yes
		if = {
			limit = {
				agot_has_dragon_storage_system_global_list_editor_trigger = yes
			}
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = {
					var:dragon_id ?= root.var:editing_dragon
				}
				save_scope_as = story
			}
			scope:story = {
				change_variable = {
					name = gene_dragon_primary_color_value
					add = root.agotaddvalue_to_change
				}
			}
		}
	}
}

# Reroll the appearance with the same generator used for newly spawned dragons
agot_dragon_customizer_randomize = {
	scope = character
	is_valid = {
		any_in_global_list = {
			variable = gl_dragon_variable_storage
			var:dragon_id ?= root.var:editing_dragon
		}
	}
	effect = {
		agot_dragon_customizer_clear_fire_effect = yes
		if = {
			limit = {
				agot_has_dragon_storage_system_global_list_editor_trigger = yes
			}
			every_in_global_list = {
				variable = gl_dragon_variable_storage
				limit = {
					var:dragon_id ?= root.var:editing_dragon
				}
				save_scope_as = story
			}
			# Remove color values and regenerate
			scope:story = {
				remove_variable = primary_color
				remove_variable = secondary_color
				remove_variable = tertiary_color
				agot_gen_appearance_variables = yes
			}
			# Set new color values
			agot_change_dragon_color_flags = { DRAGON = root.var:editing_dragon }
		}
	}
}
