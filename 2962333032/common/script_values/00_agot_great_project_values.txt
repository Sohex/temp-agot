agot_great_project_construction_time = 1825
agot_great_project_construction_time_multiplier_value = 1

agot_great_project_initial_cost = 1000
agot_great_project_work_force_contribution_cost = 500
agot_great_project_contribution_cost = 1000
agot_great_project_material_contribution_cost = 750

agot_great_projects_contribution_ai_will_do_base_value = {
	value = -25

	if = {
		limit = { this = scope:great_project.great_project_owner }
		add = 25
	}
	# you are a governor
	if = {
		limit = { is_governor = yes }
		add = {
			value = 20
			if = {
				limit = { has_trait = ambitious }
				add = 20
			}
		}
	}

	# reduce if the project is far away
	if = {
		limit = {
			"capital_province.squared_distance(scope:great_project.great_project_location)" > squared_distance_large
		}
		add = -25
	}

	add = {
		value = ai_honor
		multiply = 0.5
		min = 0 # don't reduce if dishonorable
	}
	add = {
		value = -0.25
		multiply = ai_greed
	}
	add = {
		value = 0.25
		multiply = ai_energy
	}

	if = {
		limit = { is_vassal_or_below_of = scope:great_project.great_project_owner }
		if = {
			limit = { has_trait = loyal }
			add = 20
		}
		else_if = {
			limit = { has_trait = disloyal }
			add = -20
		}
	}
	if = {
		limit = { is_at_war = yes }
		multiply = 0
	}
}
