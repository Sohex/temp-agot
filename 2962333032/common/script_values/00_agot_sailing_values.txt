great_fleet_great_projects_contribution_ai_will_do_base_value = {
	value = -25

	if = {
		limit = { this = scope:great_project.great_project_owner }
		add = 50
	}
	if = {
		limit = {
			is_liege_or_above_of = scope:great_project.great_project_owner
			NOT = { this = scope:great_project.great_project_owner }
		}
		multiply = 0
	}
	if = {
		limit = {
			any_held_county = {
				any_county_province = {
					has_holding = yes
					is_coastal = yes
				}
			}
		}
		add = 25
	}
	if = {
		limit = {
			culture = { has_cultural_parameter = ironborn_great_fleet_bonuses }
			scope:great_project.great_project_owner = {
				culture = { has_cultural_parameter = ironborn_great_fleet_bonuses }
			}
			is_vassal_or_below_of = scope:great_project.great_project_owner
		}
		add = 50
	}

	if = {
		limit = {
			is_vassal_or_below_of = scope:great_project.great_project_owner
		}
		if = {
			limit = { has_trait = loyal }
			add = 20
		}
		else_if = {
			limit = { has_trait = disloyal }
			add = -20
		}
	}
	if = {
		limit = {
			is_at_war = yes
		}
		multiply = 0
	}
}
