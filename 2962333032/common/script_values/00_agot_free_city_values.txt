candidate_score_value = { # Irritatingly, we can't just add the votes for an individual character because it would be add = number_of_election_votes = { scope:term_limit_country } and we can't have the two equals
	value = 0
	save_temporary_scope_as = candidate
	scope:term_limit_country = {
		every_elector = {
			save_temporary_scope_as = person
			if = {
				limit = { scope:candidate = { has_election_vote_of = { who = scope:person title = scope:term_limit_country } } }
				add = gold # Since gold is the vote power of each elector, we just add that
			}
		}
	}
}

war_vote_declaration = {
	value = 0
	add = {
		value = ai_boldness
	}
	add = {
		value = ai_energy
		multiply = 0.5
	}
	if = {
		limit = {
			opinion = {
				target = scope:recipient
				value = { 50 100 }
			}
		}
		subtract = 50
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:recipient
				value = { 0 50 }
			}
		}
		subtract = 25
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:recipient
				value = { -50 0 }
			}
		}
		add = 25
	}
	else = {
		add = 50
	}
	if = {
		limit = {
			opinion = {
				target = scope:actor
				value = { 50 100 }
			}
		}
		add = 50
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:actor
				value = { 0 50 }
			}
		}
		add = 25
	}
	else_if = {
		limit = {
			opinion = {
				target = scope:actor
				value = { -50 0 }
			}
		}
		subtract = 25
	}
	else = {
		subtract = 50
	}
	if = {
		limit = {
			scope:actor.current_military_strength >= scope:recipient.current_military_strength
		}
		add = 50
	}
	else = {
		subtract = 50
	}
	if = {
		limit = {
			scope:actor.max_military_strength >= scope:recipient.max_military_strength
		}
		add = 50
	}
	else = {
		subtract = 50
	}
}

free_city_term_length = {
	value = 0

	#Historical Terms
	if = {
		limit = { holder = { has_realm_law = term_length_historical } }
		switch = {
			trigger = this
			title:e_narrow_sea = { value = 3 }
			title:e_daoryrdembos = { value = 7 }
			title:k_lys = { value = 4 }
			title:k_myr = { value = 6 }
			title:k_tyrosh = { value = 9 }
			title:k_braavos = { value = -1 }
			title:k_pentos = { value = -1 }
			title:k_norvos = { value = 10 }
			title:k_lorath = { value = 7 }
			title:k_qohor = { value = 13 }
			fallback = { value = 5 } # If you somehow get the historical law and weren't supposed to; fallback to 5
		}
	}
	#Short Terms
	if = {
		limit = { holder = { has_realm_law = term_length_short } }
		value = 5
	}
	#Long Terms
	if = {
		limit = { holder = { has_realm_law = term_length_long } }
		value = 10
	}
	#Life Terms
	if = {
		limit = {
			holder = { has_realm_law = term_length_life }
		}
		value = -1
	}
	#Dictatorship
	else_if = {
		limit = {
			has_title_law = dictatorship_succession_law
		}
		value = -3
	}
	# Delayed Due to War
	else_if = {
		limit = { holder = { has_character_flag = election_delayed_till_war_ends } }
		value = -2
	}
}

free_city_election_progress = {
	if = {
		limit = {
			exists = var:years_since_election
		}
		value = var:years_since_election
		divide = free_city_term_length  # Calculate the progress towards the term length
		multiply = 100  # To get the percentage
	}
	else = {
		value = 0
	}
}

free_city_election_progress_loc = {
	value = free_city_term_length
	subtract = var:years_since_election
}

norvos_pv_overthrow_threshold_value = {
	value = 0
	# Grab our sum total of powerful vassals.
	root = {
		every_powerful_vassal = { add = 1 }
		every_powerful_family = {
			limit = {
				house_head ?= { is_powerful_vassal = no }
			}
			add = 1
		}
	}
	# Halve it.
	multiply = 0.5
	# Now round down to even us out.
	floor = yes
	# And add +1 to get a simple majority.
	add = 1
}

norvos_pv_overthrow_tally_value = {
	value = 0
	every_in_list = {
		variable = coup_pv_supporters_list
		limit = {
			OR = {
				is_powerful_vassal_of = root.liege
				house ?= { is_powerful_family = yes }
			}
			is_alive = yes
		}
		add = 1
	}
}
