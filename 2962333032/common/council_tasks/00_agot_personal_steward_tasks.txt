task_personal_steward_court_politics = {
	position = councillor_personal_steward

	skill = diplomacy
	task_type = task_type_general
	task_progress = task_progress_infinite

	effect_desc = task_personal_steward_court_politics_effect_desc

	asset = {
		icon = "gfx/interface/icons/council_task_types/task_castellan_default_5.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}

	council_owner_modifier = {
		name = task_court_politics_modifier
		diplomacy = 1
		scale = personal_steward_task_court_politics_scale
	}

	monthly_on_action = court_politics_setup

	ai_will_do = {
		value = 0
		if = {
			limit = {
				diplomacy >  very_high_skill_rating
			}
			add = 1
		}
		else_if = {
			limit = {
				diplomacy >  high_skill_rating
			}
			add = 0.5
		}

		if = { # Keep doing it once it is started
			limit = {
				is_performing_council_task = task_court_politics
			}
			add = 1000
		}

		scope:councillor_liege = {
			if = {
				limit = {
					diplomacy <  decent_skill_rating
				}
				multiply = 3
			}
			else_if = {
				limit = {
					diplomacy <  medium_skill_rating
				}
				multiply = 2
			}
		}
	}
}

task_collect_pirate_domicile_taxes = {
	default_task = no
	position = councillor_steward

	task_type = task_type_general
	task_progress = task_progress_infinite

	asset = {
		icon = "gfx/interface/icons/council_task_types/task_collect_pirate_domicile_taxes.dds"
		background = "gfx/interface/buttons/button_round_bg.dds"
		frame = "gfx/interface/buttons/button_round_frame.dds"
		glow = "gfx/particles/halo.dds"
	}

	is_shown = {
		has_council_position = councillor_steward
	}
	effect_desc = {
		desc = task_collect_taxes_effect_desc
	}

	is_valid_showing_failures_only = {
		tgp_natural_disaster_councillor_available_for_task_trigger = yes
		OR = {
			custom_tooltip = {
				text = you_have_one_of_the_following_domiciles
				scope:councillor_liege = { agot_is_landless_pirate_character = yes }
			}
		}
	}

	council_owner_modifier = {
		name = task_collect_pirate_domicile_taxes
		domicile_monthly_gold_mult = 1
		scale = steward_collect_taxes_total_scale
	}
	council_owner_modifier = {
		name = task_domicile_generic_build_speed_inc_modifier
		domicile_build_speed = -0.1
		scale = steward_collect_taxes_total_scale
	}

	ai_will_do = {
		value = 1 # Always a good backup

		if = { # Very greedy rulers prefer this over increasing Acceptance, in most cases
			limit = {
				scope:councillor_liege.ai_greed > 0
			}
			add = {
				value = scope:councillor_liege.ai_greed
				divide = 5
			}
		}

		if = {
			# Wealth Focused characters like this
			limit = {
				scope:councillor_liege = {
					has_focus = stewardship_wealth_focus
				}
			}
			add = 15
		}

		if = {
			limit = {
				scope:councillor_liege.primary_title.tier >= tier_duchy
				scope:councillor_liege.gold < 100
			}
			add = 500
		}
		else_if = {
			limit = {
				scope:councillor_liege.gold < 10
			}
			add = 500
		}
		if = { # Conquerors want more money
			limit = {
				scope:councillor_liege = {
					has_variable = conqueror
				}
			}
			add = 500
		}
	}
}
