is_available_for_reconstruction_generic = {
	age >= 14
	is_available = yes
}

is_available_for_reconstruction_mc = {
	is_available_for_reconstruction_generic = yes
	OR = {
		has_character_flag = rebuilding_moat_cailin
		has_character_flag = started_rebuilding_moat_cailin
	}
}

is_available_for_reconstruction_hh = {
	is_available_for_reconstruction_generic = yes
	OR = {
		has_character_flag = rebuilding_harrenhal
		has_character_flag = started_rebuilding_harrenhal
	}
}
