agot_kingdom_tier_is_real_kingdom_trigger = {
	has_government = lp_feudal_government
	highest_held_title_tier >= tier_kingdom
	OR = {
		agot_flavour_palatinate_is_kingdom_trigger = { CHAR = this }
		agot_is_independent_ruler = yes
	}
}

agot_can_create_duchy_title = {
	NAND = {
		primary_title.tier = tier_county
		has_variable = agot_pwl_direct
		var:agot_pwl_direct.primary_title.tier = tier_duchy
	}
	is_human = yes
}

agot_can_create_kingdom_title = {
	NAND = {
		primary_title.tier <= tier_duchy
		has_variable = agot_pwl_direct
		var:agot_pwl_direct.primary_title.tier = tier_kingdom
	}
	trigger_if = {
		limit = {
			government_has_flag = government_is_free_city
		}
		primary_title.tier != tier_kingdom
	}
}

agot_can_create_empire_title = {
	trigger_if = {
		limit = {
			NOT = { government_has_flag = government_is_free_city }
		}
		any_held_title = {
			tier = tier_kingdom
			count >= 2
		}
	}
}
