can_have_norvoshi_regency_succession_law_trigger = {
	norvos_are_the_bearded_priests_the_real_power_trigger = yes
	government_has_flag = government_is_norvos
	OR = {
		any_vassal_or_below = { has_title = title:d_norvos_high_priesthood }
		top_liege ?= {
			any_vassal_or_below = { has_title = title:d_norvos_high_priesthood }
		}
	}
}

prohibit_figurehead_for_norvos_ceremonial_priest_trigger = {
	NOT = {
		$RECIPIENT$ = {
			primary_title ?= title:d_norvos_high_priesthood
		}
	}
}

has_two_or_more_sinner_traits_in_norvos_trigger = {
	$CHARACTER$ = {
		any_character_trait = {
			count >= 2
			OR = {
				this = trait:generous
				this = trait:lazy
				this = trait:arbitrary
				this = trait:pilgrim
				this = trait:dull
			}
		}
	}
}

has_two_or_more_pious_traits_in_norvos_trigger = {
	$CHARACTER$ = {
		any_character_trait = {
			count >= 2
			OR = {
				this = trait:just
				this = trait:shrewd
				this = trait:greedy
				this = trait:diligent
			}
		}
	}
}

norvos_are_the_bearded_priests_the_real_power_trigger = {
	exists = global_var:norvos_has_priesthood_shadowy_leadership
}

norvos_ceremonial_monarch_trigger_check = {
	norvos_are_the_bearded_priests_the_real_power_trigger = yes
	has_tgp_dlc_trigger = yes
	scope:actor = {
		government_has_flag = government_is_norvos
		OR = {
			top_liege.primary_title.var:administrative_ui_special_title.holder = title:d_norvos_high_priesthood.holder
			primary_title.var:administrative_ui_special_title.holder = title:d_norvos_high_priesthood.holder
		}
	}
	scope:recipient = {
		this != scope:actor
		top_liege = scope:actor.top_liege
		primary_title = title:d_norvos_high_priesthood
	}
}
