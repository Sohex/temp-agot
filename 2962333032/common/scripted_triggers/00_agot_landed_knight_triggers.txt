agot_can_revoke_lordship_punishment_trigger = {
	$STRIPPER$ = {
		is_human = yes
		is_ai = no
	}
	$STRIP_TARGET$ = {
		is_imprisoned_by = $STRIPPER$
		is_vassal_of = $STRIPPER$
		is_landed = yes
		OR = {
			highest_held_title_tier = tier_duchy
			highest_held_title_tier = tier_county
		}
		NOT = {
			vassal_contract_has_flag = has_landed_knight_contract
		}
	}
}
