	is_adult = yes
	OR = {
		capital_province ?= {
			has_building_or_higher = common_tradeport_01
		}
		house = {
			has_house_modifier = owns_big_ship
		}
		house = {
			has_house_modifier = owns_small_ship
		}
	}
}
