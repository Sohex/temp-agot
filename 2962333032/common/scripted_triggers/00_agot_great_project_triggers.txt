agot_is_chroyane_restoration_great_project = {
	scope:great_project ?= {
		OR = {
			agot_is_chroyane_ruins_great_project = yes
			agot_is_chroyane_special_building_great_project = yes
		}
	}
}

	scope:great_project ?= {
		OR = {

			great_project_type = agot_ruins_tul_poy
			great_project_type = agot_ruins_shroudfall
			great_project_type = agot_ruins_the_visage
			great_project_type = agot_ruins_denllal

	}
}
