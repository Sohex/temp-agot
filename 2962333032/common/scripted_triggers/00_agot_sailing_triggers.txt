is_valid_held_province_for_sailing = {
	scope:host = {
		any_realm_province = {
			this = root
			is_coastal = yes
			has_holding = yes
			OR = {
				barony.county.holder = scope:host
				barony.county.holder = {
					any_liege_or_above = { this = scope:host }
				}
			}
		}
	}
}

						has_sailing_special_building = yes

					}
