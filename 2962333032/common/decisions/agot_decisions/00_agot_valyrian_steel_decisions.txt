reforge_valyrian_steel_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	desc = reforge_valyrian_steel_decision_desc
	sort_order = 76

	decision_group_type = artifact

	cost = { }


	is_shown = {
		any_character_artifact = {
			has_variable = valyrian_steel
			NOT = { has_artifact_modifier = vs_dawn_modifier }
			trigger_if = {
				limit = { root.dynasty ?= dynasty:dynn_Gardener }
				NOT = { has_artifact_modifier = vs_longthorn_ruined_modifier }
			}
		}
	}

	is_valid = {
		employs_court_position = court_smith_court_position
		custom_tooltip = {
			text = reforge_valyrian_steel_excellent_aptitude
			any_court_position_holder = {
				type = court_smith_court_position
				aptitude = {
					court_position = court_smith_court_position
					value >= 4
				}
			}
		}
		custom_tooltip = {
			text = reforge_valyrian_steel_additional_requirements
			any_court_position_holder = {
				type = court_smith_court_position
				OR = {
					faith = { has_doctrine_parameter = reforge_valyrian_steel }
					has_character_flag = can_reforge_valyrian_steel
				}
			}
		}
		prestige >= massive_prestige_value
		short_term_gold >= 1000
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	ai_potential = { always = no }

	effect = {
		trigger_event = agot_valyrian_steel.8000

		custom_tooltip = reforge_valyrian_steel_decision.tt
	}
}

recover_brightroar_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	desc = recover_brightroar_decision_desc
	sort_order = 1

	decision_group_type = artifact

	cost = {
		gold = 500
	}

	is_shown = {
		dynasty = dynasty:dynn_Lannister
		has_global_variable = brightroar_valyria
	}

	is_valid_showing_failures_only = {
		is_available_adult = yes
		is_landed = yes
		gold > 500
		is_imprisoned = no
	}

	cooldown = { days = 3975 }

	ai_check_interval_by_tier = {
		barony = 0
		county = 60
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	ai_potential = {
		dynasty = dynasty:dynn_Lannister
		is_landed = yes
		is_available_adult = yes
		gold > 500
	}

	ai_will_do = {
		base = 20
		modifier = {
			add = 100
			AND = {
				current_date >= 8292.1.1
				this = character:Lannister_1
				gold >= 500
			}
		}
		modifier = {
			add = -100
			NOT = { gold >= 500}
		}
		modifier = {
			add = 30
			gold >= 500
		}
		modifier = {
			add = 20
			has_trait = ambitious
		}
		modifier = {
			add = 20
			has_trait = greedy
		}
		modifier = {
			add = 50
			has_trait = adventurer
		}
	}

	effect = {
		if = {
			limit = {
				character:Lannister_5 ?= {
					is_alive = yes
					is_landed = no
					is_available_adult = yes
					is_imprisoned = no
				}
			}
			character:Lannister_5 = {
				save_scope_as = explorer
			}
		}
		else = {
			create_character = {
				location = root.capital_province
				gender_female_chance = 0
				template_character = root
				faith = root.faith
				culture = root.culture
				random_traits = yes
				dynasty = none
				age = { 18 55 }
				martial = { 3 15 }
				save_scope_as = explorer
			}
		}

		trigger_event = {
			id = agot_valyrian_steel.4000
		}
	}
}

decision_reforge_longthorn = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_misc.dds"
	}
	title = {
		first_valid = {
			triggered_desc = {
				trigger = { dynasty ?= dynasty:dynn_Gardener }
				desc = decision_reforge_longthorn
			}
			desc = decision_reforge_longthorn_alt
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { dynasty ?= dynasty:dynn_Gardener }
				desc = decision_reforge_longthorn_desc
			}
			desc = decision_reforge_longthorn_desc_alt
		}
	}
	sort_order = 1
	decision_group_type = artifact
	cost = {
		gold = 750
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 60
		kingdom = 60
		empire = 60
		hegemony = 60
	}

	is_shown = {
		any_character_artifact = {
			has_artifact_modifier = vs_longthorn_ruined_modifier
		}
		NOT = { exists = global_var:longthorn_restored }
	}

	is_valid = {
		trigger_if = {
			limit = { dynasty ?= dynasty:dynn_Gardener }
			house = character:Gardener_1.house
			has_title = title:c_highgarden
			has_title = title:d_highgarden
			prestige_level >= 4
		}
		custom_tooltip = {
			text = reforge_longthorn_good_aptitude
			any_court_position_holder = {
				type = court_smith_court_position
				aptitude = {
					court_position = court_smith_court_position
					value >= 3
				}
			}
		}
	}

	effect = {
		set_global_variable = longthorn_restored
		random_character_artifact = {
			limit = { has_artifact_modifier = vs_longthorn_ruined_modifier }
			save_scope_as = ruined_longthorn
		}
		destroy_artifact = scope:ruined_longthorn
		show_as_tooltip = { agot_create_artifact_vs_longthorn_effect = { OWNER = ROOT } }
		trigger_event = agot_valyrian_steel.5000
	}
}
