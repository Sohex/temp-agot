agot_buy_ship = {
	picture = {
		reference = "gfx/interface/illustrations/activity_header_backgrounds/activity_agot_sailing.dds"
	}

	sort_order = 80

	desc = agot_buy_ship_desc
	selection_tooltip = agot_buy_ship_tooltip
	confirm_text = agot_buy_ship_confirm_text_desc

	is_shown = {
		is_lowborn = no
		has_access_to_a_big_ship = no
		NOT = { government_has_flag = government_is_pirate }
		NOT = { government_has_flag = government_is_nw }
		NOT = { government_has_flag = government_is_uninteractable }
		is_ai = no
		agot_can_sail = yes
	}

	widget = {
		gui = "decision_view_widget_option_list_generic"
		controller = decision_option_list_controller
		decision_to_second_step_button = "SELECT_ACTION"

		item = {
			value = agot_buy_ship_small_ship
			current_description = agot_buy_ship_small_ship_desc
			localization = agot_buy_ship_small_ship
			icon = "gfx/interface/icons/government_types/pirate_no_dlc_government.dds"
			is_shown = {
				has_access_to_a_small_ship = no
			}
			is_valid = {
				has_access_to_a_small_ship = no
			}
		}
		item = {
			value = agot_buy_ship_big_ship
			current_description = agot_buy_ship_big_ship_desc
			localization = agot_buy_ship_big_ship
			icon = "gfx/interface/icons/government_types/pirate_government.dds"
			is_shown = {
				has_access_to_a_big_ship = no
			}
			is_valid = {
				has_access_to_a_big_ship = no
			}
		}
	}

	effect = {
		hidden_effect = {
			if = {
				limit = { scope:agot_buy_ship_big_ship = yes }
				if = {
					limit = {
						has_access_to_a_diving_ship = yes
					}
					hidden_effect = {
						house = { remove_house_modifier = owns_small_ship_diving }
					}
					house = { add_house_modifier = owns_big_ship_diving }
				}
				else_if = {
					limit = {
						has_access_to_a_small_ship = yes
					}
					hidden_effect = {
						house = { remove_house_modifier = owns_small_ship }
					}
					house = { add_house_modifier = owns_big_ship }
				}
				else = {
					house = { add_house_modifier = owns_big_ship }
				}
			}
			else = {
				house = { add_house_modifier = owns_small_ship }
			}
		}
		if = {
			limit = { has_access_to_a_ship = no }
			custom_tooltip = agot_buy_ship_effect_1_tooltip
			custom_tooltip = agot_buy_ship_effect_2_tooltip
		}
		else = {
			show_as_tooltip = {
				if = {
					limit = {
						has_access_to_a_diving_ship = yes
					}
					hidden_effect = {
						house = { remove_house_modifier = owns_small_ship_diving }
					}
					house = { add_house_modifier = owns_big_ship_diving }
				}
				else_if = {
					limit = {
						has_access_to_a_small_ship = yes
					}
					house = { add_house_modifier = owns_big_ship }
				}
				else = {
					house = { add_house_modifier = owns_big_ship }
				}
			}
		}
		set_variable = bought_the_ship
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = { scope:agot_buy_ship_small_ship = yes }
				add = agot_sailing_small_ship_cost
			}
			else_if = {
				limit = { scope:agot_buy_ship_big_ship = yes }
				add = agot_sailing_big_ship_cost
				if = {
					limit = { has_access_to_a_small_ship = yes }
					subtract = agot_sailing_small_ship_cost
				}
			}
		}
	}
	ai_potential = {
		always = no
	}
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 0
		hegemony = 0
	}
	ai_will_do = {
		base = 0
	}
}

agot_buy_diving_bell = {
	picture = {
		reference = "gfx/interface/illustrations/activity_header_backgrounds/activity_agot_sailing.dds"
	}

	sort_order = 80

	desc = agot_buy_diving_bell_desc
	selection_tooltip = agot_buy_diving_bell_tooltip
	confirm_text = agot_buy_diving_bell_confirm_text_desc

	is_shown = {
		is_lowborn = no
		has_access_to_a_ship = yes
		has_access_to_a_diving_ship = no
		NOT = { government_has_flag = government_is_pirate }
		NOT = { government_has_flag = government_is_nw }
		NOT = { government_has_flag = government_is_uninteractable }
	}

	effect = {
		if = {
			limit = {
				house = { has_house_modifier = owns_small_ship }
			}
			hidden_effect = {
				house = { remove_house_modifier = owns_small_ship }
			}
			house = { add_house_modifier = owns_small_ship_diving }
		}
		else = {
			hidden_effect = {
				house = { remove_house_modifier = owns_big_ship }
			}
			house = { add_house_modifier = owns_big_ship_diving }
		}
		custom_tooltip = agot_buy_diving_bell_effect_tooltip
	}

	cost = {
		gold = 3000
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 12
		empire = 12
		hegemony = 12
	}

	ai_potential = { always = yes }
	ai_will_do = {
		base = 0
		modifier = {
			add = 20
			short_term_gold >= 5000
		}
		modifier = {
			add = 20
			has_trait = lifestyle_mariner
		}
		modifier = {
			add = 20
			any_character_artifact = { has_variable = sunken_ship }
		}
		modifier = {
			add = 20
			any_held_county = {
				any_county_province = {
					OR = {
						has_building_or_higher = common_tradeport_01
						has_sailing_special_building = yes
					}
				}
			}
		}
		modifier = {
			factor = 0
			is_at_war = yes
		}
		modifier = {
			factor = 0
			short_term_gold <= 3500
		}
	}
}
