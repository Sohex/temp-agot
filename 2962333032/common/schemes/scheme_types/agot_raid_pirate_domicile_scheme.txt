agot_raid_pirate_domicile = {
	# Basic Setup
	skill = intrigue
	desc = "agot_raid_pirate_domicile_desc_general"
	success_desc = "RAID_ESTATE_SUCCESS_DESC"
	discovery_desc = "RAID_ESTATE_DISCOVERY_DESC"
	icon = icon_raid_estate_scheme
	illustration = "gfx/interface/illustrations/event_scenes/docks.dds"
	category = political
	target_type = character
	cooldown = { years = 10 }

	# Parameters
	speed_per_skill_point = t2_spsp_owner_value
	speed_per_target_skill_point = t2_spsp_target_value
	spymaster_speed_per_skill_point = t2_spsp_owner_spy_value
	target_spymaster_speed_per_skill_point = t2_spsp_target_spy_value
	tier_speed = 1
	base_progress_goal = t3_base_phase_length_value
	maximum_secrecy = 80
	base_maximum_success = t2_base_max_success_value
	phases_per_agent_charge = 1
	success_chance_growth_per_skill_point = t3_scgpsp_value

	is_secret = yes
	maximum_breaches = 5

	# Core Triggers
	allow = {
		is_adult = yes
		is_imprisoned = no
		government_has_flag = government_is_pirate
		any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
		scope:target = {
			any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
			is_adult = yes
			is_imprisoned = no
			government_has_flag = government_is_pirate
			domicile ?= { is_domicile_type = agot_pirate_ship }
		}
		trigger_if = {
			limit = {
				scope:target = {
					is_landed = yes
				}
			}
			scope:owner.domicile.domicile_location.county.holder != scope:target
		}
	}

	valid = {
		scope:owner = {
			any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
		}
		scope:target = {
			any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
			domicile ?= { is_domicile_type = agot_pirate_ship }
			trigger_if = {
				limit = {
					is_landed = yes
				}
				scope:owner.domicile.domicile_location.county.holder != scope:target
			}
		}
	}

	# Agents
	agent_leave_threshold = -25
	agent_join_chance = {
		base = 0
		ai_agent_join_chance_basic_suite_modifier = yes
		ai_agent_join_chance_hostile_grievous_modifier = yes
	}
	valid_agent = { is_valid_agent_standard_trigger = yes }
	agent_groups_owner_perspective = { courtiers guests scripted_relations }

	odds_prediction = {
		add = hostile_scheme_base_odds_prediction_target_is_char_value
		add = odds_skill_contribution_intrigue_value
		add = agent_groups_owner_perspective_value
		# same as odds_dispute_border_scheme_misc_value but target is a character
		add = {
			value = 0
			desc = scheme_odds.misc_modifiers
			if = {
				limit = {
					scope:target.domicile.domicile_location.county.holder = {
						government_has_flag = government_is_pirate
						has_realm_law = pirate_tax_rate_04
					}
				}
				add = -3
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_01
					}
				}
				add = -2
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_02
					}
				}
				add = -4
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_03
					}
				}
				add = -6
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_aftcastle_04
					}
				}
				add = -8
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_aftcastle_05
					}
				}
				add = -10
			}
			if = {
				limit = {
					scope:target.domicile = {
						has_domicile_building_or_higher = agot_forecastle_aftcastle_06
					}
				}
				add = -12
			}
		}
		min = 0
	}

	# Base Chances
	base_success_chance = {
		base = 0
		scheme_type_skill_success_chance_modifier = { SKILL = INTRIGUE }
		hostile_scheme_base_chance_modifier = yes

		# Family Feud
		house_feud_hostile_scheme_success_modifier = yes
		# Diarchs are better at schemes within their liege's realm
		diarch_scheming_within_realm_bonus_modifier = yes
		# Modifiers
		# Context
		modifier = {
			scope:target.domicile.domicile_location.county ?= scope:owner.domicile.domicile_location.county
			desc = RAID_PIRATE_SHIP_IN_SUBREALM_DESC
			add = 5
		}
		modifier = {
			scope:target.domicile.domicile_location.county.holder = {
				government_has_flag = government_is_pirate
				has_realm_law = pirate_tax_rate_04
			}
			add = -3
			desc = RAID_PIRATE_SHIP_IN_HIGH_PORT_RATE_DESC
		}
		modifier = {
			exists = scope:target.house
			scope:actor.top_liege.house ?= scope:target.house
			desc = RAID_ESTATE_TOP_LIEGE_DESC
			add = -15
		}
		modifier = {
			scope:target = {
				agot_is_landless_pirate_character = yes
				has_perk = enduring_hardships_perk
			}
			desc = RAID_PIRATE_SHIP_ENDURING_HARDSHIPS_PERK_DESC
			add = -15
		}
		# Domicile
		modifier = { # FORECASTLE
			scope:target.domicile ?= { has_domicile_building_or_higher = agot_forecastle_01 }
			desc = OWNER_DOMICILE_FORCASTLE_DESC
			add = {
				value = -2.5
				scope:target.domicile = {
					if = {
						limit = { has_domicile_building_or_higher = agot_forecastle_aftcastle_06 }
						multiply = 6
					}
					else_if = {
						limit = { has_domicile_building_or_higher = agot_forecastle_aftcastle_05 }
						multiply = 5
					}
					else_if = {
						limit = { has_domicile_building_or_higher = agot_forecastle_aftcastle_04 }
						multiply = 4
					}
					else_if = {
						limit = { has_domicile_building_or_higher = agot_forecastle_03 }
						multiply = 3
					}
					else_if = {
						limit = { has_domicile_building_or_higher = agot_forecastle_02 }
						multiply = 2
					}
				}
			}
		}
	}
	base_secrecy = {
		add = secrecy_base_value
		add = countermeasure_apply_secrecy_maluses_value
	}

	# On Actions
	on_start = {
		set_variable = {
			name = apply_countermeasures
			value = flag:calculating
		}
		add_scheme_starting_opportunities_intrigue_effect = yes
		# Fallback agents — a balanced config if you start the scheme from script.
		if = {
			limit = {
				scheme_owner = {
					NOT = { has_variable = agents_added }
				}
			}
			add_agent_slot = agent_scout
			add_agent_slot = agent_scout
			add_agent_slot = agent_muscle
			add_agent_slot = agent_infiltrator
			add_agent_slot = agent_lookout
		}
	}
	on_phase_completed = {
		suppress_scheme_follow_up_event_till_input_given_effect = yes
		agot_raid_pirate_domicile_prep_effect = yes
		cap_schemes_and_fire_reminders_effect = yes
	}
	on_hud_click  = { agot_raid_pirate_domicile_prep_effect = yes }
	on_semiyearly = {
		if = {
			limit = {
				scheme_owner = { is_ai = yes }
			}
			agot_raid_pirate_domicile_prep_effect = yes
		}
	}
	on_monthly = {
		hostile_scheme_monthly_discovery_chance_effect = yes
		if = {
			limit = {
				NOT = { exists = scope:discovery_event_happening }
			}
			fire_generic_contract_scheme_on_going_events_effect = yes
		}
	}

	## Invalidation
	on_invalidated = {
		scheme_target_character = { save_scope_as = target }
		scheme_owner = {
			save_scope_as = owner
			if = {
				limit = {
					scope:target = {
						is_alive = no
					}
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = agot_raid_pirate_domicile_invalidated_title
					left_icon = scope:target
					custom_tooltip = scheme_target_not_alive
				}
			}
			else_if = { # Not Pirate Domicile
				limit = {
					NOT = {
						scope:target = {
							any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
						}
					}
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = agot_raid_pirate_domicile_invalidated_title
					left_icon = scope:target
					custom_tooltip = scheme_target_not_agot_pirate
				}
			}
			else_if = {
				limit = {
					any_held_title = { agot_is_pirate_domicile_title_trigger = yes }
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = agot_raid_pirate_domicile_invalidated_title
					left_icon = scope:target
					custom_tooltip = scheme_owner_not_agot_pirate
				}
			}
		}
	}
}
