agot_remove_realms = {
	trigger = {
		NOT = { has_game_rule = agot_essos_enabled }
	}
	effect = {
		every_living_character = {
			limit = {
				location ?= { geographical_region = world_westeros }
			}
			add_character_flag = {
				flag = sent_relevant_death_event
				days = 3
			}
		}
		agot_remove_realms_scenario_effect = yes
		if = {
			limit = { agot_realm_narrow_sea_enabled = no }
			agot_remove_realm_effect = { REALM = title:e_narrow_sea }
		}
		if = {
			limit = { agot_realm_daoryrdembos_enabled = no }
			agot_remove_realm_effect = { REALM = title:e_daoryrdembos }
		}
		if = {
			limit = { agot_realm_rhoyne_enabled = no }
			agot_remove_realm_effect = { REALM = title:e_rhoyne }
		}
		title:c_unknown.holder = {
			every_held_title = {
				set_color_from_title = title:c_unknown
			}
			every_courtier_or_guest = {
				death = {
					death_reason = death_vanished
				}
			}
		}
	}
}
