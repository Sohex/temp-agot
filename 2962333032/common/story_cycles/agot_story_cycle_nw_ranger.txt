nw_ranger_story = {
	on_setup = {
		story_owner = {
			if = {
				limit = { has_character_flag = fr_is_a_lowborn }
				set_to_lowborn = yes
			}
			if = {
				limit = { NOT = { has_character_flag = nw_ranger_title } }
				add_character_flag = nw_ranger_title
			}
			save_scope_as = new_fr
			title:d_nw_landless_first_ranger.holder = { save_scope_as = new_first_ranger }
			title:k_the_wall.holder = { trigger_event = agot_nw_ranger_events.0002 }
			#Set Up AI Values (Should be applied on the player too in case the AI takes over)
			set_variable = {
				name = remaining_contract_counter_nw
				value = 0
			}
			set_variable = {
				name = remaining_employer_counter_nw
				value = 0
			}
			#Add a ranger to the camp
			if = {
				limit = {
					title:k_the_wall.holder = {
						any_courtier = {
							agot_valid_ranger_camper = yes
						}
					}
				}
				save_scope_as = first_ranger
				trigger_event = {
					id = agot_nw_ranger_events.0016
					days = 1
				}
			}
			#Inform playable house member
			if = {
				limit = {
					has_multiple_players = no
					has_dlc_feature = landless_playable
					exists = house
					house = {
						any_house_member = {
							is_playable_character = yes
							is_ai = no
						}
					}
				}
				save_scope_as = first_ranger
				house = {
					random_house_member = {
						limit = {
							is_playable_character = yes
							is_ai = no
							agot_can_be_ruler = yes
						}
						trigger_event = agot_nw_ranger_events.0020
					}
				}
			}
			#Add a courtier to avoid crash upon attempting to appoint a councillor
			agot_nw_ranger_on_creation = yes
		}
	}

	on_end = {
		story_owner = {
			if = {
				limit = {
					is_alive = yes
					has_character_flag = nw_ranger_title
				}
				remove_character_flag = nw_ranger_title
			}
		}
	}

	on_owner_death = {
		scope:story = { end_story = yes }
	}

			trigger = {
				story_owner = {
					any_courtier = {
						count <= 7

					}

					NOT = { has_character_flag = new_ranger_arrived }

					exists = title:k_the_wall.holder
					NOT = { this = title:k_the_wall.holder }
				}

				title:k_the_wall.holder = {
					any_courtier = {
						agot_valid_ranger_camper = yes
					}
				}
			}

					save_scope_as = first_ranger

			}
		}

					}

					exists = title:k_the_wall.holder
					NOT = { this = title:k_the_wall.holder }
				}

				title:k_the_wall.holder = {
					any_courtier = {
						agot_valid_ranger_camper = yes
					}
				}
			}
			effect = {
				story_owner = {
					save_scope_as = first_ranger

						chance = {
							value = 0
							if = {
								limit = {
									scope:first_ranger = {
										any_courtier = {
											count < 2
											agot_valid_ranger_camper = yes
										}

last_ranger_story = {
	on_setup = {
		story_owner = {
			save_scope_as = first_ranger
			primary_title = {
				set_variable = post_nw_quest
				set_title_name = d_nw_landless_first_ranger_broken_nw
				set_coa = title:d_nw_landless_first_ranger
			}
			title:c_castle_black = {
				set_variable = post_nw_quest_blocker
				set_variable = {
					name = agot_last_ranger_cooldown
					years = 10
				}
			}
		}
		scope:story = {
			set_variable = {
				name = ex_ranger_title
				value = scope:first_ranger.primary_title
			}
		}
	}

	on_end = {
		story_owner = {
			if = {
				limit = { is_alive = yes }
				if = {
					limit = { has_character_flag = already_arrived_to_the_northern_lord }
					remove_character_flag = already_arrived_to_the_northern_lord
				}
				if = {
					limit = { has_character_flag = agot_nw_ranger_events_0100_cooldown }
					remove_character_flag = agot_nw_ranger_events_0100_cooldown
				}
				if = {
					limit = { has_character_flag = post_nw_reclaim_quest }
					remove_character_flag = post_nw_reclaim_quest
				}
			}
			if = {
				limit = { title:c_castle_black = { has_variable = post_nw_quest_blocker } }
				title:c_castle_black = { remove_variable = post_nw_quest_blocker }
			}
			if = {
				limit = {
					exists = primary_title
					primary_title = { has_variable = post_nw_quest }
				}
				destroy_title = primary_title
			}
		}
	}

	on_owner_death = {
		scope:story = { end_story = yes }
	}

	### DAILY
	effect_group = {
		days = 1

		#Send the event
		triggered_effect = {
			trigger = {
				story_owner = {
					NOT = { has_character_flag = agot_nw_ranger_events_0100_first_event_timer }
					NOT = { has_character_flag = agot_nw_ranger_events_0100_cooldown }
				}
			}
			effect = {
				story_owner = {
					save_scope_as = first_ranger
					trigger_event = agot_nw_ranger_events.0100
				}
			}
		}

		#Refill provisions until arrived
		triggered_effect = {
			trigger = {
				story_owner = {
					NOT = { has_character_flag = last_ranger_provisions }
					NOT = { has_character_flag = already_arrived_to_the_northern_lord }
				}
			}
			effect = {
				story_owner = {
					add_character_flag = last_ranger_provisions
					domicile = {
						if = {
							limit = { provisions < max_provisions }
							change_provisions = max_provisions
						}
					}
				}
			}
		}

		#Check if still valid
		triggered_effect = {
			trigger = {
				story_owner = {
					OR = {
						trigger_if = {
							limit = {
								exists = scope:story.var:ex_ranger_title
								exists = primary_title
							}
							NOT = { primary_title = scope:story.var:ex_ranger_title }
						}
						NOT = { is_playable_character = yes }
					}
				}
			}
			effect = {
				scope:story = { end_story = yes }
			}
		}
	}
}
