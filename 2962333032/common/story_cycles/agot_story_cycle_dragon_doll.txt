story_dragon_doll = {

	on_setup = {
		if = {
			limit = {
				exists = story_owner.var:child
			}
			story_owner = {
				var:child = { save_scope_as = child }
			}
			set_variable = {
				name = child
				value = scope:child
			}
			story_owner = { remove_variable = child }
		}

		set_variable = {
			name = events_fired
			value = 0
		}

		if = {
			limit = {
				any_living_dragon = {
					count > 0
				}
			}
			story_owner = {
				trigger_event = agot_childhood_events.0029
			}
		}
		else = {
			story_owner = {
				trigger_event = agot_childhood_events.0022
			}
		}
	}

	on_owner_death = {

	}

	on_end = {
		story_owner = {
			if = {
				limit = {
					has_character_modifier = agot_craft_focused_major
				}
				remove_character_modifier = agot_craft_focused_major
			}
			else_if = {
				limit = {
					has_character_modifier = agot_craft_focused_minor
				}
				remove_character_modifier = agot_craft_focused_minor
			}
		}

		### doll variation:
		## Color: var:doll_color
		# flag:vibrant_fit
		# flag:vibrant_off
		# flag:vibrant_incoherent
		# flag:simple_complement
		# flag:simple_clash
		# flag:monotone
		## Wing Span: var:doll_wings
		# flag:large_fit
		# flag:large_oversize
		# flag:large_goat
		# flag:small_fit
		# flag:small_undersized
		# flag:small_whale
		# flag:proportional_lopsided
		## Eyes: var:doll_eyes
		# var:doll_eyes
		# flag:buttons
		# flag:gems_off
		# flag:gems_pop
	}

	effect_group = {
		months = 1
		triggered_effect = {
			trigger = {
				var:child = {
					any_relation = {
						type = agot_dragon
						count > 0
						save_temporary_scope_as = dragon_temp
					}
				}
			}
			effect = {
				var:child = { save_scope_as = child }
				scope:dragon_temp = { save_scope_as = dragon }
				story_owner = { # invalidation event, child has a dragon
					trigger_event = agot_childhood_events.9001
				}
				end_story = yes
			}
		}
	}
	effect_group = {
		months = 2
		triggered_effect = {
			trigger = {
				story_owner = {
					any_owned_story = {
						story_type = story_dragon_doll
						var:child ?= { is_alive = yes }
					}
				}
			}
			effect = {
				save_scope_as = story
				var:child = { save_scope_as = child }
				story_owner = {
					if = {
						limit = {
							prev = { var:events_fired = 3 }
						}
						trigger_event = {
							id = agot_childhood_events.0021
							days = { 1 25 }
						}
					}
					else = {
						prev = {
							change_variable = {
								name = events_fired
								add = 1
							}
						}
						random_list = {
							1 = {
								modifier = {
									OR = {
										scope:story = { has_variable = chosen_dragon } # colors are already chosen by inspiration
										has_character_flag = has_had_event_agot_childhood_events.0024
									}
									factor = 0
								}
								trigger_event = {
									id = agot_childhood_events.0024
									days = { 1 25 }
								}
							}
							1 = {
								modifier = {
									has_character_flag = has_had_event_agot_childhood_events.0025
									factor = 0
								}
								trigger_event = {
									id = agot_childhood_events.0025
									days = { 1 25 }
								}
							}
							1 = {
								modifier = {
									has_character_flag = has_had_event_agot_childhood_events.0026
									factor = 0
								}
								trigger_event = {
									id = agot_childhood_events.0026
									days = { 1 25 }
								}
							}
							1 = {
								modifier = {
									has_character_flag = has_had_event_agot_childhood_events.0027
									factor = 0
								}
								trigger_event = {
									id = agot_childhood_events.0027
									days = { 1 25 }
								}
							}
						}
					}
				}
			}
		}
	}
}
