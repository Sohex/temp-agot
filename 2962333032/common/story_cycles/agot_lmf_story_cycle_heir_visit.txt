lmf_heir_visit_story_cycle = {

	on_setup = {
		set_variable = { name = visit_success value = 0 }
		set_variable = { name = visit_diplomacy value = 0 }
		set_variable = { name = visit_martial value = 0 }
		set_variable = { name = visit_stewardship value = 0 }
		set_variable = { name = visit_intrigue value = 0 }
		set_variable = { name = visit_learning value = 0 }
		set_variable = { name = visit_prowess value = 0 }
		set_variable = { name = visit_language value = 0 }
		set_variable = { name = visit_hunter value = 0 }
		set_variable = { name = visit_reveler value = 0 }
		set_variable = { name = visit_time value = 0 }
		set_variable = { name = visit_time_castle value = 0 }
		set_variable = { name = visit_time_target value = 0 }
		set_variable = { name = visit_time_sick value = 0 }
		set_variable = { name = visit_time_hunt value = 0 }
		set_variable = { name = visit_time_tour value = 0 }
		set_variable = { name = visit_time_party value = 0 }
		set_variable = { name = visit_time_nothing value = 0 }
	}

	on_owner_death = {
		story_owner = { save_scope_as = your_heir }
		var:heir_visit_target = {
			send_interface_message = {
				type = event_generic_bad_with_text
				title = lmf_heir.1630.t.cancel
				desc = lmf_heir.1630.desc.heir.death
				left_icon = scope:your_heir
			}
		}
		end_story = yes
	}

	on_end = {
		if = {
			limit = {
				var:heir_visit_target ?= {
					is_alive = yes
					has_variable = visited_by_heir
				}
			}
			var:heir_visit_target = { remove_variable = visited_by_heir }
		}
	}
}
