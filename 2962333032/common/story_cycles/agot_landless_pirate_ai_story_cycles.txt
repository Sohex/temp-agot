agot_story_landless_pirate_ai = {

	on_setup = {
		story_owner = { save_scope_as = owner }
	}

	on_end = {}

	on_owner_death = {}

	effect_group = {
		months = { 3 12 }
		trigger = {
			story_owner ?= {
				is_ai = yes
				agot_ai_pirate_wants_to_move = yes
				is_at_war = no
				is_travelling = no
				NOT = { has_pending_interaction_of_type = imprison_interaction }
			}
		}
		triggered_effect = {
			effect = {
				story_owner = {
					agot_pirate_ai_find_next_county_effect = yes
				}
			}
		}
	}

	effect_group = {
		months = 12
		trigger = {
			story_owner ?= {
				is_ai = no
			}
		}
		triggered_effect = {
			effect = {
				scope:story = {
					end_story = yes
				}
			}
		}
	}

	# Landless Pirate Subsidies since landless cannot raid due to the game engine
	effect_group = {
		months = { 12 14 }
		trigger = {
			story_owner ?= {
				is_alive = yes
				is_ai = yes
			}
		}
		triggered_effect = {
			effect = {
				story_owner = {
					add_gold = { 75 325 }
				}
			}
		}
	}
}
