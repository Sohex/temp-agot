@pirate_port_law_cooldown_years = 5

pirate_haven_port_laws = {
	default = pirate_tax_rate_01
	cumulative = no

	flag = agot_realm_law
	pirate_tax_rate_01 = {
		should_start_with = {
			is_landed = yes
			government_has_flag = government_is_pirate
			capital_county = {
				title_province = {
					has_building_or_higher = pirate_den_01
				}
			}
		}
		can_have = {
			is_landed = yes
			government_has_flag = government_is_pirate
			capital_county = {
				title_province = {
					has_building_or_higher = pirate_den_01
				}
			}
		}
		can_pass = {
			custom_description = {
				subject = root
				text = "has_pirate_port_tax_rate_law_cooldown"
				NOT = { has_variable = pirate_port_law_cooldown }
			}
		}
		pass_cost = {
			prestige = increase_pirate_port_tax_prestige_cost
		}
		modifier = {
			raid_speed = 0.1
			men_at_arms_recruitment_cost = -0.1
		}
		on_pass = {
			set_variable = {
				name = pirate_port_law_cooldown
				years = @pirate_port_law_cooldown_years
			}
			hidden_effect = {
				every_held_title = {
					limit = {
						tier = tier_county
					}
					every_county_province = {
						limit = {
							OR = {
								has_holding_type = pirate_den_holding
								has_holding_type = settlement_holding
							}
						}
						every_province_domicile = {
							limit = {
								is_domicile_type = agot_pirate_ship
							}
							owner = {
								send_interface_message = {
									type = msg_pirate_ship_changed_port_rates_taxes
									title = agot_pirate_port_rate_changed.t
									desc = agot_pirate_port_rate_changed.desc_01
								}
							}
						}
					}
				}
				every_held_title = {
					limit = {
						tier = tier_county
						title_province = { has_holding_type = pirate_den_holding }
						agot_is_pirate_domicile_title_trigger = no
						NOT = {
							has_county_modifier = pirate_tax_rate_01_displayer_modifier
						}
					}
					add_county_modifier = pirate_tax_rate_01_displayer_modifier
					if = {
						limit = {
							has_county_modifier = pirate_tax_rate_02_displayer_modifier
						}
						remove_county_modifier = pirate_tax_rate_02_displayer_modifier
					}
					else_if = {
						limit = {
							has_county_modifier = pirate_tax_rate_03_displayer_modifier
						}
						remove_county_modifier = pirate_tax_rate_03_displayer_modifier
					}
					else_if = {
						limit = {
							has_county_modifier = pirate_tax_rate_04_displayer_modifier
						}
						remove_county_modifier = pirate_tax_rate_04_displayer_modifier
					}
				}
			}
		}
		ai_will_do = {
			value = 0
			if = {
				limit = {
					capital_county = {
						title_province = {
							any_province_domicile = {
								is_domicile_type = agot_pirate_ship
								count >= 1
							}
							any_province_domicile = {
								is_domicile_type = agot_pirate_ship
								count <= 3
							}
						}
					}
				}
				add = 1
			}
			if = {
				limit = {
					ai_greed >= high_positive_ai_value
				}
				multiply = 0
			}
			if = {
				limit = {
					ai_greed <= very_low_negative_ai_value
				}
				add = 1
			}
		}
	}
