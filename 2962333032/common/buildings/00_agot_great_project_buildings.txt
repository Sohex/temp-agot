agot_great_fleet_01 = {
	type_icon = "icon_fleetport.dds"

	effect_desc = agot_sailing_activity_cost_discount_max_desc

	can_construct_potential = {
		always = no # This building is the result of a great project
	}

	show_disabled = yes

	rebuild_cost = {
		gold = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = {
				value = {
					add = standard_rebuild_value
					multiply = 2
				}
			}
		}
		piety = {
			if = {
				limit = { is_ai = yes }
				value = ai_rebuild_value
			}
			else = {
				value = {
					add = standard_rebuild_value
					multiply = 2
				}
			}
		}
	}

	province_modifier = {
		tax_mult = 0.2
		development_growth_factor = 0.3
		supply_limit_mult = 0.2
		monthly_income = 1
		stationed_maa_pursuit_add = 10
	}

	character_modifier = {
		monthly_dynasty_prestige_mult = 0.1
		monthly_prestige = 0.5
		naval_movement_speed_mult = 0.25
		embarkation_cost_mult = -0.25
		supply_duration = 2
	}

	character_government_modifier = {
		parameter = government_is_pirate_trigger_check
		hostile_raid_time = 0.5
		raid_speed = 0.5
	}

	character_culture_modifier = {
		parameter = ironborn_great_fleet_bonuses
		hostile_raid_time = 0.5
		raid_speed = 0.5
	}

	next_building = agot_great_fleet_02
	great_project_type = construct_great_fleet_01

	type = great_building

	flag = travel_point_of_interest_economic
	flag = agot_great_fleet_01
}
