# b_chroyane
agot_ruins_chroyane = {
	icon = { reference = @icon_great_project_agot_ruins_chroyane }
	illustration = { reference = @agot_ruins_chroyane_illustration }
	name = { desc = great_project_type_agot_ruins_chroyane }

	is_important = yes

	is_shown = {
		game_start_date >= 7297.9.4
		NOT = { has_global_variable = chroyane_ruins_restoration_completed }
		NOT = { is_planning_great_project = agot_ruins_chroyane }
		title:b_chroyane.title_province = { has_holding_type = ruin_holding }
		title:b_chroyane.county.holder ?= root
	}

	can_start_planning = { # These are the displayed requirements of the project
		custom_tooltip = {
			text = only_single_great_project_type_allowed
			NOT = { any_great_project = { great_project_type = agot_ruins_chroyane } }
		}
		title:b_chroyane.title_province = { has_building = ruin_stone_men_02 }
	}

	is_valid = {} # These will abandon the project when triggered

	province_filter = landed_title
	province_filter_target = b_chroyane

	cost = {
		treasury_or_gold = {
			value = agot_great_project_initial_cost
			multiply = 1.2
		}
	}

	construction_time = {
		value = agot_great_project_construction_time
		multiply = agot_great_project_construction_time_multiplier_value
	}

	contributor_cooldown = standard_contributor_cooldown_value
	invite_interaction = request_great_project_contribution_interaction

	allowed_contributor_filter = {
		owner = yes
		vassals = yes
		tributaries = yes
	}

	can_cancel = {}

	on_cancel = { agot_chroyane_gp_contributor_disappointment_effect = yes }

	on_plan_build = {
		agot_chroyane_ruins_gp_owner_planning_memory_effect = { PROVINCE = title:b_chroyane }
	}

	on_start_build = {
		scope:owner = {
			if = {
				limit = { has_legitimacy = yes }
				add_legitimacy_effect = { LEGITIMACY = minor_legitimacy_gain }
			}
		}
		scope:great_project = {
			every_contribution = {
				limit = { contribution_is_funded = yes }
				contributor = { add_prestige = minor_prestige_value }
			}
		}
	}

	on_invalidated = {}

	on_complete = {
		if = {
			limit = { NOT = { has_global_variable = chroyane_ruins_restoration_completed } }
			set_global_variable = chroyane_ruins_restoration_completed
		}
		if = {
			limit = {
				title:b_chroyane.title_province = {
					has_holding_type = ruin_holding
					NOT = { has_building_or_higher = agot_ruins_chroyane_walls_03 }
				}
			}
			title:b_chroyane.title_province = {
				remove_building = ruin_stone_men_02
				set_holding_type = castle_holding
				remove_building = agot_ruins_chroyane_walls_02
				add_building = agot_ruins_chroyane_walls_04
			}
			# Clean up the county
			agot_chroyane_gp_ruins_conversion_effect = { PROVINCE = title:b_chroyane }
		}
		# Ah, the memories
		agot_chroyane_ruins_gp_owner_completed_memory_effect = { PROVINCE = title:b_chroyane }
		# Tally the rewards
		scope:great_project ?= {
			agot_chroyane_gp_determine_if_contributed_and_save_scope_effect = { CONTRIBUTION = materials }
			agot_chroyane_gp_determine_if_contributed_and_save_scope_effect = { CONTRIBUTION = work_force }
			agot_chroyane_gp_determine_if_contributed_and_save_scope_effect = { CONTRIBUTION = b_chroyane_holding }
			agot_chroyane_gp_determine_if_contributed_and_save_scope_effect = { CONTRIBUTION = b_chroyane_building }
		}
		# Dole out some rewards!
		agot_chroyane_gp_gather_contributors_in_list_effect = yes
		every_in_list = {
			list = agot_contributors_to_reward
			if = {
				limit = { this = scope:owner }
				trigger_event = agot_great_project_events.1000
			}
			else = {
				if = {
					limit = { scope:owner = { is_ai = yes } }
					show_as_tooltip = { agot_chroyane_gp_contributor_reward_effect = { OVERLORD = scope:owner } }
				}
				trigger_event = agot_great_project_events.1010
			}
		}
	}

	project_contributions = {
		materials = {
			is_required = yes

			contributor_is_valid = {
				OR = {
					top_liege = scope:owner
					is_tributary_of = scope:owner
				}
			}

			cost = { treasury_or_gold = agot_great_project_material_contribution_cost }

			contributor_cooldown = 0

			on_contribution_funded = { great_project_notify_contribution = yes }
			on_complete = { agot_chroyane_gp_materials_contributor_reward_effect = yes }

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = agot_great_projects_contribution_ai_will_do_base_value }
		}

		work_force = {
			is_required = yes

			contributor_is_valid = {
				OR = {
					top_liege = scope:owner
					is_tributary_of = scope:owner
				}
			}
			context_allows_contributions = {
				is_at_war = no
			}

			cost = { treasury_or_gold = agot_great_project_work_force_contribution_cost }

			contributor_cooldown = 0

			on_contribution_funded = { great_project_notify_contribution = yes }
			on_complete = { agot_chroyane_gp_work_force_contributor_reward_effect = yes }

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = agot_great_projects_contribution_ai_will_do_base_value }
		}

		b_chroyane_holding = {
			is_shown = { title:b_chroyane.title_province = { has_holding_type = ruin_holding } }
			is_required = yes

			cost = { treasury_or_gold = agot_great_project_contribution_cost }

			contributor_cooldown = individual_contribution_cooldown_value

			on_contribution_funded = { great_project_notify_contribution = yes }
			on_complete = { agot_chroyane_gp_advanced_contributor_reward_effect = yes }

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = agot_great_projects_contribution_ai_will_do_base_value }
		}

		b_chroyane_building = {
			is_shown = { title:b_chroyane.title_province = { NOT = { has_building_or_higher = agot_ruins_chroyane_walls_03 } } }
			is_required = yes

			cost = { treasury_or_gold = agot_great_project_contribution_cost }

			contributor_cooldown = individual_contribution_cooldown_value

			on_contribution_funded = { great_project_notify_contribution = yes }
			on_complete = { agot_chroyane_gp_advanced_contributor_reward_effect = yes }

			ai_check_interval_by_tier = {
				barony = 0
				county = 12
				duchy = 9
				kingdom = 6
				empire = 6
				hegemony = 6
			}
			ai_will_do = { value = agot_great_projects_contribution_ai_will_do_base_value }
		}
	}

	ai_target_quick_trigger = {
		adult = yes
		rank = empire
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 0
		empire = 6
		hegemony = 6
	}

	ai_will_do = {
		value = 0
		if = {
			limit = {
				has_title = title:b_chroyane.county
			}
			add = 500
		}
		if = {
			limit = {
				is_at_war = yes
			}
			multiply = 0
		}
	}

	completion_sound_effect = @msg_completion_effect_chroyane
}
