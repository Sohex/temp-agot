repair_living_dragon_personalities_decision = {
    picture = {
        reference = "gfx/interface/illustrations/decisions/decision_dragons.dds"
    }
    decision_group_type = minor

    desc = repair_living_dragon_personalities_decision_desc
    selection_tooltip = repair_living_dragon_personalities_decision_tooltip
    confirm_text = repair_living_dragon_personalities_decision_confirm

    is_shown = {
        is_ai = no
        any_living_dragon = {
            trigger_if = {
                limit = { has_game_rule = agot_canon_dragons_enabled_traits }
                NOT = { has_character_flag = agot_developing_canon_dragon_traits }
            }
            number_of_personality_traits < personality_trait_limit
            OR = {
                AND = { age > 2 age < 6 number_of_personality_traits < 1 }
                AND = { age > 5 age < 10 number_of_personality_traits < 2 }
                AND = { age > 9 number_of_personality_traits < 3 }
            }
        }
    }

    is_valid = {
        any_living_dragon = {
            trigger_if = {
                limit = { has_game_rule = agot_canon_dragons_enabled_traits }
                NOT = { has_character_flag = agot_developing_canon_dragon_traits }
            }
            number_of_personality_traits < personality_trait_limit
            OR = {
                AND = { age > 2 age < 6 number_of_personality_traits < 1 }
                AND = { age > 5 age < 10 number_of_personality_traits < 2 }
                AND = { age > 9 number_of_personality_traits < 3 }
            }
        }
    }

    effect = {
        every_living_dragon = {
            limit = {
                age > 2
                trigger_if = {
                    limit = { has_game_rule = agot_canon_dragons_enabled_traits }
                    NOT = { has_character_flag = agot_developing_canon_dragon_traits }
                }
                number_of_personality_traits < personality_trait_limit
            }

            if = {
                limit = { age < 6 }
                while = {
                    limit = {
                        number_of_personality_traits < 1
                        number_of_personality_traits < personality_trait_limit
                    }
                    agot_give_random_dragon_personality_trait = yes
                }
            }
            else_if = {
                limit = { age < 10 }
                while = {
                    limit = {
                        number_of_personality_traits < 2
                        number_of_personality_traits < personality_trait_limit
                    }
                    agot_give_random_dragon_personality_trait = yes
                }
            }
            else = {
                while = {
                    limit = {
                        number_of_personality_traits < 3
                        number_of_personality_traits < personality_trait_limit
                    }
                    agot_give_random_dragon_personality_trait = yes
                }
            }
        }
        custom_tooltip = repair_living_dragon_personalities_decision_effect_tooltip
    }

    ai_potential = { always = no }
    ai_will_do = { base = 0 }
}
