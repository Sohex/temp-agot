dragonlord_council_grant_war_authorization_effect = {
    dragonlord_council_clear_war_authorization_effect = yes
    if = {
        limit = {
            has_variable = dragonlord_council_selected_war_type
            has_variable = dragonlord_council_selected_target
            OR = {
                var:dragonlord_council_selected_war_type = flag:dragonlord_colonization_war_agenda
                var:dragonlord_council_selected_war_type = flag:dragonlord_conquest_war_agenda
                var:dragonlord_council_selected_war_type = flag:dragonlord_trade_enclave_war_agenda
                var:dragonlord_council_selected_war_type = flag:dragonlord_tributary_war_agenda
                var:dragonlord_council_selected_war_type = flag:dragonlord_slave_raid_war_agenda
            }
            var:dragonlord_council_selected_target = {
                is_alive = yes
                is_landed = yes
                top_liege = this
                this != root
            }
        }
        add_character_flag = { flag = dragonlord_council_war_authorized years = 1 }
        set_variable = {
            name = dragonlord_council_war_authorization_type
            value = var:dragonlord_council_selected_war_type
        }
        set_variable = {
            name = dragonlord_council_war_authorization_target
            value = var:dragonlord_council_selected_target
        }
        send_interface_message = {
            type = event_generic_good
            title = dragonlord_council_war_authorization_granted_message
            left_icon = root
        }
    }
    else = {
        send_interface_message = {
            type = event_generic_bad
            title = dragonlord_council_war_authorization_invalid_message
            left_icon = root
        }
    }
}
