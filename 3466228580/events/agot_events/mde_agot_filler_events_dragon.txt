mde_filler_dragon.0001 = { #children fight
	hidden = yes
	cooldown = {
		years = 10 #doesnt happen to the same kids?
	}
	trigger = {
		has_character_flag = owned_dragon
		exists = var:current_rider 

		AND = {
			var:current_rider = {
				is_available = yes
				is_landed = yes
				location = capital_province
				any_court_position_holder = {
					type = wet_nurse_court_position
					has_court_event_flag = no
					is_available_ai_adult = yes
				}
				any_character_artifact = {
					artifact_type = dragon_egg 
				}
				any_child = { #missing traits trigger
					NOT = { has_trait = dragonrider }
					is_available = yes
					count > 2
					age >= 4
					age <= 15
					NOT = { has_character_flag = mde_mummers_egg }
				}
			}
			location = var:current_rider.capital_province
		}
		location = var:current_rider.capital_province
	}
	immediate = {
		var:current_rider = {
			trigger_event = {
				id = mde_filler_dragon.0002
			}
		}
	}
}
