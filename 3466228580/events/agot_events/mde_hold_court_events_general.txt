namespace = mde_hold_court_dragon


mde_hold_court_dragon.0001 = { #burnt bones prove nothing
	type = court_event
	title = mde_hold_court_dragon.0001.t
	desc = mde_hold_court_dragon.0001.desc #make it happen often until dragon is in cage, but different text
	theme = court
	cooldown = { years = 2 }

	court_scene = {
		button_position_character = scope:mde_sad_dad
		court_event_force_open = yes
		show_timeout_info = no
		should_pause_time = yes
		roles = {
			scope:mde_sad_dad = {
				group = petitioners_group
				animation = beg
			}
		}
	}

	widget = {
		gui = "event_window_widget_event_chain_progress"
		container = "custom_widgets_container"
		controller = event_chain_progress
	}

	weight_multiplier = {
		base = 1
	}

	trigger = {
		NOT = { has_character_flag = had_mde_hold_court_dragon_0001 }
		has_trait = dragonrider
	}

	immediate = {
		if = {
			random_held_county = {
				limit = {
					OR = {
						any_county_province = {
							has_holding_type = castle_holding
							has_building_or_higher = generic_dragon_pit_01
						}
						any_county_province = {
							OR = {
								has_building_or_higher = dragonpit_01 # Historical Version
								has_building = dragonpit_ruins_03 # Fully Restored Ruin Version
								has_building = agot_dragonmont_01 # Dragonmont Version
							}
						}
					}
					has_variable = has_dragonkeeper_order
				}
				save_scope_as = dragon_pit_location
			}
		}

		add_character_flag = {
			flag = had_mde_hold_court_dragon_0001
			years = 2
		}
		create_character = {
 			location = root.capital_province
			age = { 25 60 }
			culture = root.location.culture
			faith = root.location.faith
			dynasty = none
			gender_female_chance = 10

			random_traits = yes
			random_traits_list = {
				count = 1
				education_stewardship_1 = {}
				education_stewardship_2 = {}
				education_stewardship_3 = {}
				education_stewardship_4 = {}
			}
 			save_scope_as = mde_sad_dad
		}
		random_knight = {
			limit = {
				NOT = { is_player_heir_of = root }
				is_available_adult = yes
				is_landed = no
				is_married = no
				is_incapable = no
				is_courtier_of = root
			}

			save_scope_as = mde_bad_knight
		}
		var:current_dragon = {
			save_scope_as = active_dragon
		}
	}

	option = {
		name = mde_hold_court_dragon.0001.a #“Silence! (guard name) you are insufferable, compensate the man.”

		remove_short_term_gold = 150

		add_piety = 50

		reverse_add_opinion = {
			target = scope:mde_bad_knight
			modifier = angry_opinion
			opinion = -25
		}

		ai_chance = {
			base = 1
		}

	}

	option = {
		name = mde_hold_court_dragon.0001.b

		add_piety = -50

		rightfully_imprison_character_effect = {
			IMPRISONER = root
			TARGET = scope:mde_sad_dad
		}

		ai_chance = {
			base = 1
		}

	}

	option = {
		name = mde_hold_court_dragon.0001.c

		scope:active_dragon = {
			agot_send_to_dragonpit_effect = { DRAGON = scope:active_dragon DRAGONPIT_HOLDER = root DRAGONPIT_COUNTY = scope:dragon_pit_location }
		}

		ai_chance = {
			base = 1
		}

	}

	option = {
		name = mde_hold_court_dragon.0001.d

		reverse_add_opinion = {
			target = scope:mde_bad_knight
			modifier = angry_opinion
			opinion = -10
		}

	}

	after = {
		hidden_effect = {
			scope:mde_sad_dad = {
				death = {
					death_reason = death_vanished
				}
			}
	}
	}
}
