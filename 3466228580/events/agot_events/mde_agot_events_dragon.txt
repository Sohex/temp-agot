mde_dragon.0001 = { # I lose interest
	type = character_event
	window = agot_dragon_character_event
	title = mde_dragon.0001.t
	desc = mde_dragon.0001.desc
	theme = dragon

	left_portrait = {
		character = root
		animation = personality_irrational
	}

	right_portrait = {
		character = var:current_dragon
		animation = dragon_main
		camera = camera_dragon_event_standing

		#Add this so dragon fits within event window
		outfit_tags = { linear_camera_zoom }
	}


	immediate = {
		play_music_cue = CK3AGOT_dragonflight_wonder
		var:current_dragon = {
			save_scope_as = ridden_dragon
		}
		add_character_flag = currently_riding_dragon
		agot_set_dragon_animation_flag_effect = { FLAG = dragon_flying RIDING_DRAGON = YES }

		stress_impact = {
			depressed_genetic = medium_stress_impact_gain
			depressed_1 = medium_stress_impact_gain
			fickle = minor_stress_impact_gain
			shy = medium_stress_impact_gain
			lazy = medium_stress_impact_gain
			widowed = medium_stress_impact_gain
			eccentric = minor_stress_impact_gain
			craven = medium_stress_impact_gain
			paranoid = minor_stress_impact_gain
		}
	
	}

	option = {
		name = mde_dragon.0001.a #no I must ride
		custom_tooltip = mde_dragon.0001.a.tt
		add_trait_xp = {
			trait = dragonrider
			track = dragon_bond
			value = -8
		}
		add_stress = minor_stress_gain
		trigger_event = { id = agot_dragon.0010 days = 0 } 
	}

	option = {
		name = mde_dragon.0001.b #maybe just some alone time with <name>
		flavor = mde_dragon.0001.b.f #you will force yourself to ride

		trigger = {
			OR = {
				has_trait = loyal
				has_trait = compassionate
				has_trait = calm
				has_trait = diligent
				has_trait = trusting
			}
		}
		add_stress = minor_stress_loss
		add_trait_xp = {
			trait = dragonrider
			track = dragon_bond
			value = 12
		}
		remove_character_flag = currently_riding_dragon
		if = {
			limit = { has_character_flag = dragon_night_ride }
			remove_character_flag = dragon_night_ride
		}
		agot_clear_dragon_animation_flags = yes
		ai_chance = {
			base = 10
			modifier = {
				OR = {
					has_trait = loyal
					has_trait = compassionate
					has_trait = calm
					has_trait = diligent
					has_trait = trusting
				}
				add = 50
			}
			ai_value_modifier = {
				ai_sociability = 2
				ai_compassion = 2
			}
		}
	}

	option = {
		name = mde_dragon.0001.c #I shouldnt be here
		flavor = mde_dragon.0001.c.f #you will force yourself to ride

		stress_impact = {
			shy = medium_stress_impact_loss
			craven = medium_stress_impact_loss
			paranoid = major_stress_impact_loss
			brave = major_stress_impact_gain
			ambitious = medium_stress_impact_gain
		}
		remove_character_flag = currently_riding_dragon
		if = {
			limit = { has_character_flag = dragon_night_ride }
			remove_character_flag = dragon_night_ride
		}
		agot_clear_dragon_animation_flags = yes
		ai_chance = {
			base = 1
			modifier = {
				OR = {
					has_trait = shy
					has_trait = craven
					has_trait = paranoid
				}
				add = 100
			}
			ai_value_modifier = {
				ai_energy = 2
				ai_vengefulness = 2
			}
		}
	}
	after = {
		add_character_flag = {
			flag = mde_why_am_i_here
			days = 5460
		}
	}
}
