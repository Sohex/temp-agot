### Casting the rite: 0093 (asking) -> 0094 (the rite itself) ###
# Started by the Conceive a Shadow interaction, which saves its target
# into var:agot_shadow_target before triggering 0093 (a plain trigger_event
# cannot otherwise carry scope:recipient across the narrative chain).

agot_shadow.0093 = {
	type = character_event
	title = agot_shadow.0093.t
	desc = agot_shadow.0093.desc

	left_portrait = {
		character = root
		animation = personality
	}

	right_portrait = {
		character = root.primary_spouse
		animation = worry
	}

	theme = witchcraft

	option = {
		name = agot_shadow.0093.a # Ask this of them
		trigger_event = { id = agot_shadow.0094 days = 3 }
	}

	option = {
		name = agot_shadow.0093.b # You cannot ask this of them
		stress_impact = { base = medium_stress_impact_gain }
		remove_variable = agot_shadow_target
	}
}

agot_shadow.0094 = {
	type = character_event
	title = agot_shadow.0094.t
	desc = agot_shadow.0094.desc

	left_portrait = {
		character = root
		animation = personality
	}

	theme = witchcraft

	option = {
		name = agot_shadow.0094.a # Give what must be given
		trigger = {
			exists = var:agot_shadow_target
		}
		primary_spouse = {
			add_character_modifier = {
				modifier = agot_shadow_lifefire_spent_modifier
			}
		}
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = agot_shadow_conceive_a_shadow_notification
			left_icon = root
			right_icon = var:agot_shadow_target
		}
		var:agot_shadow_target = {
			trigger_event = { id = agot_shadow.0095 days = 12 }
		}
		remove_variable = agot_shadow_target
	}
}

# The shadow finds its mark - fired from the rite above, scoped so that
# ROOT is already the intended victim. Hidden: this is the assassin
# arriving, not something the victim chooses. Mastery (0-3, from Steel
# Your Resolve) improves the strike's chance by up to ~20 points.
agot_shadow.0095 = {
	type = character_event
	hidden = yes

	immediate = {
		random_list = {
			15 = {
				modifier = { factor = 0.6 var:agot_shadow_mastery >= 1 }
				modifier = { factor = 0.6 var:agot_shadow_mastery >= 2 }
				modifier = { factor = 0.4 var:agot_shadow_mastery >= 3 }
			}
			85 = {
				death = { death_reason = death_murder }
			}
		}
	}
}
