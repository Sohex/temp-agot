namespace = agot_ambient

# Hidden wrapper for the yearly per-character pulse (see
# 00_agot_ambient_on_actions.txt). Rolls privately so nothing shows for
# the ~90%+ of playable characters untouched by magic, and even touched
# characters only see this rarely (roughly once per 8 years on average).
agot_ambient.0001 = {
	type = character_event
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = agot_touched_by_magic
			}
			random = {
				chance = 12
				trigger_event = agot_ambient.0002
			}
		}
	}
}

# The actual ambient moment - unbidden, no decision/interaction needed.
# Text varies per tradition; the two options are a light, genuinely
# optional flavor choice (lean in vs. push it away), not a system to
# min-max.
agot_ambient.0002 = {
	type = character_event
	title = agot_ambient.0002.t
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_rhllor }
				desc = agot_ambient.0002.desc.rhllor
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_warging }
				desc = agot_ambient.0002.desc.warging
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_blood }
				desc = agot_ambient.0002.desc.blood
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_drowned }
				desc = agot_ambient.0002.desc.drowned
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_greenseer }
				desc = agot_ambient.0002.desc.greenseer
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_shadow }
				desc = agot_ambient.0002.desc.shadow
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_faceless }
