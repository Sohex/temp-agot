# The "Spellbook" menu - opened via the agot_magic_open_spellbook_decision.
# Self-targeted spells are cast directly from here. Targeted spells
# (Warg Scout, Warg Sway, Blood Curse) are character interactions used the
# normal CK3 way, from the target's own character screen - an event popup
# has no character-picker widget, so they can't live here too. This event
# just explains that.
agot_magic.0001 = {
	type = character_event
	title = agot_magic.0001.t
	desc = {
		desc = agot_magic.0001.desc.intro
		first_valid = {
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_rhllor }
				desc = agot_magic.0001.desc.rhllor
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_warging }
				desc = agot_magic.0001.desc.warging
			}
			triggered_desc = {
				trigger = { has_variable = agot_gift_of_blood }
				desc = agot_magic.0001.desc.blood
			}
		}
	}

	left_portrait = {
		character = root
		animation = personality
	}

	theme = witchcraft

	option = {
		name = agot_magic.0001.a
		trigger = {
			has_perk = agot_rhllor_ember_sight_perk
		}
		if = {
			limit = { learning >= 10 }
			random_list = {
				70 = {
					custom_tooltip = {
						text = agot_magic.0001.a.vision_good
					}
					add_character_modifier = {
						modifier = agot_magic_vision_foresight_modifier
						years = 1
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = agot_magic_vision_success_toast
						left_icon = root
					}
				}
				30 = {
					custom_tooltip = {
						text = agot_magic.0001.a.vision_bad
					}
					stress_impact = { base = medium_stress_impact_gain }
					send_interface_toast = {
						type = event_toast_effect_bad
						title = agot_magic_vision_failure_toast
						left_icon = root
					}
				}
			}
		}
		else = {
			random_list = {
				45 = {
					custom_tooltip = {
						text = agot_magic.0001.a.vision_good
					}
					add_character_modifier = {
						modifier = agot_magic_vision_foresight_modifier
						years = 1
					}
					send_interface_toast = {
						type = event_toast_effect_good
						title = agot_magic_vision_success_toast
						left_icon = root
					}
				}
				55 = {
					custom_tooltip = {
						text = agot_magic.0001.a.vision_bad
					}
					stress_impact = { base = medium_stress_impact_gain }
					send_interface_toast = {
						type = event_toast_effect_bad
						title = agot_magic_vision_failure_toast
						left_icon = root
					}
				}
			}
		}
		agot_magic_track_rhllor_overuse_effect = yes
		agot_magic_risk_discovery_effect = yes
	}

	option = {
		name = agot_magic.0001.close
	}
}

# ============================================================
# Casting chains - the three targeted spells (Cast a Blood Curse,
# Send Your Beast to Watch, Slip Into Their Mind) used to be instant,
# one-click character_interactions. Each now saves its target into a
# variable and opens with a short narrative event before the existing
# outcome logic resolves - the outcome odds/effects themselves are
# unchanged, only relocated out of on_accept.
# ============================================================

### Blood Magic - Cast a Blood Curse: 0060 (the working) -> 0061 (the toll) ###

agot_magic.0060 = {
	type = character_event
	title = agot_magic.0060.t
	desc = agot_magic.0060.desc

	left_portrait = {
		character = root
		animation = personality
	}

	theme = witchcraft

	option = {
		name = agot_magic.0060.a # A lock of their hair, stolen or bought
		trigger_event = { id = agot_magic.0061 days = 3 }
	}

	option = {
		name = agot_magic.0060.b # Something closer - blood, taken in secret
		trigger_event = { id = agot_magic.0061 days = 3 }
	}

	option = {
		name = agot_magic.0060.c # Let it go
		stress_impact = { base = minor_stress_impact_gain }
		remove_variable = agot_bloodcurse_target
	}
}

agot_magic.0061 = {
	type = character_event
	title = agot_magic.0061.t
	desc = agot_magic.0061.desc

	left_portrait = {
		character = root
		animation = personality
	}

	theme = witchcraft

	option = {
		name = agot_magic.0061.a # Open your own blood
		trigger = {
			exists = var:agot_bloodcurse_target
		}
		add_character_modifier = {
			modifier = agot_magic_blood_price_modifier
			years = 2
		}
		random_list = {
			35 = {
				custom_tooltip = {
					text = agot_bloodmagic_curse_outcome_infertility
				}
				var:agot_bloodcurse_target = {
					add_character_modifier = {
						modifier = agot_magic_blood_curse_infertility_modifier
					}
				}
			}
			35 = {
				custom_tooltip = {
					text = agot_bloodmagic_curse_outcome_wasting
				}
				var:agot_bloodcurse_target = {
					add_character_modifier = {
						modifier = agot_magic_blood_curse_wasting_modifier
					}
				}
			}
			30 = {
				custom_tooltip = {
					text = agot_bloodmagic_curse_outcome_backfire
				}
				add_character_modifier = {
					modifier = agot_magic_blood_curse_backfire_modifier
				}
			}
		}
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = agot_bloodmagic_curse_interaction_notification
			left_icon = root
		}
		agot_magic_track_blood_overuse_effect = yes
		agot_magic_risk_discovery_effect = yes
		remove_variable = agot_bloodcurse_target
	}
}
