# flavor - all three options lead to the same roll, the same way the
# original decision's chance never depended on the choice either.

	desc = agot_glass_candle.0001.desc

		character = root
		animation = personality
	}

	theme = witchcraft

		trigger_event = { id = agot_glass_candle.0002 days = 1 }
	}

	option = {
		name = agot_glass_candle.0001.b # A looted ruin's curiosities
		trigger_event = { id = agot_glass_candle.0002 days = 1 }
	}

	option = {
		name = agot_glass_candle.0001.c # A private collector, for the right price
		trigger_event = { id = agot_glass_candle.0002 days = 1 }
	}
}

agot_glass_candle.0002 = {
	type = character_event
	title = agot_glass_candle.0002.t
	desc = agot_glass_candle.0002.desc

	left_portrait = {
		character = root
		animation = personality
	}

	theme = witchcraft

	option = {
		name = agot_glass_candle.0002.a # See what turned up
		random_list = {
			8 = {
				custom_tooltip = {
					text = agot_glass_candle.0002.a.success
				}
				agot_glass_candle_create_effect = yes
			}
			92 = {
				custom_tooltip = {
					text = agot_glass_candle.0002.a.failure
				}
				send_interface_toast = {
					type = event_toast_effect_bad
					title = agot_glass_candle_search_failed_toast
					left_icon = root
				}
			}
		}
	}
}
