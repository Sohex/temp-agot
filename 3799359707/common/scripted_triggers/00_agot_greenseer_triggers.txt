# AGOT: Magic of Westeros v3 - Greenseer (Grünes Netz) branch
# Same eligible population as Warging (Old Gods faith / First Men and
# Wildling cultures) - lore-accurate (all greenseers could also warg), but
# kept as a fully separate, parallel gift for game-mechanical simplicity,
# same as R'hllor/Warging/Blood Magic already are parallel/exclusive.

agot_magic_eligible_for_greenseer_trigger = {
	custom_tooltip = {
		text = agot_magic_not_yet_awakened_tooltip
		NOR = {
			has_variable = agot_touched_by_magic
			has_variable = agot_gift_of_greenseer
		}
	}
	is_adult = yes
	OR = {
		faith = { has_doctrine = tenet_greensight }
		culture = culture:northman
		culture = culture:crannogman
		culture = culture:wolfswood_clansman
		culture = culture:skagosi
		culture = culture:barrowman
		culture = culture:bearman
		culture = culture:thenn
		culture = culture:fangman
		culture = culture:forestman
		culture = culture:hornfoot
		culture = culture:nightrunner
		culture = culture:lakefolk
		culture = culture:frozen_shoreman
		culture = culture:milkwaterman
		culture = culture:cave_dweller
		culture = culture:ice_riverman
		culture = culture:moon_clan
	}
}

# Perk 3 (Third Eye) must be held to attempt the Hammer of the Waters -

agot_greenseer_can_call_hammer_of_the_waters_trigger = {
	has_perk = agot_greenseer_third_eye_perk
