# AGOT: Magic of Westeros - Shared triggers

## Does this character qualify to try awakening R'hllor's gift?
## Real R'hllor faith key from AGOT: rhllor_asshai
agot_magic_eligible_for_rhllor_trigger = {
	custom_tooltip = {
		text = agot_magic_not_yet_awakened_tooltip
		NOR = {
			has_variable = agot_touched_by_magic
			has_variable = agot_gift_of_rhllor
		}
	}
	is_adult = yes
	OR = {
		faith = { has_doctrine = tenet_heart_of_fire }
		faith = { religion = religion:the_rhllor_religion }
	}
}

## Does this character qualify to try awakening the Warg's gift?

## Blood magic is a desperate, universal art - anyone can seek it out, no
## faith or culture gate. It is simply rarer to find someone willing to
## teach it, and far more dangerous to be seen doing it.
agot_magic_eligible_for_bloodmagic_trigger = {
	custom_tooltip = {
		text = agot_magic_not_yet_awakened_tooltip
		NOR = {
			has_variable = agot_touched_by_magic
			has_variable = agot_gift_of_blood
		}
	}
	is_adult = yes
}

## Whether this character has been Awakened at all (any tradition).
## Wrapped in custom_tooltip for the same reason as the eligibility
## triggers above - CK3's auto-tooltip generator can't describe
## has_variable on its own ("BUG: has_variable missing perspective").
agot_magic_has_any_gift_trigger = {
	custom_tooltip = {
		text = agot_magic_has_gift_tooltip
		has_variable = agot_touched_by_magic
	}
}

## Faiths descended from the Faith of the Seven treat all magic as heresy.
## Real id from AGOT: fots_seven
agot_magic_faith_persecutes_magic_trigger = {
	faith = { religion = religion:the_seven_religion }
}

## Shared AI gate for every "deadly" magic interaction (curses, kills,
## bindings with a real cost). Called from scope:actor with scope:recipient
## already set. The AI should only ever reach for lethal/harmful magic
## against someone it already has a real reason to want gone - a rival,
## a nemesis, an active war enemy, or someone it despises outright - never
## at random against courtiers/guests it merely dislikes a little.
## has_relation_rival/nemesis confirmed against vanilla
## (activities/activity_types/witch_ritual.txt uses the same check).
agot_magic_ai_has_valid_grudge_trigger = {
	OR = {
		has_relation_rival = scope:recipient
		has_relation_nemesis = scope:recipient
		is_at_war_with = scope:recipient
		opinion = {
			target = scope:recipient
			value <= -30
		}
	}
}

## Vision of What Is to Come, claim branch: would this character
## plausibly want the vision's target's crown if they suddenly had a
## real claim to it? Kept narrow on purpose - not every vassal/courtier
## is a plausible future pretender.
agot_magic_plausible_pretender_trigger = {
	OR = {
		has_trait = ambitious
		opinion = {
			target = root
			value <= 0
		}
	}
}
