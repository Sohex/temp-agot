# AGOT: Magic of Westeros v2 - Drowned God
# Extends the base game's has_natural_death_second_chance rule (originally
# only for the "Know Thyself" dynasty legacy perk) with our own branch, so
# our finisher perk taps the exact same tested vanilla event chain
# (on_natural_death_second_chance on_action -> death_management.9998/9999).
# This file must fully redefine the rule (CK3 scripted_rules are replaced
# by key, not merged) - the AND/OR branches below are copied verbatim from
# the base game's common/scripted_rules/00_rules.txt, with one new branch
# appended for agot_drowned_reborn_of_the_deep_perk.
#
# IMPORTANT: for same-key conflicts across different files in the same
# common/ subfolder, Clausewitz resolves them by FILENAME ALPHABETICAL
# ORDER, not by mod load order/priority (confirmed via Paradox forum: "The
# code loads scripts by alphabetical order, so you just need to make sure
# your file gets read after the file you're overriding"). AGOT's own file
# is common/scripted_rules/00_rules.txt - naming ours "00_agot_drowned..."
# sorted BEFORE it ("00_a" < "00_r"), so AGOT's plain version was silently
# winning regardless of mod order in the playset. Hence the "zzz_" prefix:
# guarantees this file is read last within scripted_rules, whatever else
# ever gets added there. This is the same known conflict class documented
# by the real "AGOT: Supernatural & The Long Night Compatibility Patch"
# mod, which merges multiple mods' has_natural_death_second_chance clauses
# for the same reason.

has_natural_death_second_chance = {
	OR = {
		AND = {
			has_perk = know_thyself_perk
			NOT = { has_character_flag = know_thyself_has_triggered_delay }
		}
		has_character_flag = know_thyself_perk_delay_period
		AND = {
			has_perk = agot_drowned_reborn_of_the_deep_perk
			NOT = { has_character_flag = know_thyself_has_triggered_delay }
		}
	}
}
