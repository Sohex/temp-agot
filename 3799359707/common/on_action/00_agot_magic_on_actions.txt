# AGOT: Magic of Westeros
# Hooks the rare "born touched by magic" roll onto the vanilla birth
# on_action. IMPORTANT (corrected after testing): only the on_action's
# `events` list merges across mod files - a bare `effect = {}` block does
# NOT merge, the last-loaded file's effect silently replaces every other
# mod's effect for that on_action (confirmed via error.log: "There is more
# than one 'effect' defined using most recent"). AGOT's own on_birth_child
# effect (twin handling, dragon egg cradling, etc.) would otherwise be
# silently discarded by our mod. Fix: hook in via the merging `events`
# list instead, pointing at a tiny hidden wrapper event.

on_birth_child = {
	events = {
		agot_magic.0020
	}
}
