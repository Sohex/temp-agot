# AGOT: Magic of Westeros v3 - Faceless Men
# Distinct risk model: a "Debt to the Many-Faced God" counter that, unlike
# every other tradition's overuse tracker, is resolved through a genuine
# player CHOICE (pay a real, heavy price - or refuse and risk the Guild's
# own knives turning on you). See agot_faceless_events.txt .0102.

agot_magic_grant_faceless_gift_effect = {
	set_variable = { name = agot_touched_by_magic value = yes }
	set_variable = { name = agot_gift_of_faceless value = yes }
	set_variable = { name = agot_faceless_debt value = 0 }
	send_interface_toast = {
		type = event_toast_effect_good
		title = agot_faceless_gift_granted_toast
		left_icon = root
	}
}

agot_faceless_ritual_failure_effect = {
	agot_magic_seek_gift_failure_effect = yes
}

agot_faceless_track_debt_effect = {
	if = {
		limit = { NOT = { exists = var:agot_faceless_debt } }
		set_variable = { name = agot_faceless_debt value = 0 }
	}
	change_variable = { name = agot_faceless_debt add = 1 }
	if = {
		limit = {
			var:agot_faceless_debt >= 5
			NOT = { has_character_flag = agot_faceless_debt_choice_pending }
		}
		set_character_flag = agot_faceless_debt_choice_pending
		trigger_event = agot_faceless.0102
	}
}
