# AGOT: Magic of Westeros v3 - Alchemists' Guild / Wildfire

agot_wildfire_grant_knowledge_effect = {
	set_variable = { name = agot_wildfire_knowledge value = yes }
	set_variable = { name = agot_wildfire_mastery value = 0 }
	send_interface_toast = {
		type = event_toast_effect_good
		title = agot_wildfire_knowledge_granted_toast
		left_icon = root
	}
}

# Discovery is a reputation/suspicion problem, not a witchcraft accusation
# (this is mundane knowledge) - its own small modifier, not the shared
# agot_magic_risk_discovery_effect used by the real magic traditions.
agot_wildfire_discovery_effect = {
	random = {
		chance = 12
		add_character_modifier = {
			modifier = agot_wildfire_reckless_reputation_modifier
			years = 5
		}
		if = {
			limit = { exists = liege }
			liege = {
				add_opinion = {
					target = root
					modifier = suspicion_opinion
					opinion = -15
				}
			}
		}
		send_interface_toast = {
			type = event_toast_effect_bad
			title = agot_wildfire_discovered_toast
			left_icon = root
		}
	}
}
