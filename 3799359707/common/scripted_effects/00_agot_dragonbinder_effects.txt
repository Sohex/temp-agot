# AGOT: Magic of Westeros v3 - Blut fuer Feuer, Feuer fuer Blut
# Hooks directly into AGOT's own existing dragon-taming machinery
# (common/scripted_effects/00_agot_dragon_effects.txt: agot_tame_dragon,
# agot_untame_dragon, both already used by AGOT's own debug tools) rather
# than reimplementing dragon-binding from scratch. Reuses AGOT's own
# registered "dragon_horn" artifact type - no new artifact type needed.

agot_dragonbinder_create_horn_effect = {
	create_artifact = {
		name = agot_dragonbinder_artifact_name
		description = agot_dragonbinder_artifact_desc
		visuals = dragon_horn
		type = dragon_horn
		quality = 90
		wealth = 80
		modifier = agot_dragonbinder_owned_modifier
	}
	set_variable = { name = agot_dragonbinder_mastery value = 0 }
	send_interface_toast = {
		type = event_toast_effect_good
		title = agot_dragonbinder_found_toast
		left_icon = root
	}
}
