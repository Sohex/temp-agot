# AGOT: Magic of Westeros v3 - Mastery system
# Each of the 4 riskiest v3 abilities (Dragonbinder, Shadowbinding,
# Faceless Men, Wildfire) can be practiced/studied via a repeatable
# decision. Mastery caps at 3 and shifts the odds of the matching ritual
# in its favor - without it, the base (harsher) odds apply. Practice
# itself is not risk-free: it can also go nowhere, with varied flavor
# either way so repeated use does not always read the same.

agot_dragonbinder_study_effect = {
	if = {
		limit = { NOT = { exists = var:agot_dragonbinder_mastery } }
		set_variable = { name = agot_dragonbinder_mastery value = 0 }
	}
	random_list = {
		55 = {
			if = {
				limit = { var:agot_dragonbinder_mastery < 3 }
				custom_tooltip = {
					text = agot_dragonbinder_study_success
				}
				change_variable = { name = agot_dragonbinder_mastery add = 1 }
				send_interface_toast = {
					type = event_toast_effect_good
					title = agot_dragonbinder_study_progress_toast
					left_icon = root
				}
			}
			else = {
				custom_tooltip = {
					text = agot_dragonbinder_study_already_mastered
				}
				send_interface_toast = {
					type = event_toast_effect_neutral
					title = agot_dragonbinder_study_no_progress_toast
					left_icon = root
				}
			}
		}
		45 = {
			random_list = {
				50 = { custom_tooltip = { text = agot_dragonbinder_study_setback_a } }
				50 = { custom_tooltip = { text = agot_dragonbinder_study_setback_b } }
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = agot_dragonbinder_study_no_progress_toast
				left_icon = root
			}
		}
	}
}
