agot_magic_grant_shadow_gift_effect = {
	set_variable = { name = agot_touched_by_magic value = yes }
	set_variable = { name = agot_gift_of_shadow value = yes }
	set_variable = { name = agot_shadow_mastery value = 0 }
	send_interface_toast = {
		type = event_toast_effect_good
		title = agot_shadow_gift_granted_toast
		left_icon = root
	}
}

			if = {
				limit = { NOT = { has_trait = agot_shadow_scarred } }

			}

	}
