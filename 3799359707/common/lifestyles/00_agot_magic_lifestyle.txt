# AGOT: Magic of Westeros
# A dedicated lifestyle for characters who have been Awakened to magic.
# Gated by the hidden variable `agot_touched_by_magic` (never a trait - a
# marker trait would be visible to every other player, which defeats the
# point of secret magic). See 00_agot_magic_effects.txt (grant_X_gift_effect)
# and 00_agot_magic_decisions.txt (the "Seek the Gift" decisions) for how a
# character gains access to this lifestyle in the first place.

agot_magic_lifestyle = {
	icon = agot_magic_lifestyle

	is_valid = {
		has_variable = agot_touched_by_magic
	}

	is_highlighted = {
		has_variable = agot_touched_by_magic
	}

	xp_per_level = 1000
	base_xp_gain = 25
}
