# AGOT: Magic of Westeros v3 - Faceless Men
# Unlike every other targeted ability in the mod, this one has NO range
# restriction and no is_shown relation requirement - any known character,
# anywhere. The outcome reads as a natural death (no murder/scheme flag),
# unlike every other lethal ability in the mod.

agot_faceless_issue_a_contract_interaction = {
	interface_priority = 94
	category = interaction_category_hostile
	icon = agot_faceless_issue_a_contract

	desc = agot_faceless_issue_a_contract_interaction_desc

	cooldown = { years = 2 }

	is_shown = {
		scope:actor = {
			agot_faceless_can_issue_contract_trigger = yes
		}
		scope:actor != scope:recipient
	}

	is_valid = {
		custom_tooltip = {
			text = agot_magic_not_self_target_tooltip
			scope:actor != scope:recipient
		}
	}

	on_accept = {
		scope:actor = {
			set_variable = {
				name = agot_faceless_target
				value = scope:recipient
			}
			trigger_event = agot_faceless.0104
		}
	}

	auto_accept = yes

	ai_potential = {
		agot_faceless_can_issue_contract_trigger = yes
	}

	ai_targets = {
		ai_recipients = scripted_relations
		ai_recipients = war_enemies
		ai_recipients = primary_war_enemies
		ai_recipients = courtiers
		max = 15
	}

	ai_frequency = 3

	ai_will_do = {
		base = -70
		modifier = {
			add = 100
			scope:actor = { has_relation_nemesis = scope:recipient }
		}
		modifier = {
			add = 70
			scope:actor = { has_relation_rival = scope:recipient }
		}
		modifier = {
			add = 50
			scope:actor = { is_at_war_with = scope:recipient }
		}
	}
}
