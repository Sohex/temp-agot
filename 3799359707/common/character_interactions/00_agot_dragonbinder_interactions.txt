# AGOT: Magic of Westeros v3 - Blut fuer Feuer, Feuer fuer Blut
# "No mortal man shall sound me and live" - highest single-use death risk
# in the whole mod (40%), but on success permanently binds ANY dragon,
# even one already ridden by someone else (stealing it from them via
# AGOT's own agot_untame_dragon/agot_tame_dragon effects).

agot_dragonbinder_blow_the_horn_interaction = {
	interface_priority = 99
	category = interaction_category_hostile
	icon = agot_dragonbinder_blow_the_horn

	desc = agot_dragonbinder_blow_the_horn_interaction_desc

	cooldown = { years = 20 }

	is_shown = {
		scope:actor = {
			any_character_artifact = { artifact_type = dragon_horn }
		}
		scope:recipient = {
			has_trait = dragon
		}
		scope:actor != scope:recipient
	}

	is_valid = {
		custom_tooltip = {
			text = agot_dragonbinder_recipient_must_be_dragon_tooltip
			scope:recipient = { has_trait = dragon }
		}
	}

	on_accept = {
		scope:actor = {
			stress_impact = { base = massive_stress_impact_gain }
			set_variable = {
				name = agot_dragonbinder_target_dragon
				value = scope:recipient
			}
			trigger_event = agot_dragonbinder.0105
		}
	}

	auto_accept = yes

	# The target is a dragon, not a person - "valid grudge" does not apply
	# here, this is ambition versus a real death risk (40%) instead. Kept
	# deliberately rare, matching the search decision's own "even the AI
	# is wary of this one" weighting.
	ai_targets = {
		ai_recipients = realm_characters
		ai_recipients = neighboring_rulers
		ai_recipients = war_enemies
		max = 10
	}

	ai_frequency = 6

	ai_will_do = {
		base = -20
	}
}
