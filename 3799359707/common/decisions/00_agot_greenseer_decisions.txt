### Scry - Perk 1. Senses whether magic is present nearby, no exact name. ###
agot_greenseer_scry_decision = {
	ai_check_interval = 12

	picture = {
		reference = "gfx/interface/illustrations/decisions/agot_decision_greenseer.dds"
	}

	desc = agot_greenseer_scry_decision_desc
	selection_tooltip = agot_greenseer_scry_decision_tooltip

	cooldown = { years = 1 }

	is_shown = {
		has_perk = agot_greenseer_awakened_sight_perk
	}

	is_valid = {
		has_perk = agot_greenseer_awakened_sight_perk
	}

	effect = {
		random_courtier = {
			limit = {
				has_variable = agot_touched_by_magic
			}
			save_scope_as = agot_greenseer_scry_target
		}
		if = {
			limit = {
				exists = scope:agot_greenseer_scry_target
				scope:agot_greenseer_scry_target = { is_landed = yes }
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = agot_greenseer_scry_sensed_landed_toast
				left_icon = root
				right_icon = scope:agot_greenseer_scry_target
			}
		}
		else_if = {
			limit = { exists = scope:agot_greenseer_scry_target }
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = agot_greenseer_scry_sensed_toast
				left_icon = root
				right_icon = scope:agot_greenseer_scry_target
			}
		}
		else = {
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = agot_greenseer_scry_nothing_toast
				left_icon = root
			}
		}
		agot_greenseer_track_rootedness_effect = yes
	}

	ai_potential = {
		has_perk = agot_greenseer_awakened_sight_perk
	}

	ai_will_do = {
		base = 10
	}
}
