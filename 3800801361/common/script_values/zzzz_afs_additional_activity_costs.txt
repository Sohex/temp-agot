coronation_cost = {
	add = {
		add = coronation_cost_additive
		if = {
			limit = {
				confederation ?= { has_cohesion_level_parameter = any_member_activity_cost_reduced }
			}
			add = {
				value = coronation_cost_additive
				multiply = -0.1
				desc = CB_COHESION_PARAMETER_DISCOUNT
			}
		}
		#Government types
		if = {
			limit = {
				government_has_flag = government_is_tribal
			}
			add = {
				value = coronation_cost_additive
				multiply = -0.34
				desc = coronation_tribal_reg
			}
		}
		else_if = {
			limit = {
				government_has_flag = government_is_nomadic
			}
			add = {
				value = coronation_cost_additive
				multiply = -0.9
				desc = coronation_nomadic_reg
			}
		}

		# players can not see AI cost, so let it be!
		if = {
			limit = { is_ai = yes }
			multiply = coronation_cost_factor_for_ai
		}
	}

if = { limit = { scope:province ?= { has_building = afs_andal_palace_01 agot_generic_is_special_building_enabled = yes } } multiply = { value = 0.8 desc = building_type_afs_andal_palace_01 } }
