# AGOT 0.5.2.1 compatibility: AFS seats are dragonpits in their own right.
has_more_than_one_dragonpit = {
	any_held_county = {
		OR = {
			any_county_province = {
				has_holding_type = castle_holding
				OR = { has_building_or_higher = generic_dragon_pit_01 OR = { has_building = afs_valyrian_spire_01 has_building = afs_wyrmrest_terraces_01 has_building = afs_cinderwing_aerie_01 } }
			}
			any_county_province = {
				OR = {
					has_building_or_higher = dragonpit_01 # Historical Version
					has_building = dragonpit_ruins_03 # Fully Restored Ruin Version
					has_building = agot_dragonmont_01 # Dragonmont Version
				}
			}
		}
		count > 1
	}
}
