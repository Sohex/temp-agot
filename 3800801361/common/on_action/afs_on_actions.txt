# Appended hooks, including the existing-save lobby entry. One persisted global watchdog chain.
on_game_start = { on_actions = { afs_bootstrap } }
on_game_start_after_lobby = { on_actions = { afs_bootstrap } }
afs_bootstrap = {
    effect = {
        every_player = { afs_check_founder_effect = yes afs_migrate_dragon_seats_effect = yes }
        afs_refresh_dragon_care_effect = yes
        if = {
            limit = { NOT = { exists = global_var:afs_watchdog_running } }
            set_global_variable = { name = afs_watchdog_running value = 1 }
            trigger_event = { on_action = afs_watchdog days = 1 }
        }
    }
}
afs_watchdog = {
    effect = {
        every_player = { afs_check_founder_effect = yes afs_migrate_dragon_seats_effect = yes }
        afs_refresh_dragon_care_effect = yes
        trigger_event = { on_action = afs_watchdog days = 1 }
    }
}
quarterly_playable_pulse = { on_actions = { afs_training_pulse } }
afs_training_pulse = { effect = { afs_migrate_dragon_seats_effect = yes afs_train_duelist_effect = yes afs_train_mariner_effect = yes } }
