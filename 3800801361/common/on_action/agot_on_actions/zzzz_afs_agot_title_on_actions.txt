# on_title_lost supplies the old holder as root, not scope:previous_holder.
on_title_lost = { on_actions = { afs_on_title_lost_dragonpit } }
afs_on_title_lost_dragonpit = {
 trigger = {
  scope:title.tier = tier_county
  scope:title = { any_county_province = { OR = { has_building = afs_valyrian_spire_01 has_building = afs_wyrmrest_terraces_01 has_building = afs_cinderwing_aerie_01 } } }
  is_alive = yes
  exists = var:current_dragon
  var:current_dragon = {
   has_character_flag = dragon_put_in_different_pit
   exists = var:pitted_dragon_location
   scope:title = var:pitted_dragon_location
  }
 }
 effect = {
  var:current_dragon = {
   save_scope_as = dragon
   agot_remove_from_dragonpit_skip_county_effects = { DRAGON = scope:dragon }
   return_to_court = yes
  }
 }
}
