# Character scope; AGOT stores a county title in pitted_dragon_location.
afs_update_dragon_care_effect = {
 if = {
  limit = {
   has_trait = dragon
   has_character_flag = in_dragonpit
   exists = var:pitted_dragon_location
   var:pitted_dragon_location ?= {
    agot_title_is_an_active_dragonpit = yes
    any_county_province = {
     has_building = afs_wyrmrest_terraces_01
     agot_generic_is_special_building_enabled = yes
    }
   }
  }
  if = { limit = { NOT = { has_character_modifier = afs_wyrmrest_dragon_care } } add_character_modifier = { modifier = afs_wyrmrest_dragon_care } }
 }
 else_if = { limit = { has_character_modifier = afs_wyrmrest_dragon_care } remove_character_modifier = afs_wyrmrest_dragon_care }
}
# Reuse the existing single daily watchdog; never add a second timer chain.
afs_refresh_dragon_care_effect = { every_living_dragon = { afs_update_dragon_care_effect = yes } }
