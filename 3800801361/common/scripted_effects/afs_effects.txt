afs_postpone_effect = {
 if = {
  limit = { has_character_flag = afs_pending NOT = { has_character_flag = afs_reminder_used } }
  if = { limit = { NOT = { has_variable = afs_cycle } } set_variable = { name = afs_cycle value = 0 } }
  add_character_flag = afs_reminder_used
  add_character_flag = afs_reminder_scheduled
  save_scope_value_as = { name = afs_reminder_cycle value = var:afs_cycle }
  trigger_event = { id = afs.0003 months = 3 }
 }
}
# Character scope. Shared by the player-only startup hook.
afs_schedule_effect = {
    if = {
        limit = {
            has_trait = afs_founders_ambition
            NOR = { has_character_flag = afs_scheduled has_character_flag = afs_pending }
        }
        if = { limit = { NOT = { has_variable = afs_cycle } } set_variable = { name = afs_cycle value = 0 } }
        change_variable = { name = afs_cycle add = 1 }
        remove_character_flag = afs_reminder_used
        remove_character_flag = afs_reminder_scheduled
        add_character_flag = afs_scheduled
        set_variable = { name = afs_schedule_guard value = 1 days = 12 }
        save_scope_value_as = { name = afs_start_cycle value = var:afs_cycle }
        trigger_event = { id = afs.0004 days = 10 }
    }
}
# Character scope. Prefer the capital, then a personally held castle.
afs_open_selection_effect = {
    if = {
        limit = { has_character_flag = afs_pending  }

    if = {
        limit = { afs_has_eligible_site = yes }
        if = {
            limit = { capital_province = { afs_eligible_site = yes } }
            capital_province = { save_scope_as = afs_site }
        }
        else = {
            random_held_title = {
                limit = {
                    tier = tier_barony
                    title_province = { afs_eligible_site = yes }
                }
                title_province = { save_scope_as = afs_site }
            }
        }
        afs_reset_picker_effect = yes
        if = { limit = { any_held_title = { count >= 2 tier = tier_barony title_province = { afs_eligible_site = yes } } } trigger_event = afs.0150 }
        else = { trigger_event = afs.0100 }
    }
    else = { trigger_event = afs.0400 }
    }
}
# Called only on confirmation; check ownership and the slot again.
afs_build_seat_effect = {
    if = {
        limit = {
            has_character_flag = afs_pending
            
            exists = scope:afs_site
            $FOUNDER_RULE$ = yes
            scope:afs_site = {
                afs_eligible_site = yes
                OR = { always = $ALLOW_INLAND$ is_coastal = yes }
                $SITE_RULE$ = yes
            }
        }
        scope:afs_site = {
            add_special_building_slot = $BUILDING$
            add_special_building = $BUILDING$
            afs_install_native_dragonpit_effect = yes
        }
        remove_character_flag = afs_paid_receipt
        add_character_flag = afs_completed
        remove_character_flag = afs_pending
    }
    else = { afs_open_selection_effect = yes }
}

# Province scope. Legacy effect ID retained for saves; no ordinary dragonpit is added.
afs_install_native_dragonpit_effect = {
 if = { limit = { has_holding_type = castle_holding OR = { has_building = afs_valyrian_spire_01 has_building = afs_wyrmrest_terraces_01 has_building = afs_cinderwing_aerie_01 } }
  # Preserve a separately constructed native pit when first integrating an unmarked seat.
  if = { limit = { NOT = { has_variable = afs_native_pit_installed } has_building = generic_dragon_pit_01 } set_variable = { name = afs_preserve_native_pit value = 1 } }
  # Collapse the old two-building representation only in a marked legacy AFS seat.
  # Native pits in other holdings, and unmarked independently built pits, remain intact.
  if = { limit = { has_variable = afs_native_pit_installed NOT = { has_variable = afs_preserve_native_pit } has_building = generic_dragon_pit_01 }
   remove_building = generic_dragon_pit_01
  }
  set_variable = { name = afs_native_pit_installed value = 1 }
  afs_initialize_native_dragonpit_effect = yes
 }
}
